1//! Engine-constraint verification: the user-visible DSL contract for each Tensor-Unit engine.
2//!
3//! Unlike the lowering algorithms in the crate root (thin FFI wrappers over the hidden
4//! `furiosa-opt-lower-impl`), these are plain, published functions. They state *what* mapping a stage
5//! must produce, in pure `furiosa_mapping` terms. Both the frontend (`furiosa-opt-std`, whose
6//! type-level `verify_*` delegate here) and the IR translator (`npu-visa-translate`) call them, so a
7//! kernel is checked identically whether it runs under an interpreting backend or is compiled.
8//!
9//! Each check returns a typed, per-engine error enum carrying the offending mappings; both callers
10//! render it (`panic!` in the frontend, a spanned diagnostic in the translator).
11//!
12//! When a check needs a hidden algorithm (e.g. exact division), it calls the FFI wrapper in the crate
13//! root rather than reimplementing it, keeping the algorithm private while the constraint stays visible.
14//!
15//! The submodule layout follows `furiosa-opt-std/src/engine`: each verification lives under the matching
16//! engine subtree. The vector engine's five checks are consolidated into one `vector/tensor.rs` rather
17//! than split as they are in the frontend.

34/// Bits in a byte.
35pub const BITS_PER_BYTE: usize = 8;
36/// Size of a single flit in bytes; the switching network moves data in flit-sized units.
37pub const FLIT_BYTES: usize = 32;
38/// Columns of the temporal accumulator (the packet reducer's output-width bound).
39pub const TEMPORAL_ACCUMULATOR_COLS: usize = 32;
40/// Elements of the lane-folder / packet-reducer output packet (one flit of `i32`/`f32`).
41pub(crate) const CONTRACT_LANE_OUT_PACKET_ELEMENTS: usize = 8;

43/// Byte size of `length` elements of an `element_bits`-wide type. Assumes the total bit count is
44/// byte-aligned; sub-byte types must supply a length that fills whole bytes.
45pub(crate) fn size_in_bytes(element_bits: usize, length: usize) -> usize {
46    debug_assert!(
47        (length * element_bits).is_multiple_of(BITS_PER_BYTE),
48        "size_in_bytes: {length} x {element_bits} bits is not byte-aligned"
49    );
50    length * element_bits / BITS_PER_BYTE
51}
52
53/// Element count that fills `bytes` bytes of an `element_bits`-wide type.
54pub(crate) fn length_from_bytes(element_bits: usize, bytes: usize) -> usize {
55    bytes * BITS_PER_BYTE / element_bits
56}

58/// `a` rounded up to a multiple of `b`.
59pub(crate) fn align_up(a: usize, b: usize) -> usize {
60    a.div_ceil(b) * b
61}

63/// The hardware supports 1, 2, 4, or 8 lanes.
64pub(crate) fn is_valid_lane_size(size: usize) -> bool {
65    matches!(size, 1 | 2 | 4 | 8)
66}

68/// Padded extent of each axis at its cumulative stride. On a normalized `Mapping` each axis carries
69/// its padded extent as `size()`, and the running stride is the product of the inner axes' extents;
70/// the contraction verifies key into this by a division term's `dividend_stride`.
71pub(crate) fn padding_per_stride(m: &Mapping) -> BTreeMap<usize, usize> {
72    let mut extents = Vec::new();
73    collect_axis_extents(&m.normalize(), &mut extents);
74    let mut map = BTreeMap::new();
75    let mut stride = 1;
76    for extent in extents {
77        map.insert(stride, extent);
78        stride *= extent;
79    }
80    map
81}
82
83/// Padded extent of a division term's axis: the `padding_per_stride` entry at the term's
84/// `dividend_stride`, falling back to the term's own `resize` when that stride is absent.
85pub(crate) fn padded_extent_at(padding_per_stride: &BTreeMap<usize, usize>, term: &DivideTerm) -> usize {
86    padding_per_stride
87        .get(&term.dividend_stride)
88        .copied()
89        .unwrap_or(term.resize)
90}
91
92/// Pushes each axis's padded extent (`size()`), innermost (right) first, down the normalized `Pair`
93/// spine. Relies on `normalize` fully decomposing composite leaves into the `Pair` spine so each
94/// non-`Pair` node is a single axis whose stride key matches a division term's `dividend_stride`.
95fn collect_axis_extents(m: &Mapping, out: &mut Vec<usize>) {
96    match m {
97        Mapping::Pair { left, right } => {
98            collect_axis_extents(right, out);
99            collect_axis_extents(left, out);
100        }
101        other => out.push(other.size()),
102    }
