7/// Elements in a one-flit vector packet (8 x 32-bit lanes).
8const ONE_FLIT_ELEMENTS: usize = 8;
9/// Elements in a half-flit vector packet (the front-4 lanes, and the partial-reduction width).
10const HALF_FLIT_ELEMENTS: usize = 4;
11/// Inner `Time` cells folded into the packet by Way4 concat.
12const WAY4_TIME_INNER: usize = 2;
13
14/// Why a vector packet reshape or intra-slice reduce is not realizable on the Vector engine.
15#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
16pub enum VectorError {
17    /// `vector_narrow_split` input packet is not one flit.
18    #[error("Split requires Packet of 8 elements (one flit).")]
19    SplitRequiresOneFlit,
20    /// `vector_narrow_split` time factor mismatch.
21    #[error("Vector_Split time factor mismatch. Expected: {expected}, got: {got}")]
22    SplitTimeMismatch {
23        /// Expected time.
24        expected: Mapping,
25        /// Declared output time.
26        got: Mapping,
27    },
28    /// `vector_narrow_split` output packet is not the front half of a flit.
29    #[error("Vector_Split output Packet2 must have 4 elements (front half of flit), got: {0}")]
30    SplitOutputSize(usize),
31    /// `vector_narrow_split` packet mismatch.
32    #[error("Vector_Split packet mismatch. Expected: {expected}, got: {got}")]
33    SplitPacketMismatch {
34        /// Expected packet.
35        expected: Mapping,
36        /// Declared output packet.
37        got: Mapping,
38    },
39    /// `vector_widen_concat` input packet is not Way4.
40    #[error("Concat requires Packet of 4 elements (Way4 mode).")]
41    ConcatRequiresWay4,
42    /// `vector_widen_concat` output packet is not one flit.
43    #[error("Vector_Concat output Packet2 must have 8 elements (one flit), got: {0}")]
44    ConcatOutputSize(usize),
45    /// `vector_widen_concat` time factor mismatch.
46    #[error("Vector_Concat time factor mismatch. Expected: {expected}, got: {got}")]
47    ConcatTimeMismatch {
48        /// Expected time.
49        expected: Mapping,
50        /// Declared output time.
51        got: Mapping,
52    },
53    /// `vector_widen_concat` packet mismatch.
54    #[error("Vector_Concat packet mismatch. Expected: {expected}, got: {got}")]
55    ConcatPacketMismatch {
56        /// Expected packet.
57        expected: Mapping,
58        /// Declared output packet.
59        got: Mapping,
60    },
61    /// `vector_narrow_trim` input packet is not one flit.
62    #[error(
63        "vector_narrow_trim: input Packet must have 8 elements (one flit), got {0}. \
64         vector_narrow_trim strips the back-4 dummy lanes before float operations; \
65         if Packet is already 4, you don't need it."
66    )]
67    TrimInputSize(usize),
68    /// `vector_narrow_trim` back-4 lanes are not dummy padding.
69    #[error(
70        "vector_narrow_trim: the back 4 lanes of the packet must be dummy (padding), but got: {0}. \
71         If they contain real data, use vector_narrow_split() instead."
72    )]
73    TrimBackNotDummy(Mapping),
74    /// `vector_narrow_trim` output packet is not the front 4.
75    #[error("vector_narrow_trim: output Packet2 must have 4 elements, got {0}.")]
76    TrimOutputSize(usize),
77    /// `vector_narrow_trim` output packet does not match the front 4 of the input.
78    #[error("vector_narrow_trim: Packet2 must match the front 4 of Packet. Expected: {expected}, got: {got}.")]
79    TrimPacketMismatch {
80        /// Expected packet (front 4 of input).
81        expected: Mapping,
82        /// Declared output packet.
83        got: Mapping,
84    },
85    /// `vector_widen_pad` input packet is not the trimmed half.
86    #[error("vector_widen_pad: input Packet must have 4 elements (after vector_narrow_trim), got {0}.")]
87    PadInputSize(usize),
88    /// `vector_widen_pad` output packet is not one flit.
89    #[error("vector_widen_pad: output Packet2 must have 8 elements (one flit), got {0}.")]
90    PadOutputSize(usize),
91    /// `vector_widen_pad` output packet is not the input padded to one flit.
92    #[error("vector_widen_pad: Packet2 must be Packet padded to 8. Expected: {expected}, got: {got}.")]
93    PadPacketMismatch {
94        /// Expected packet (input padded to 8).
95        expected: Mapping,
96        /// Declared output packet.
97        got: Mapping,
98    },
99    /// Intra-slice reduce: output does not divide input.
100    #[error("[Intra-slice reduce] divide failed: output shape must divide input shape: {0}")]
101    ReduceIndivisible(String),
102    /// Intra-slice reduce: a reduced axis does not carry `reduce_label`.
103    #[error(
104        "IntraSliceReduce: all reduced axes must match the specified reduce_label {reduce_label}, got quotient {quotient}"
105    )]
106    ReduceLabelMismatch {
107        /// The required reduce label.
108        reduce_label: Ident,
109        /// The quotient (the reduced axes).
110        quotient: Mapping,
111    },
112    /// Intra-slice reduce: a `reduce_label` axis survives into the retained division terms.
113    #[error(
114        "IntraSliceReduce: all the reduce axes must be fully reduced, but reduce_label {reduce_label} is still present in a retained axis"
115    )]
116    ReduceAxisNotFullyReduced {
117        /// The reduce label found in a retained axis.
118        reduce_label: Ident,
119    },
120    /// Intra-slice reduce: packet is neither preserved nor reduced to 4.
121    #[error(
122        "IntraSliceReduce: Packet should be either preserved or reduced to 4 (for partial reduction), got Packet {packet} -> OutPacket {out_packet}"
123    )]
124    ReducePacketMismatch {
125        /// The input packet.
126        packet: Mapping,
127        /// The declared output packet.
128        out_packet: Mapping,
129    },
130}

132/// `vector_narrow_split`: split a one-flit (8-element) packet, folding the front 4 into `Packet2` and
133/// the back 4 onto `Time2`.
134pub fn config_vector_narrow_split(
135    time: &Mapping,
136    packet: &Mapping,
137    time2: &Mapping,
138    packet2: &Mapping,
139) -> Result<(), VectorError> {
140    if packet.size() != ONE_FLIT_ELEMENTS {
141        return Err(VectorError::SplitRequiresOneFlit);
142    }
143    let (packet_outer, packet_inner) = packet.split_at(HALF_FLIT_ELEMENTS);
144    let expected_time = time.clone().pair(packet_outer).normalize();
145    let expected_packet = packet_inner.normalize();
146
147    let out_time = time2.normalize();
148    if expected_time != out_time {
149        return Err(VectorError::SplitTimeMismatch {
150            expected: expected_time,
151            got: out_time,
152        });
153    }
154    if packet2.size() != HALF_FLIT_ELEMENTS {
155        return Err(VectorError::SplitOutputSize(packet2.size()));
156    }
157    let out_packet = packet2.normalize();
158    if expected_packet != out_packet {
159        return Err(VectorError::SplitPacketMismatch {
160            expected: expected_packet,
161            got: out_packet,
162        });
163    }
164    Ok(())
165}

167/// `vector_widen_concat`: concatenate a 4-element packet with the inner 2 `Time` cells into a one-flit
168/// (8-element) `Packet2` (Way4 mode).
169pub fn config_vector_widen_concat(
170    time: &Mapping,
171    packet: &Mapping,
172    time2: &Mapping,
173    packet2: &Mapping,
174) -> Result<(), VectorError> {
175    if packet.size() != HALF_FLIT_ELEMENTS {
176        return Err(VectorError::ConcatRequiresWay4);
177    }
178    let (time_outer, time_inner) = time.split_at(WAY4_TIME_INNER);
179    let expected_time = time_outer.normalize();
180    let expected_packet = time_inner.pair(packet.clone()).normalize();
181
182    let out_time = time2.normalize();
183    let out_packet = packet2.normalize();
184    if packet2.size() != ONE_FLIT_ELEMENTS {
185        return Err(VectorError::ConcatOutputSize(packet2.size()));
186    }
187    if expected_time != out_time {
188        return Err(VectorError::ConcatTimeMismatch {
189            expected: expected_time,
190            got: out_time,
191        });
192    }
193    if expected_packet != out_packet {
194        return Err(VectorError::ConcatPacketMismatch {
195            expected: expected_packet,
196            got: out_packet,
197        });
198    }
199    Ok(())
200}

202/// `vector_narrow_trim`: strip the back-4 dummy lanes of a one-flit packet, keeping the front 4.
203pub fn config_vector_narrow_trim(packet: &Mapping, packet2: &Mapping) -> Result<(), VectorError> {
204    if packet.size() != ONE_FLIT_ELEMENTS {
205        return Err(VectorError::TrimInputSize(packet.size()));
206    }
207    let (packet_outer, packet_inner) = packet.split_at(HALF_FLIT_ELEMENTS);
208    // The back 4 must be dummy padding (`[1 # 2]`); otherwise use vector_narrow_split.
209    if packet_outer.normalize() != <m![1 # 2]>::to_value().normalize() {
210        return Err(VectorError::TrimBackNotDummy(packet_outer));
211    }
212    if packet2.size() != HALF_FLIT_ELEMENTS {
213        return Err(VectorError::TrimOutputSize(packet2.size()));
214    }
215    if packet_inner.normalize() != packet2.normalize() {
216        return Err(VectorError::TrimPacketMismatch {
217            expected: packet_inner,
218            got: packet2.clone(),
219        });
220    }
221    Ok(())
222}

224/// `vector_widen_pad`: restore the back-4 dummy lanes stripped by `vector_narrow_trim` (4 -> 8).
225pub fn config_vector_widen_pad(packet: &Mapping, packet2: &Mapping) -> Result<(), VectorError> {
226    if packet.size() != HALF_FLIT_ELEMENTS {
227        return Err(VectorError::PadInputSize(packet.size()));
228    }
229    if packet2.size() != ONE_FLIT_ELEMENTS {
230        return Err(VectorError::PadOutputSize(packet2.size()));
231    }
232    let expected = packet.clone().padding(ONE_FLIT_ELEMENTS, PaddingKind::Top).normalize();
233    if expected != packet2.normalize() {
234        return Err(VectorError::PadPacketMismatch {
235            expected,
236            got: packet2.clone(),
237        });
238    }
239    Ok(())
240}

242/// Intra-slice reduce: `[OutTime, OutPacket]` must divide `[Time, Packet]` exactly, every reduced axis
243/// must carry `reduce_label`, and the packet is preserved or reduced to 4.
244pub fn config_reduce_label(
245    time: &Mapping,
246    packet: &Mapping,
247    out_time: &Mapping,
248    out_packet: &Mapping,
249    reduce_label: &Ident,
250) -> Result<(), VectorError> {
251    let input = time.clone().pair(packet.clone());
252    let output = out_time.clone().pair(out_packet.clone());
253
254    let division_terms = crate::config_divide_exact(&input, &output).map_err(VectorError::ReduceIndivisible)?;
255
256    let quotient = crate::config_divide_relaxed(&input, &output).dividend_residue;
257    if !quotient.idents().iter().all(|ident| ident == reduce_label) {
258        return Err(VectorError::ReduceLabelMismatch {
259            reduce_label: *reduce_label,
260            quotient,
261        });
262    }
263    if !division_terms
264        .iter()
265        .all(|d| d.idents.iter().all(|ident| ident != reduce_label))
266    {
267        return Err(VectorError::ReduceAxisNotFullyReduced {
268            reduce_label: *reduce_label,
269        });
270    }
271
272    let packet = packet.normalize();
273    let out_packet = out_packet.normalize();
274    if packet != out_packet && out_packet != <m![1 # 4]>::to_value().normalize() {
275        return Err(VectorError::ReducePacketMismatch { packet, out_packet });
276    }
277    Ok(())
278}
