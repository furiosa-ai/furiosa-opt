1//! Collect engine: normalize the packet to exactly one flit, and store to the TRF (`to_trf`).
2
3use furiosa_mapping::{Mapping, MappingExt, PaddingKind};
4
5use crate::verify::{FLIT_BYTES, is_valid_lane_size, length_from_bytes, size_in_bytes};
6
7/// Why a collect is not realizable on the Collect engine.
8#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
9pub enum CollectError {
10    /// The output packet is not exactly one flit.
11    #[error("Collect output packet must be exactly {FLIT_BYTES} bytes (one flit).")]
12    OutputNotOneFlit,
13    /// The output packet is not the inner flit of the padded input.
14    #[error("Collect packet mismatch. Expected: {expected}, got: {got}")]
15    PacketMismatch {
16        /// The inner flit of the padded input packet.
17        expected: Mapping,
18        /// The declared output packet.
19        got: Mapping,
20    },
21    /// The output time is not the input time folded with the outer flit portion.
22    #[error("Collect time mismatch. Expected: {expected}, got: {got}")]
23    TimeMismatch {
24        /// The input time folded with the outer flit portion.
25        expected: Mapping,
26        /// The declared output time.
27        got: Mapping,
28    },
29}
30
31/// Why a `to_trf` is not realizable.
32#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
33pub enum ToTrfError {
34    /// `Lane` is not 1, 2, 4, or 8.
35    #[error("Lane::SIZE must be 1, 2, 4, or 8, got {0}")]
36    LaneSize(usize),
37    /// The TRF data does not fit the register file.
38    #[error(
39        "TRF data ({total_bytes} bytes = {lanes} lanes x {per_lane_bytes} bytes) exceeds register file capacity ({capacity} bytes)"
40    )]
41    ExceedsCapacity {
42        /// Total byte size across all lanes.
43        total_bytes: usize,
44        /// Lane count.
45        lanes: usize,
46        /// Byte size of one lane's element.
47        per_lane_bytes: usize,
48        /// Register file capacity in bytes.
49        capacity: usize,
50    },
51    /// `Lane::SIZE` does not divide `Time::SIZE`.
52    #[error("Lane::SIZE ({lane}) does not divide Time::SIZE ({time})")]
53    LaneDoesNotDivideTime {
54        /// Lane size.
55        lane: usize,
56        /// Time size.
57        time: usize,
58    },
59    /// The outer factors of `Time` do not equal `Lane`.
60    #[error("`to_trf` lane mismatch: time_outer != Lane: {time_outer} != {lane}")]
61    LaneMismatch {
62        /// Outer factors of `Time`.
63        time_outer: Mapping,
64        /// The declared `Lane`.
65        lane: Mapping,
66    },
67    /// The inner factors of `Time` concatenated with `Packet` do not equal `Element`.
68    #[error("`to_trf` element mismatch: [time_inner, Packet] != Element: {expected} != {got}")]
69    ElementMismatch {
70        /// `[time_inner, Packet]`.
71        expected: Mapping,
72        /// The declared `Element`.
73        got: Mapping,
74    },
75}

77/// Pads the input packet to a flit-aligned boundary, then splits at the flit: the inner flit must be
78/// `packet2`, and the outer portion folded onto `time` must be `time2`.
79pub fn config_collect(
80    time: &Mapping,
81    packet: &Mapping,
82    time2: &Mapping,
83    packet2: &Mapping,
84    element_bits: usize,
85) -> Result<(), CollectError> {
86    let in_packet_bytes = size_in_bytes(element_bits, packet.size());
87    let aligned_bytes = in_packet_bytes.div_ceil(FLIT_BYTES) * FLIT_BYTES;
88    let flit_elements = length_from_bytes(element_bits, FLIT_BYTES);
89
90    let out_packet_bytes = size_in_bytes(element_bits, packet2.size());
91    if out_packet_bytes != FLIT_BYTES {
92        return Err(CollectError::OutputNotOneFlit);
93    }
94
95    let padded = packet
96        .clone()
97        .padding(length_from_bytes(element_bits, aligned_bytes), PaddingKind::Top);
98    let (in_outer, in_flit) = padded.split_at(flit_elements);
99
100    let expected_packet = in_flit.normalize();
101    let out_packet = packet2.normalize();
102    if expected_packet != out_packet {
103        return Err(CollectError::PacketMismatch {
104            expected: expected_packet,
105            got: out_packet,
106        });
107    }
108
109    let expected_time = time.clone().pair(in_outer).normalize();
110    let out_time = time2.normalize();
111    if expected_time != out_time {
112        return Err(CollectError::TimeMismatch {
113            expected: expected_time,
114            got: out_time,
115        });
116    }
117    Ok(())
118}

120/// `to_trf`: reshape the collected `[Time, Packet]` into the TRF `[Lane, Element]`.
121///
122/// `Lane` must be 1/2/4/8 and fit `capacity` bytes; the outer factors of `Time` must equal `Lane`,
123/// and the remaining inner factors concatenated with `Packet` must equal `Element`.
124pub fn config_to_trf(
125    lane: &Mapping,
126    time: &Mapping,
127    packet: &Mapping,
128    element: &Mapping,
129    capacity: usize,
130    element_bits: usize,
131) -> Result<(), ToTrfError> {
132    let lane_size = lane.size();
133    if !is_valid_lane_size(lane_size) {
134        return Err(ToTrfError::LaneSize(lane_size));
135    }
136
137    let total_trf_bytes = size_in_bytes(element_bits, lane_size * element.size());
138    if total_trf_bytes > capacity {
139        return Err(ToTrfError::ExceedsCapacity {
140            total_bytes: total_trf_bytes,
141            lanes: lane_size,
142            per_lane_bytes: size_in_bytes(element_bits, element.size()),
143            capacity,
144        });
145    }
146
147    let time_size = time.size();
148    if !time_size.is_multiple_of(lane_size) {
149        return Err(ToTrfError::LaneDoesNotDivideTime {
150            lane: lane_size,
151            time: time_size,
152        });
153    }
154    let (time_outer, time_inner) = time.split_at(time_size / lane_size);
155    let time_outer = time_outer.normalize();
156    let lane_n = lane.normalize();
157    if time_outer != lane_n {
158        return Err(ToTrfError::LaneMismatch {
159            time_outer,
160            lane: lane_n,
161        });
162    }
163
164    let expected_element = time_inner.pair(packet.clone()).normalize();
165    let element_n = element.normalize();
166    if expected_element != element_n {
167        return Err(ToTrfError::ElementMismatch {
168            expected: expected_element,
169            got: element_n,
170        });
171    }
172    Ok(())
173}
