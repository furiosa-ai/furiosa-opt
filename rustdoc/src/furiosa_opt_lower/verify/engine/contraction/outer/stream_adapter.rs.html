1//! Stream adapter of `contract_outer`: pack the innermost `Time` cells into `OutPacket` and broadcast.
2
3use furiosa_mapping::{Mapping, MappingExt};
4
5use crate::verify::{FLIT_BYTES, is_valid_lane_size, length_from_bytes, size_in_bytes};
6
7/// Why a stream adapter is not realizable on `contract_outer`.
8#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
9pub enum StreamAdapterError {
10    /// `Lane` is not 1, 2, 4, or 8.
11    #[error("Lane::SIZE must be 1, 2, 4, or 8, got {0}")]
12    LaneSize(usize),
13    /// `OutPacket` is not one or two flits.
14    #[error("OutPacket must be 1 or 2 flits ({FLIT_BYTES} bytes each), got {0} bytes")]
15    OutPacketFlits(usize),
16    /// The inner flit of `OutPacket` does not equal the input `Packet`.
17    #[error("`contract_outer`: inner flit of OutPacket must equal the input Packet")]
18    FlitMismatch,
19    /// The packed cells of `OutPacket` do not equal the innermost `Time` cells.
20    #[error("`contract_outer`: OutPacket's packed cells must equal the innermost {0} cells of Time")]
21    PackedCellsMismatch(usize),
22    /// `Time` does not divide `OutTime`.
23    #[error("`contract_outer`: Time does not divide OutTime")]
24    TimeIndivisible,
25    /// `OutTime` is not the outer `Time` with broadcast axes innermost.
26    #[error("`contract_outer`: OutTime must be the outer Time with broadcast (tiling) axes innermost")]
27    OutTimeMismatch,
28}
29
30/// `OutPacket` = [packed innermost `Time` cells, input `Packet` flit]; `OutTime` = outer `Time` with
31/// broadcast (tiling) axes innermost. `Lane` must be 1/2/4/8 and `OutPacket` one or two flits.
32pub fn config_stream_adapter(
33    lane: &Mapping,
34    time: &Mapping,
35    packet: &Mapping,
36    out_time: &Mapping,
37    out_packet: &Mapping,
38    element_bits: usize,
39) -> Result<(), StreamAdapterError> {
40    if !is_valid_lane_size(lane.size()) {
41        return Err(StreamAdapterError::LaneSize(lane.size()));
42    }
43
44    let out_packet_bytes = size_in_bytes(element_bits, out_packet.size());
45    if !out_packet_bytes.is_multiple_of(FLIT_BYTES) || ![1, 2].contains(&(out_packet_bytes / FLIT_BYTES)) {
46        return Err(StreamAdapterError::OutPacketFlits(out_packet_bytes));
47    }
48
49    // Packing pulls the innermost `pack_size` cells of `Time` into the packet.
50    let pack_size = out_packet_bytes / FLIT_BYTES;
51    let (time_outer, time_packed) = time.split_at(pack_size);
52
53    // `OutPacket = [packed cells of Time, inner flit]`; the inner flit is the input `Packet`.
54    let flit_elements = length_from_bytes(element_bits, FLIT_BYTES);
55    let (out_packet_packed, out_packet_flit) = out_packet.split_at(flit_elements);
56    if out_packet_flit.normalize() != packet.normalize() {
57        return Err(StreamAdapterError::FlitMismatch);
58    }
59    if out_packet_packed.normalize() != time_packed.normalize() {
60        return Err(StreamAdapterError::PackedCellsMismatch(pack_size));
61    }
62
63    // `OutTime = [outer Time, broadcast]`; stripping the tiling axes off the bottom leaves outer `Time`.
64    if !out_time.size().is_multiple_of(time_outer.size()) {
65        return Err(StreamAdapterError::TimeIndivisible);
66    }
67    let tiling_size = out_time.size() / time_outer.size();
68    let (out_time_outer, _broadcast) = out_time.split_at(tiling_size);
69    if out_time_outer.normalize() != time_outer.normalize() {
70        return Err(StreamAdapterError::OutTimeMismatch);
71    }
72    Ok(())
73}
