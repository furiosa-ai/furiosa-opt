1//! Furiosa-opt lowering engines.
2//!
3//! Thin safe wrappers over the prebuilt `furiosa-opt-lower-impl` static library (linked by this
4//! crate's build script). The resolved results cross the `extern "C-unwind"` boundary as
5//! [`furiosa_opt_lower_types`] StableAbi values; each wrapper converts back to a plain `Result`. The
6//! frontend (`furiosa-opt-std`) and the IR backend (`tu-ops`) both call these.
7
8use abi_stable::std_types::{RResult, RString};
9use furiosa_mapping::Mapping;
10
11use abi_stable::std_types::RVec;
12pub use furiosa_opt_lower_types::{
13    COMMIT_VALID_PACKET_SIZES, CommitError, DivideTerm, FETCH_ALIGN_BYTES, FactorLeaf, FetchError,
14    MAX_SEQUENCER_ENTRIES, RelaxedDivision, StreamSequencerConfig, SwitchConfig, TransposeConfig,
15};
16

68/// Resolve transpose-engine hardware parameters, or the rendered error.
69pub fn config_transpose(
70    time: &Mapping,
71    packet: &Mapping,
72    out_time: &Mapping,
73    out_packet: &Mapping,
74    element_bits: usize,
75) -> Result<TransposeConfig, String> {
76    unsafe { sys::config_transpose(time, packet, out_time, out_packet, element_bits) }
77        .into_result()
78        .map_err(String::from)
79}
80
81/// Synthesize the fetch read descriptors, or the [`FetchError`].
82pub fn config_fetch(
83    in_time: &Mapping,
84    in_packet: &Mapping,
85    out_time: &Mapping,
86    out_packet: &Mapping,
87) -> Result<StreamSequencerConfig, FetchError> {
88    unsafe { sys::config_fetch(in_time, in_packet, out_time, out_packet) }.into_result()
89}
90
91/// Synthesize the commit write descriptors, or the [`CommitError`].
92pub fn config_commit(
93    in_time: &Mapping,
94    in_packet: &Mapping,
95    element: &Mapping,
96    element_bits: usize,
97) -> Result<StreamSequencerConfig, CommitError> {
98    unsafe { sys::config_commit(in_time, in_packet, element, element_bits) }.into_result()
99}
100
101/// Carve `divisor` out of `dividend` exactly; the matched axes, or a rendered error.
102pub fn config_divide_exact(dividend: &Mapping, divisor: &Mapping) -> Result<Vec<DivideTerm>, String> {
103    unsafe { sys::config_divide_exact(dividend, divisor) }
104        .into_result()
105        .map(Vec::from)
106        .map_err(String::from)
107}
108
109/// Relaxed carve: matched axes, both residues (`Mapping`), and the contiguous tail.
110pub fn config_divide_relaxed(dividend: &Mapping, divisor: &Mapping) -> RelaxedDivision {
111    unsafe { sys::config_divide_relaxed(dividend, divisor) }
112}

114/// `Mapping::factor_leaves()`: the factor decomposition, innermost-first — each leaf a named axis or an
115/// untagged composite/padding run.
116pub trait FactorLeavesExt {
117    /// The factor leaves of this mapping, innermost-first.
118    fn factor_leaves(&self) -> Vec<FactorLeaf>;
119}
120
121impl FactorLeavesExt for Mapping {
122    fn factor_leaves(&self) -> Vec<FactorLeaf> {
123        unsafe { sys::mapping_factor_leaves(self) }.into()
124    }
125}

127/// Validate a switch topology against the slice/time shapes, or the rendered error.
128pub fn config_switch(
129    config: &SwitchConfig,
130    in_slice: &Mapping,
131    in_time: &Mapping,
132    out_slice: &Mapping,
133    out_time: &Mapping,
134) -> Result<SwitchConfig, String> {
135    unsafe { sys::config_switch(config, in_slice, in_time, out_slice, out_time) }

137        .map_err(String::from)

140/// Validate a `tile` view (`index` divides `element` into `expected`), or the rendered error.
