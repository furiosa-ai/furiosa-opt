8/// Device function that transposes a tensor from shape [A, B] to [B, A].
9#[device(chip = 1)]
10pub fn tile_simple(ctx: &mut Context, input: HbmTensorView<'_, i8, m![1], m![A, B]>) -> HbmTensor<i8, m![1], m![B, A]> {
11    let mut output = unsafe { HbmTensor::<i8, m![1], m![B, A]>::from_addr(0x3000) };
12    for b in 0..32 {
13        // TODO: replace with <m![B]>::SIZE
14        let input_slice = input.tile::<m![B], 1, m![A, 1 # 32]>(b);
15        let output_slice = output.view_mut().tile::<m![B], 1, m![1 #{!} 32, A]>(b);
16        input_slice.to_hbm_view(&mut ctx.tdma, output_slice);
17    }
18    output
19}

21/// Reads a DM tensor column-by-column where each column is selected by a *computed* tile offset
22/// `b = g * GROUP + j` rather than a bare loop variable, returning an identity copy of the input.
23/// Splits the `B = 32` axis into `GROUPS = 2` groups of `GROUP = 16`; for each `(g, j)` it reads
24/// `input[.., b]` (the load-bearing computed-offset tile) into a loop-local scratch sink.
25///
26/// The tiled view is **DM (SRAM)**, so the `for g { for j { .. } }` nest is unrolled at
27/// `visa -> lir` and the computed offset's loop-variable leaves are substituted and folded to the
28/// concrete column constant. (An HBM-tiled loop is *not* unrolled — it lowers as a kept VISA `Loop`
29/// whose offset must be a bare scalar variable — so a computed offset must tile an SRAM view.)
30///
31/// Exercises computed tile offsets through the vISA-to-LIR unroller. The matching end-to-end test
32/// remains ignored because of a downstream mutable-view write-back limitation documented there.
33#[device(chip = 1)]
34pub fn tile_computed_offset(
35    ctx: &mut Context,
36    input_hbm: &HbmTensor<i8, m![1], m![A, B]>,
37) -> HbmTensor<i8, m![1], m![A, B]> {
38    const GROUPS: usize = 2;
39    const GROUP: usize = 16;
40    let input = input_hbm.to_dm::<m![A / 256], m![A % 256], m![B]>(&mut ctx.tdma);
41    let mut output = unsafe { DmTensor::<i8, m![1], m![A / 256], m![A % 256], m![B]>::from_addr(0x3000) };
42
43    // Define the whole output up front (identity copy), so the result equals the input.
44    input.view().to_dm_view(&mut ctx.tdma, output.view_mut());
45
46    // Read each column at the *computed* offset `b = g * GROUP + j` into a loop-local scratch tile.
47    // This is the read-side computed-offset pattern (a Q-head select `qh = kv_head * GQA_GROUP + g`):
48    // the GROUPS x GROUP iterations sweep every column, so the loop nest is SRAM-tiled and unrolled
49    // and the computed offset folds. The read is the load-bearing exercise; `scratch` (allocated
50    // inside the loop body, so the write target is in scope) is a per-iteration sink.
51    for g in 0..GROUPS {
52        for j in 0..GROUP {
53            let b = g * GROUP + j;
54            let mut scratch = unsafe { DmTensor::<i8, m![1], m![A / 256], m![A % 256], m![1 # 32]>::from_addr(0x6000) };
55            let input_col = input.view().tile::<m![B], 1, m![B = 1 # 32]>(b);
56            input_col.to_dm_view(&mut ctx.tdma, scratch.view_mut());
57        }
58    }
59
60    output.to_hbm(&mut ctx.tdma)
61}

67/// Commits a fetched 32-wide window into the *upper* half of a 64-wide DM via `commit_view` into a
68/// `view_mut().tile()`. The tile's out-of-tile cells are down (Bottom) padding, so the commit
69/// sequencer places the `/ 8` strided time within the live prefix and leaves them unwritten rather
70/// than rejecting the write. Copies `input[0..32]` into `result[32..64]`. Regression for a
71/// `view_mut().tile()` commit into a windowed-and-down-padded DM that surfaced as
72/// `StreamUnmatchedSegment`.
73#[device(chip = 1)]
74pub fn tile_window_commit(ctx: &mut Context, input: &HbmTensor<f32, Chip, m![D]>) -> HbmTensor<f32, Chip, m![D]> {
75    let tensor: DmTensor<f32, Chip, Cluster, Slice, m![D]> = input.to_dm(&mut ctx.tdma);
76    let tile_one = tensor.view().tile::<m![D], 32, m![D = 32 # 64]>(0);
77
78    let mut result: DmTensor<f32, Chip, Cluster, Slice, m![D]> = unsafe { DmTensor::from_addr(1 << 12) };
79
80    ctx.main
81        .begin(tile_one)
82        .fetch::<m![1], m![D = 32]>()
83        .fetch_cast::<f32>()
84        .collect::<m![D = 32 / 8], m![D = 32 % 8]>()
85        .commit_trim::<m![D = 32 % 8]>()
86        .commit_view(result.view_mut().tile::<m![D], 32, m![D = 32 #{!} 64]>(32));
87
88    result.to_hbm(&mut ctx.tdma)
89}

91/// Same transpose as [`tile_simple`], but the input is aliased with a bare `view()`
92/// inside the loop before tiling. The bare view resolves directly to the outer
93/// param tensor, which the loop-body builder must find among the parent's tensors;
94/// without that it fails VISA -> LIR with "no tensor exists for T{n}".
95#[device(chip = 1)]
96pub fn tile_view_in_loop(ctx: &mut Context, input: &HbmTensor<i8, m![1], m![A, B]>) -> HbmTensor<i8, m![1], m![B, A]> {
97    let mut output = unsafe { HbmTensor::<i8, m![1], m![B, A]>::from_addr(0x3000) };
98    for b in 0..32 {
99        let input_slice = input.view().tile::<m![B], 1, m![A, 1 # 32]>(b);
100        let output_slice = output.view_mut().tile::<m![B], 1, m![1 #{!} 32, A]>(b);
101        input_slice.to_hbm_view(&mut ctx.tdma, output_slice);
102    }
103    output
104}

106/// Identity copy through a two-level HBM-tiled loop nest whose *inner* body tiles a view
107/// produced by the *outer* loop variable -- the loop-body capture shape of the attention-decode
108/// kernels. The outer `for g` slices `input` into a 16-wide `B`-group (`in_group`, tiled by the
109/// outer index `g`); the inner `for h` tiles *that group* by the inner index, so the inner loop
110/// body reads a tensor whose `IndexAccess` `table` traces back through `in_group` to the OUTER
111/// loop's index tensor -- an index allocated in the enclosing scope, absent from the inner body's
112/// own tensors.
113///
114/// An HBM-tiled loop is not unrolled (contrast the SRAM nest in [`tile_computed_offset`]), so both
115/// `for`s survive as kept VISA `Loop`s. `convert_loop` must therefore (1) thread the outer index
116/// into the inner loop's `captured_inputs` -- else VISA -> LIR hits the loop-body-graph verify wall
117/// (`Loop::build_body_graph -> verify_structural_validity`, "no tensor exists for T{n}") -- and
118/// (2) keep the inner nest's own locals OUT of the outer loop's `local_tensors` -- else the LIR
119/// loop-body completeness check rejects them ("contained in local_tensors but not referenced by any
120/// local instructions"). It is the in-repo regression for both facets of that fix; the matching
121/// `test_tile_view_in_nested_loop` is a lowering-only check (`build_lir`, not `compare_lir!`) because
122/// the host backend's tile-shape validator does not accept this two-level HBM tile.
123#[device(chip = 1)]
124pub fn tile_view_in_nested_loop(
125    ctx: &mut Context,
126    input: &HbmTensor<i8, m![1], m![A, B]>,
127) -> HbmTensor<i8, m![1], m![A, B]> {
128    let mut output = unsafe { HbmTensor::<i8, m![1], m![A, B]>::from_addr(0x3000) };
129    for g in 0..2 {
130        // Outer-index tile: a 16-wide `B`-group selected by the enclosing loop var `g`. Its
131        // `tile` takes `&self`, so the inner loop can re-slice it every iteration; the index
132        // operand it carries (`g`) is allocated in the enclosing loop scope.
133        let in_group = input.view().tile::<m![B / 16], 1, m![A, 1 # 2, B % 16]>(g);
134        for h in 0..16 {
135            // Inner-index tile of the outer-index view: the read tensor's index operand (`h`)
136            // is inner-local, but its `table` traces back through `in_group` to the outer `g`.
137            let in_col = in_group.tile::<m![B % 16], 1, m![A, 1 # 32]>(h);
138            // The write side re-slices `output` by the same outer group index `g` and inner
139            // column index `h`; the outer `g` view is rebuilt here (the mutable `tile` consumes
140            // its receiver) but still references the enclosing `g` index tensor.
141            let out_col = output
142                .view_mut()
143                .tile::<m![B / 16], 1, m![A, 1 # 2, B % 16]>(g)
144                .tile::<m![B % 16], 1, m![A, 1 #{!} 32]>(h);
145            in_col.to_hbm_view(&mut ctx.tdma, out_col);
146        }
147    }
148    output
149}

151/// Chunked-output write created *inside* a loop and returned *after* it: the exact
152/// T1 pattern that VISA -> LIR lowering must keep visible past the loop op.
153///
154/// Copies `input[B, D]` to `output[B, D]` one `D`-chunk per iteration. Each
155/// `output.view_mut().tile::<m![D / 16], 1, m![B, 1 #{!} 4, D % 16]>(i)` is a
156/// chunked write tile whose strided `D_16` chunk-index axis is absent from the
157/// contiguous output shape, so lowering must materialize it (`split_chunk_index_axis`).
158/// The write tile is a loop-body-local tensor whose writeback resolves against the
159/// returned `output` *after* the loop closes; dropping loop-body locals from the parent
160/// regresses this to a "no tensor T{n}" panic. Together this is a single runnable value
161/// oracle (`compare_lir!`) for both the loop-tensor re-merge and the chunk-axis fix.
162#[device(chip = 1)]
163pub fn tile_chunked_output(
164    ctx: &mut Context,
165    input: &HbmTensor<i8, m![1], m![B, D]>,
166) -> HbmTensor<i8, m![1], m![B, D]> {
167    let mut output = unsafe { HbmTensor::<i8, m![1], m![B, D]>::from_addr(0x3000) };
168    for i in 0..4 {
169        let input_slice = input.view().tile::<m![D / 16], 1, m![B, 1 # 4, D % 16]>(i);
170        let output_slice = output.view_mut().tile::<m![D / 16], 1, m![B, 1 #{!} 4, D % 16]>(i);
171        input_slice.to_hbm_view(&mut ctx.tdma, output_slice);
172    }
173    output
174}

220/// Swaps the disjoint 256-wide `L`-chunks `[0, 256)` and `[256, 512)` via a tile window > 1,
221/// copying the trailing chunk `[512, 768)` through unchanged.
222///
223/// `start` for a tile window > 1 is the raw `L`-axis offset of the window, not a chunk index scaled
224/// by the window length -- confirmed by the working `tile_window_commit` example, where window `32`
225/// at `start = 32` lands on `result[32..64]`, not some multiple of `32`. Every `start` here (`0`,
226/// `256`, `512`) is already the exact axis offset of a chunk boundary, so the three windows are
227/// disjoint and cover the whole `L` axis exactly once (unlike `tile_with_larger_than_one_1`, whose
228/// `start` values `0`/`1`/`2` are small overlapping raw offsets that reproduce PROG-539 and leave
229/// most of the output tensor unwritten).
230#[device(chip = 1)]
231pub fn tile_size_gt1_chunk_swap(
232    ctx: &mut Context,
233    input: &HbmTensor<bf16, Chip, m![L, A]>,
234) -> HbmTensor<bf16, Chip, m![L, A]> {
235    let mut output = unsafe { HbmTensor::<bf16, Chip, m![L, A]>::from_addr(0x3000) };
236
237    let input_slice = input.view().tile::<m![L], 256, m![L = 256 # 768, A]>(0);
238    let output_slice = output.view_mut().tile::<m![L], 256, m![L = 256 #{!} 768, A]>(256);
239    input_slice.to_hbm_view(&mut ctx.tdma, output_slice);
240
241    let input_slice = input.view().tile::<m![L], 256, m![L = 256 # 768, A]>(256);
242    let output_slice = output.view_mut().tile::<m![L], 256, m![L = 256 #{!} 768, A]>(0);
243    input_slice.to_hbm_view(&mut ctx.tdma, output_slice);
244
245    let input_slice = input.view().tile::<m![L], 256, m![L = 256 # 768, A]>(512);
246    let output_slice = output.view_mut().tile::<m![L], 256, m![L = 256 #{!} 768, A]>(512);
247    input_slice.to_hbm_view(&mut ctx.tdma, output_slice);
248
249    output
