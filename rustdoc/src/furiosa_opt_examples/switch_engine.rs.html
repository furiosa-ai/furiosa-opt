1//! Switch Engine examples: each `#[device]` fn lowers a `SwitchConfig::CustomBroadcast`
2//! and is checked bit-exact (VISA vs LIR) in `npu-visa-test`.
3//!
4//! Axis scoping: `Chip`/`Cluster` and the `(B, V)` time/packet axes below are shared by
5//! every example. Every other axis is declared in its own `axes!` block directly above
6//! the single example that uses it, so each block's domain is exactly that example.
7
8use furiosa_opt_std::prelude::*;
9
10type Chip = m![1];
11type Cluster = m![1 # 2];
12// Time (`B`) and packet (`V`) axes, common to every example below.
13axes![B = 8, V = 16];
14
15// ── custom_broadcast (ring 4) ───────────────────────────────────────────────────────
16// The padding slot `1#4` becomes a fresh broadcast tile `Y`; slice volume 64*4 = 256.
17axes![A = 64, Y = 4];
18type Slice = m![A, 1 # 4];
19type OutSlice = m![A, Y];
20
21#[device(chip = 1)]
22pub fn custom_broadcast(
23    ctx: &mut Context,
24    input: &HbmTensor<bf16, Chip, m![A, B, V]>,
25) -> HbmTensor<bf16, Chip, m![A, Y, B, V]> {
26    let dm: DmTensor<bf16, Chip, Cluster, Slice, m![B, V]> = input.to_dm::<Cluster, Slice, m![B, V]>(&mut ctx.tdma);
27
28    let result: DmTensor<bf16, Chip, Cluster, OutSlice, m![B, V]> = ctx
29        .main
30        .begin(dm.view())
31        .fetch::<m![B], m![V]>()
32        .switch::<OutSlice, m![B]>(SwitchConfig::CustomBroadcast { ring_size: 4 })
33        .collect::<m![B], m![V]>()
34        .commit_trim::<m![V]>()
35        .commit();
36
37    result.to_hbm::<m![A, Y, B, V]>(&mut ctx.tdma)
38}

113// ── custom_broadcast_moved_axis (multi-bit bitmap) ──────────────────────────────────
114// Unlike the broadcasts above (a padding slot becomes a tile, nothing moves slice to time, so
115// every output snoops a single input lane), here a live slice axis `Q` moves to output time and
116// a fresh tile `Yt` fills its slot. Each output slice then reads, over the `Q` time steps, the
117// whole `Q`-block of input lanes: a multi-bit snoop bitmap. The compare_lir test checks the
118// resulting data permutation, while the bitmap bits are pinned by the sequencer-vs-enumeration
119// cross-check in `custom_snoop_bitmap` (the LIR executor computes from shapes, not the bitmap).
120axes![P = 64, Q = 4, Yt = 4];
121
122#[device(chip = 1)]
123pub fn custom_broadcast_moved_axis(
124    ctx: &mut Context,
125    input: &HbmTensor<bf16, Chip, m![P, Q, B, V]>,
126) -> HbmTensor<bf16, Chip, m![P, Yt, B, Q, V]> {
127    let dm: DmTensor<bf16, Chip, Cluster, m![P, Q], m![B, V]> =
128        input.to_dm::<Cluster, m![P, Q], m![B, V]>(&mut ctx.tdma);
129
130    let result: DmTensor<bf16, Chip, Cluster, m![P, Yt], m![B, Q, V]> = ctx
131        .main
132        .begin(dm.view())
133        .fetch::<m![B], m![V]>()
134        .switch::<m![P, Yt], m![B, Q]>(SwitchConfig::CustomBroadcast { ring_size: 4 })
135        .collect::<m![B, Q], m![V]>()
136        .commit_trim::<m![V]>()
137        .commit();
138
139    result.to_hbm::<m![P, Yt, B, Q, V]>(&mut ctx.tdma)
140}

142// ── custom_broadcast_transpose (relocation) ─────────────────────────────────────────
143// A pure slice permutation (`[Pt, Qt]` to `[Qt, Pt]`, nothing broadcast or moved to time): each
144// output snoops a single input lane, but at a relocated position. The compare_lir test checks the
145// permuted data, while the relocated base index in the bitmap is pinned by the internal
146// sequencer-vs-enumeration cross-check.
147axes![Pt = 32, Qt = 8];
148
149#[device(chip = 1)]
150pub fn custom_broadcast_transpose(
151    ctx: &mut Context,

158        .main
159        .begin(dm.view())
