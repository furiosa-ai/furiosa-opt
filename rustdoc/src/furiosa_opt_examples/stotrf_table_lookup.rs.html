1//! `f4e2m1` (NVFP4 / MXFP4) weight GEMM via decode-then-stage.
2//!
3//! Table lookup is a main-context fetch stage, so a packed `f4e2m1` weight is decoded to `f8e4m3` in
4//! a main-context fetch (`fetch -> fetch_table_lookup::<f8e4m3>() -> collect -> commit`) that writes
5//! the decoded weight to DM, and a plain convert-only StoTrf then stages that `f8e4m3` weight into
6//! the contraction TRF. Decoding cannot happen inside the StoTrf fetch itself: the sub-context Fetch
7//! Unit has no lookup-table register, so table lookup is offered only on the main context (see
8//! `computing-tensors/fetch-adapter.md`).
9//!
10//! The kernel is a tokens-on-slices GEMM: an `f8e4m3` activation streams the contraction while the
11//! decoded `f8e4m3` weight is the weight-stationary filter, the f8 MAC (`Red % 64`) reduces, and the
12//! VE tail applies the per-output-channel dequant scale and rounds to `bf16`. Only the packed weight
13//! read plus its decode pass (¼ the weight DMA volume) differs from a native f8 weight GEMM.
14
15use furiosa_opt_std::prelude::*;
16
17axes![Tok = 16, Red = 64, Out = 16];
18
19type Chip = m![1];
20// Tokens on the slices: `Tok / 8 % 2` cluster, `Tok % 8` slice padded to 256.
21type GemmCl = m![Tok / 8 % 2];
22type TokSlice = m![Tok % 8 # 256];
23
24/// `[Tok, Red] x [Out, Red]^T -> [Tok, Out]` tokens-on-slices GEMM with the `[Out, Red]` weight
25/// PACKED `f4e2m1`, decoded to `f8e4m3` by a main-context table-lookup fetch that commits to DM,
26/// then staged into the TRF by a plain convert-only StoTrf and contracted against the `f8e4m3`
27/// `[Tok, Red]` activation. Each output is `scale[o] * sum_red act[tok, red] * decode(weight[o, red])`,
28/// accumulated in `f32` and rounded to `bf16`.
29#[device(chip = 1)]
30pub fn stotrf_table_lookup_gemm(
31    ctx: &mut Context,
32    act: &HbmTensor<f8e4m3, Chip, m![Tok, Red]>,
33    weight: &HbmTensor<f4e2m1, Chip, m![Out, Red]>,
34    w_scale: &HbmTensor<bf16, Chip, m![Out]>,
35) -> HbmTensor<bf16, Chip, m![Tok, Out]> {
36    const {
37        assert!(
38            Red::SIZE % 64 == 0,
39            "Red must be a multiple of the 64-element f8 MAC width"
40        );
41        assert!(
42            Out::SIZE % 8 == 0,
43            "output width must be a multiple of the 8-lane output group"
44        );
45        assert!(Tok::SIZE <= 16, "Tok must map onto 2 clusters x 8 slices (<= 16)");
46    }
47
48    // Stream operand: the e4m3 activation, tokens on the slices, `Red` in the element.
49    let act_dm: DmTensor<f8e4m3, Chip, GemmCl, TokSlice, m![Red]> = act.to_dm(&mut ctx.tdma);
50
51    // Resident token-major output DM (tokens on slices, `Out` contiguous in the element).
52    const OUT_DM_ADDR: Address = 0x0040_0000;
53    let mut out_dm: DmTensor<bf16, Chip, GemmCl, TokSlice, m![Out]> = DmTensor::new();
54
55    // The full `[Out]` per-output-channel dequant scale, broadcast across the token-slices.
56    let scale_all_dm: DmTensor<bf16, Chip, GemmCl, TokSlice, m![Out]> = w_scale.to_dm(&mut ctx.tdma);
57
58    for g in 0..(Out::SIZE / 8) {
59        // This group's 8 packed f4e2m1 weight rows, broadcast across the token-slices.
60        let weight_group = weight
61            .view()
62            .tile::<m![Out / 8], 1, m![1 # { Out::SIZE / 8 }, Out % 8, Red]>(g);
63        let weight_dm: DmTensor<f4e2m1, Chip, GemmCl, TokSlice, m![Out % 8, Red]> = weight_group.to_dm(&mut ctx.tdma);
64
65        // Table lookup is a main-context fetch stage, so the packed f4e2m1 weight is decoded to
66        // f8e4m3 in a main-context fetch that commits the decoded stream to a resident DM. The fetch
67        // reads the packed f4 (2 codes/byte) and `fetch_table_lookup::<f8e4m3>()` decodes each 4-bit
68        // code to its e4m3 value. The decoded weight lands in a resident DM (fixed address, not a
69        // transient `commit()` scratch) so the sub-context StoTrf below can read it back.
70        const WEIGHT_F8_DM_ADDR: Address = 0x0060_0000;
71        let mut weight_f8_dm: DmTensor<f8e4m3, Chip, GemmCl, TokSlice, m![Out % 8, Red]> = DmTensor::new();
72        ctx.main
73            .begin(weight_dm.view())
74            .fetch::<m![Out % 8], m![Red]>()
75            .fetch_table_lookup::<f8e4m3>()
76            .collect::<m![Out % 8, Red / 32], m![Red % 32]>()
77            .commit_trim::<m![Red % 32]>()
78            .commit_view(weight_f8_dm.view_mut());
79
80        // Stage the decoded f8e4m3 weight into the TRF as the weight-stationary filter (a plain
81        // convert-only StoTrf, `f8e4m3 -> f8e4m3`, no table lookup on the sub-context fetch).
82        let weight_trf: TrfTensor<f8e4m3, Chip, GemmCl, TokSlice, m![Out % 8], m![Red]> = ctx
83            .sub
84            .begin(weight_f8_dm.view())
85            .fetch::<m![Out % 8], m![Red]>()
86            .collect::<m![Out % 8, Red / 32], m![Red % 32]>()
87            .to_trf();
88
89        // The group's 8 per-output-channel dequant scales, broadcast across the token-slices.
90        let scale_group = scale_all_dm
91            .view()
92            .tile::<m![Out / 8], 1, m![1 # { Out::SIZE / 8 }, Out % 8]>(g);
93        let scale_vrf: VrfTensor<f32, Chip, GemmCl, TokSlice, m![Out % 8]> = ctx
94            .sub
95            .begin(scale_group)
96            .fetch::<m![1], m![Out % 8]>()
97            .fetch_cast::<f32>()
98            .collect::<m![1], m![Out % 8]>()
99            .to_vrf();
100
101        // Contract the streamed per-token e4m3 activation against the decoded 8-row f8 weight TRF
102        // (`mac_width(f8) = 64`, so the contraction OutPacket is `Red % 64`), fold the 8 output
103        // features Interleaved into the `OutPacket`, then dequant in the VE tail and round to bf16.
104        ctx.main
105            .begin(act_dm.view())
106            .fetch::<m![Red / 32], m![Red % 32]>()
107            .collect::<m![Red / 32], m![Red % 32]>()
108            .contract_outer::<m![Red / 64], m![Red % 64], _, _, _>(&weight_trf)
109            .contract_packet::<m![1]>()
110            .contract_time::<m![1]>()
111            .contract_lane::<m![1], m![Out % 8]>(LaneMode::Interleaved)
112            .vector_init()
113            .vector_intra_slice_tag(TagMode::Zero)
114            .vector_narrow_split::<m![Out / 4 % 2], m![Out % 4]>()
115            .vector_fp_binary(FpBinaryOp::MulF(FpMulAlu::Mul0), &scale_vrf)
116            .vector_widen_concat::<m![1], m![Out % 8]>()
117            .vector_final()
118            .cast::<bf16, m![Out % 8 # 16]>()
119            .commit_trim::<m![Out % 8]>()
120            .commit_view(
121                out_dm
122                    .view_mut()
123                    .tile::<m![Out / 8], 1, m![1 #{!} { Out::SIZE / 8 }, Out % 8]>(g),
124            );
125    }
126
127    let mut out = unsafe { HbmTensor::<bf16, Chip, m![Tok, Out]>::from_addr(0x0080_0000) };
128    out_dm.view().to_hbm_view(&mut ctx.tdma, out.view_mut());
129    out
130}
