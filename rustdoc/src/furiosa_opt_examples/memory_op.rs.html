1//! Examples exercising memory-movement primitives that lack dedicated examples
2//! elsewhere: DM↔DM relayout ([`DmTensor::to_dm`]), parallel-copy
3//! ([`DmTensor::to_dm_pcopy`] / `DmTensorView::to_dm_view_pcopy`), HBM chip
4//! shuffle ([`HbmTensorView::hbm_chip_shuffle`]), and a `commit_view` into a
5//! down-padded tiled `view_mut` ([`commit_view_bottom_pad`]).
6
7#![expect(clippy::type_complexity)]
8
9use furiosa_opt_std::prelude::*;
10
11axes![A = 256, B = 4096];
12axes![Q = 896, T = 512];

14/// `DmTensor::to_dm`: relayout a DM tensor's `Element` payload (DM → DM DMA copy, auto-placed).
15/// The `Slice` size is preserved (`m![A]` and `m![1 # 2, A / 2]` are both 256).
16#[device(chip = 1)]
17pub fn dm_relayout(
18    ctx: &mut Context,
19    hbm: &HbmTensor<i32, m![1], m![A, B]>,
20) -> HbmTensor<i32, m![1], m![A / 2, A % 2, B]> {
21    let dm: DmTensor<i32, m![1], m![1 # 2], m![A], m![B]> = hbm.to_dm(&mut ctx.tdma);
22    let relaid: DmTensor<i32, m![1], m![1 # 2], m![1 # 2, A / 2], m![A % 2, B]> = dm.to_dm(&mut ctx.tdma);
23    relaid.to_hbm(&mut ctx.tdma)
24}

26/// `DmTensor::to_dm_pcopy`: copy a DM tensor into another DM tensor via parallel
27/// copy (the convenience wrapper over `to_dm_view_pcopy`).
28#[device(chip = 1)]
29pub fn dm_pcopy(ctx: &mut Context, hbm: &HbmTensor<i32, m![1], m![A, B]>) -> HbmTensor<i32, m![1], m![A, B]> {
30    let src: DmTensor<i32, m![1], m![1], m![A], m![B]> = hbm.to_dm(&mut ctx.tdma);
31    let dst: DmTensor<i32, m![1], m![1], m![A], m![B]> = src.to_dm_pcopy(&mut ctx.sub);
32    dst.to_hbm(&mut ctx.tdma)
33}

35/// `DmTensorView::to_dm_view_pcopy`: copy from a DM view into a DM `view_mut` via
36/// parallel copy.
37#[device(chip = 1)]
38pub fn dm_view_pcopy(ctx: &mut Context, hbm: &HbmTensor<i32, m![1], m![A, B]>) -> HbmTensor<i32, m![1], m![A, B]> {
39    let src: DmTensor<i32, m![1], m![1], m![A], m![B]> = hbm.to_dm(&mut ctx.tdma);
40    let mut dst: DmTensor<i32, m![1], m![1], m![A], m![B]> = unsafe { DmTensor::from_addr(0x10_0000) };
41    src.view().to_dm_view_pcopy(&mut ctx.sub, dst.view_mut());
42    dst.to_hbm(&mut ctx.tdma)
43}

45/// `HbmTensorView::hbm_chip_shuffle`: redistribute data across chips on the HBM
46/// side via DMA.
47#[device(chip = 4)]
48pub fn hbm_chip_shuffle(
49    ctx: &mut Context,
50    hbm: &HbmTensor<i32, m![A / 4 % 4], m![A / 16, A % 4, B]>,
51) -> HbmTensor<i32, m![A / 4 % 4], m![A / 16, A % 4, B]> {
52    hbm.view()
53        .hbm_chip_shuffle::<4, { Dma::Tensor }>(&mut ctx.tdma, &[1, 2, 3, 0])
54}

56type QChip = m![1];
57type QCluster = m![1 # 2];
58type QSlice = m![T / 32 # 256];

60/// `commit_view` into a tiled `view_mut` whose tile is padded back up to the
61/// destination axis width as down padding (`#{!} 56`): an 8-wide chunk is
62/// committed into a 56-wide destination axis, the rest down padding, so the
63/// commit sequencer never writes the cells outside the chunk. Minimal
64/// fetch/commit pipeline exercising that `view_mut().tile()` fills with
65/// [`PaddingKind::Bottom`].
66#[device(chip = 1)]
67pub fn commit_view_bottom_pad(
68    ctx: &mut Context,
69    input_hbm: &HbmTensor<f32, QChip, m![T, Q % 56 = 8]>,
70) -> HbmTensor<f32, QChip, m![T, Q % 56]> {
71    let input: DmTensor<f32, QChip, QCluster, QSlice, m![T % 32, Q % 56 = 8]> = input_hbm.to_dm(&mut ctx.tdma);
72
73    let mut result: DmTensor<f32, QChip, QCluster, QSlice, m![T % 32, Q % 56]> =
74        unsafe { DmTensor::from_addr(0x200000) };
75
76    ctx.main
77        .begin(input.view())
78        .fetch::<m![T % 32], m![Q % 56 = 8]>()
79        .collect::<m![T % 32], m![Q % 56 = 8]>()
80        .commit_trim::<m![Q % 56 = 8]>()
81        .commit_view(
82            result
83                .view_mut()
84                .tile::<m![Q % 56], 8, m![T % 32, Q % 56 = 8 #{!} 56]>(0),
85        );
86
87    result.to_hbm(&mut ctx.tdma)
88}
