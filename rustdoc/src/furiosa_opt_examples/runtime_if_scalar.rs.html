1//! Runtime `if` producing a *scalar* result: the merged value is a `ScalarExpr::Ternary` (the scalar
2//! analogue of a tensor phi at a branch merge), which lowers to `SymExpr::Ternary`.
3//!
4//! A scalar `if` result is usable exactly where a *runtime* scalar is:
5//!
6//! - As a view/tile index it is fine ([`runtime_if_scalar_index`]): the index becomes an SPM value.
7//! - As a vector-engine immediate it is not ([`runtime_if_scalar_immediate`]): an immediate is
8//!   encoded in the instruction, so it must const-fold to a compile-time constant. That function
9//!   fails to compile with a diagnostic saying so; it is a negative example.
10
11use furiosa_opt_std::prelude::*;
12
13axes![A = 512, B = 32, N = 1024];
14type Chip = m![1];
15
16/// Runtime `if` selecting a tile INDEX. The tile offset `if i == 0 { 0 } else { 16 }` is a runtime
17/// scalar (a `Ternary` on the loop index), and a runtime scalar is a valid view index, so this
18/// compiles and runs. Iterations `0`/`1` copy the two 16-wide halves of `B`, together covering the
19/// whole row.
20///
21/// Oracle: `output == input`.
22#[device(chip = 1)]
23pub fn runtime_if_scalar_index(
24    ctx: &mut Context,
25    input_hbm: &HbmTensor<i8, Chip, m![A, B]>,
26) -> HbmTensor<i8, Chip, m![A, B]> {
27    let input = input_hbm.to_dm::<m![A / 256], m![A % 256], m![B]>(&mut ctx.tdma);
28    let mut output = DmTensor::<i8, Chip, m![A / 256], m![A % 256], m![B]>::new();
29    for i in 0..2 {
30        let off: usize = if i == 0 { 0 } else { 16 };
31        let src = input.view().tile::<m![B], 16, m![B = 16 # 32]>(off);
32        let dst = output.view_mut().tile::<m![B], 16, m![B = 16 #{!} 32]>(off);
33        src.to_dm_view(&mut ctx.tdma, dst);
34    }
35    output.to_hbm(&mut ctx.tdma)
36}
37
38type Row = DmTensor<i32, Chip, m![1], m![N / 4], m![N % 4]>;
39
40/// Runtime `if` selecting a vector-engine IMMEDIATE: `vector_fxp(AddFxp, if i == 1 { 1 } else { 2 })`.
41/// The immediate is encoded in the instruction, so it must const-fold to a compile-time constant; the
42/// `if` result is a runtime scalar that does not, so this fails to compile (a negative example).
43///
44/// Expected: `a vector-engine immediate must be a compile-time constant, but `...` does not const-fold`.
45///
46/// Runs at a 4-PE device (1 cluster) so the `Cluster = m![1]` `Row` is a valid allocation and the
47/// compile reaches the vector-engine-immediate check that is the point of this negative example.
48#[device(chip = 1, pe = 4)]
49pub fn runtime_if_scalar_immediate(
50    ctx: &mut Context,
51    input: &HbmTensor<i32, Chip, m![N]>,
52) -> HbmTensor<i32, Chip, m![N]> {
53    let input_dm: Row = input.to_dm(&mut ctx.tdma);
54    let mut result: Row = DmTensor::new();
55    for i in 0..2 {
56        let c: i32 = if i == 1 { 1 } else { 2 };
57        result = ctx
58            .main
59            .begin(input_dm.view())
60            .fetch::<m![1], m![N % 4]>()
61            .fetch_cast::<i32>()
62            .collect::<m![1], m![N % 4 # 8]>()
63            .vector_init()
64            .vector_intra_slice_tag(TagMode::Zero)
65            .vector_fxp(FxpBinaryOp::AddFxp, c)
66            .vector_final()
67            .commit_trim::<m![N % 4]>()
68            .commit();
69    }
70    result.to_hbm(&mut ctx.tdma)
71}
