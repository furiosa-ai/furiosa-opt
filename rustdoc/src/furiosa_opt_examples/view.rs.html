1pub mod simpl {
2    use furiosa_opt_std::prelude::*;
3
4    axes![A = 512, B = 8];
5
6    #[device(chip = 1)]
7    pub fn view_simpl(
8        ctx: &mut Context,
9        input_hbm: &HbmTensor<i32, m![1], m![A, B]>,
10    ) -> HbmTensor<i32, m![1], m![A, B]> {
11        // Transfer input tensor to DM with shape:
12        // - (A=512/256) on the cluster dimension,
13        // - (A=512%256) on the slice dimension,
14        // - (B=8) on the element dimension.
15        let input = input_hbm.to_dm::<m![A / 256], m![A % 256], m![B]>(&mut ctx.tdma);
16
17        // Allocate output tensor in DM with the same shape as input.
18        let mut output = unsafe { DmTensor::<i32, m![1], m![A / 256], m![A % 256], m![B]>::from_addr(0x3000) };
19
20        // Create views on input (B=0,1,2,3) and output (B=2,3,4,5), and copy data from input to output.
21        let input0123 = input.view().tile::<m![B], 4, m![B = 4 # 8]>(0);
22        let output2345 = output.view_mut().tile::<m![B], 4, m![B = 4 #{!} 8]>(2);
23        input0123.to_dm_view(&mut ctx.tdma, output2345);
24
25        // Create views on input (B=4,5) and output (B=6,7), and copy data from input to output.
26        let input45 = input.view().tile::<m![B], 2, m![B = 2 # 8]>(4);
27        let output67 = output.view_mut().tile::<m![B], 2, m![B = 2 #{!} 8]>(6);
28        input45.to_dm_view(&mut ctx.tdma, output67);
29
30        // Create views on input (B=6,7) and output (B=0,1), and copy data from input to output.
31        let input67 = input.view().tile::<m![B], 2, m![B = 2 # 8]>(6);
32        let output01 = output.view_mut().tile::<m![B], 2, m![B = 2 #{!} 8]>(0);
33        input67.to_dm_view(&mut ctx.tdma, output01);
34
35        // Transfer output tensor back to HBM.
36        output.to_hbm(&mut ctx.tdma)
37    }
38}

40pub mod nested {
41    use furiosa_opt_std::prelude::*;
42
43    axes![A = 512, B = 528];
44
45    #[device(chip = 1)]
46    pub fn view_nested(
47        ctx: &mut Context,
48        input_hbm: &HbmTensor<i32, m![1], m![A, B]>,
49    ) -> HbmTensor<i32, m![1], m![A, B]> {
50        let input = input_hbm.to_dm::<m![A / 256], m![A % 256], m![B]>(&mut ctx.tdma);
51        let mut output = unsafe { DmTensor::<i32, m![1], m![A / 256], m![A % 256], m![B]>::from_addr(0x3000) };
52
53        // Define the whole output first (full copy) so no position is left `Uninit`.
54        input.view().to_dm_view(&mut ctx.tdma, output.view_mut());
55
56        let in_nested = input
57            .view()
58            .tile::<m![B], 264, m![B = 264 # 528]>(0)
59            .tile::<m![B = 264 # 528], 132, m![B = 132 # 528]>(0);
60        let out_nested = output
61            .view_mut()
62            .tile::<m![B], 264, m![B = 264 # 528]>(0)
63            .tile::<m![B = 264 # 528], 132, m![B = 132 # 528]>(0);
64        in_nested.to_dm_view(&mut ctx.tdma, out_nested);
65
66        output.to_hbm(&mut ctx.tdma)
67    }
68}

70pub mod padding {
71    use furiosa_opt_std::prelude::*;
72
73    axes![A = 9, B = 7];
74
75    #[device(chip = 1, pe = 1)]
76    pub fn view_padding(
77        ctx: &mut Context,
78        input_hbm: &HbmTensor<i32, m![1], m![[A, B] # 1024]>,
79    ) -> HbmTensor<i32, m![1], m![[A, B] # 1024]> {
80        let input_hbm_transposed: HbmTensor<i32, m![1], m![[A, B] # 1024 % 8, [A, B] # 1024 / 8]> =
81            input_hbm.to_hbm::<_, _>(&mut ctx.tdma);
82        // Transfer to DM: `(A=9)(B=7)=63` real values padded to 1024 so the slice is a valid 64.
83        // - cluster: 1
84        // - slice: (A,B)#1024/16 = 64
85        // - element: (A,B)#1024%8 * (A,B)#1024/8%2 = 16
86        let input = input_hbm_transposed
87            .to_dm::<m![1], m![[A, B] # 1024 / 16], m![[A, B] # 1024 % 8, [A, B] # 1024 / 8 % 2]>(&mut ctx.tdma);
88
89        // Allocate output tensor in DM with the same shape as input.
90        let mut output = unsafe {
91            DmTensor::<i32, m![1], m![1], m![[A, B] # 1024 / 16], m![[A, B] # 1024 % 8, [A, B] # 1024 / 8 % 2]>::from_addr(
92                0x4000,
93            )
94        };
95
96        // Create views on input (((A=9)(B=7)#1024%4)=0,1,2) and output (((A=9)(B=7)#1024%4)=1,2,3), and copy data from input to output.
97        let input012 = input
98            .view()
99            .tile::<m![[A, B] # 1024 % 4], 3, m![[A, B] # 1024 / 4 % 2, ([A, B] # 1024 % 4) = 3 # 4, [A, B] # 1024 / 8 % 2]>(0);
100        let output123 = output
101            .view_mut()
102            .tile::<m![[A, B] # 1024 % 4], 3, m![[A, B] # 1024 / 4 % 2, ([A, B] # 1024 % 4) = 3 #{!} 4, [A, B] # 1024 / 8 % 2]>(
103                1,
104            );
105        input012.to_dm_view(&mut ctx.tdma, output123);
106
107        // Create views on input (((A=9)(B=7)#1024%4)=3) and output (((A=9)(B=7)#1024%4)=0), and copy data from input to output.
108        let input3 = input
109            .view()
110            .tile::<m![[A, B] # 1024 % 4], 1, m![[A, B] # 1024 / 4 % 2, 1 # 4, [A, B] # 1024 / 8 % 2]>(3);
111        let output0 = output
112            .view_mut()
113            .tile::<m![[A, B] # 1024 % 4], 1, m![[A, B] # 1024 / 4 % 2, 1 #{!} 4, [A, B] # 1024 / 8 % 2]>(0);
114        input3.to_dm_view(&mut ctx.tdma, output0);
115
116        // Transfer output tensor back to HBM.
117        output.to_hbm(&mut ctx.tdma)
118    }
