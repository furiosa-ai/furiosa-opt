1//! Kernels for the `DmTensorViewMut::memset` (ParallelMemSet / `itos`) fill op.
2//!
3//! Each kernel loads an HBM input into a 2-cluster DM region, overwrites the whole region with a
4//! typed fill `value` via `dm.view_mut().memset(..)`, relayouts it (a DM->DM copy that reads the
5//! filled region), then writes it back. The input is fully overwritten, so the readback must be the
6//! fill value regardless of input: the signal that ParallelMemSet fired. The relayout reading the
7//! just-filled region also pins the in-place fill aliasing (the consumer sees the fill, not the input).
8//!
9//! Covers every `Scalar` this branch translates to vISA: the byte+ RNGD scalars (`i8`, `i16`, `i32`,
10//! `f32`, `bf16`, `f8e4m3`) and sub-byte `i4`. A plain literal folds `i8`/`i16`/`i32`/`f32`; a
11//! `const { .. }` folds a computed value (`bf16`/`f8e4m3`/`i4`). `f4e2m1` is a follow-up (a valid DSL
12//! scalar, but its vISA `mir_ast::Scalar` variant lands with the separate fetch/table-lookup work), so
13//! a `memset(f4e2m1_value)` is rejected with a diagnostic rather than mis-lowered.
14//!
15//! The DM region spans `num_clusters_per_execution` clusters (`m![1 # 2]`) and uses the
16//! `memory_op::dm_relayout` shape (the 2-cluster round-trip `m![A, B]` -> `m![A / 2, A % 2, B]`); `B`
17//! is even, so sub-byte in-slice rows are byte-aligned.
18
19use furiosa_opt_std::prelude::*;
20
21axes![
22    A = 256, // partitioned rows (2 clusters * 128 slices)
23    B = 128  // in-slice payload per row
24];
25
26/// HBM input: a plain `m![A, B]` region.
27type In<D> = HbmTensor<D, m![1], m![A, B]>;
28/// HBM output: the 2-cluster relayout shape `m![A / 2, A % 2, B]` (`dm_relayout`'s output shape).
29type Out<D> = HbmTensor<D, m![1], m![A / 2, A % 2, B]>;
30/// 2-cluster DM region loaded straight from `In`.
31type Dm<D> = DmTensor<D, m![1], m![1 # 2], m![A], m![B]>;
32/// The relaid 2-cluster DM whose layout maps cleanly onto `Out`.
33type Relaid<D> = DmTensor<D, m![1], m![1 # 2], m![1 # 2, A / 2], m![A % 2, B]>;
34
35/// Fills `dm`, relayouts it (the relayout reads the fill), and writes it back. Shared by every
36/// per-type kernel below via its concrete `value`.
37macro_rules! memset_kernel {
38    ($name:ident, $d:ty, $value:expr) => {
39        #[device(chip = 1)]
40        pub fn $name(ctx: &mut Context, input: &In<$d>) -> Out<$d> {
41            let mut dm: Dm<$d> = input.to_dm(&mut ctx.tdma);
42            dm.view_mut().memset($value, &mut ctx.sub);
43            let relaid: Relaid<$d> = dm.to_dm(&mut ctx.tdma);
44            relaid.to_hbm(&mut ctx.tdma)
45        }
46    };
47}
48
49// `PadValue::Zero` path (a zero fill folds to all-zero element bits).
50memset_kernel!(memset_i32_zero, i32, 0);
51// Signed `PadValue::Custom` (low byte `0xff`).
52memset_kernel!(memset_i8_neg_one, i8, -1);
53// 2-byte `PadValue::Custom`.
54memset_kernel!(memset_i16_300, i16, 300);
55// 4-byte float `PadValue::Custom`.
56memset_kernel!(memset_f32_one_half, f32, 1.5);
57// `const { .. }`-folded `bf16` (a computed value needs the const block to fold).
58memset_kernel!(memset_bf16_one, bf16, const { bf16::from_f32(1.0) });
59// `const { .. }`-folded `f8e4m3`.
60memset_kernel!(memset_f8e4m3_one, f8e4m3, const { f8e4m3::from_f32(1.0) });
61// Sub-byte `i4`: the fill value `-1` is the nibble `0xf`, replicated to every packed code.
62memset_kernel!(memset_i4_neg_one, i4, const { i4::from_i32(-1) });
63// Explicit in-place aliasing pin (structurally identical: the relayout consumes the fill).
64memset_kernel!(memset_alias_bf16, bf16, const { bf16::from_f32(3.0) });

66/// Sub-view fill: `memset(7.0)` writes only the upper half of the in-slice `B` axis (the `B/2 = 64`
67/// tile at offset `64`); the lower half stays the loaded input.
68///
69/// Device (LIR) translation of a sub-view fill is **rejected** today (see `memset::lower`'s TODO and
70/// `snapshot.toml`, which pins this kernel to fail at the `visa` stage): the tile lowers to a padded
71/// in-slice whose window is one block, and `with_full_padding` would overrun it. The VISA-level
72/// (emulation) calculation is still correct, which the answer-key `test_memset_subview_bf16` pins.
73#[device(chip = 1)]
74pub fn memset_subview_bf16(ctx: &mut Context, input: &In<bf16>) -> Out<bf16> {
75    let mut dm: Dm<bf16> = input.to_dm(&mut ctx.tdma);
76    dm.view_mut()
77        .tile::<m![B], 64, m![B = 64 #{!} 128]>(64)
78        .memset(const { bf16::from_f32(7.0) }, &mut ctx.sub);
79    let relaid: Relaid<bf16> = dm.to_dm(&mut ctx.tdma);
80    relaid.to_hbm(&mut ctx.tdma)
81}
82
83/// Negative fixture: a computed fill passed WITHOUT `const { .. }` is rejected at translation
84/// (`snapshot.toml` pins it to fail at the `mir` stage). Without the const block, `bf16::from_f32(1.0)`
85/// is a runtime call in this position; it has no device-translatable body, so the inline pass reports
86/// it (guiding toward `const { .. }`) instead of the compiler panicking on the unlowered call.
87/// Compiles and runs on emulation; the device translate is what requires the const.
88#[device(chip = 1)]
89pub fn memset_missing_const_bf16(ctx: &mut Context, input: &In<bf16>) -> Out<bf16> {
90    let mut dm: Dm<bf16> = input.to_dm(&mut ctx.tdma);
91    dm.view_mut().memset(bf16::from_f32(1.0), &mut ctx.sub);
92    let relaid: Relaid<bf16> = dm.to_dm(&mut ctx.tdma);
93    relaid.to_hbm(&mut ctx.tdma)
94}
