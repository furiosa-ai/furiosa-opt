1//! Runtime branches over a non-unrolled loop index.
2//!
3//! The branch condition is an arbitrary runtime scalar: each example derives a `bool` variable from
4//! the loop index (`let take_then = i == 1;`) and branches on it, exercising the condition path end to
5//! end (comparison, `>`, a constant, a named `bool`).
6//!
7//! All but one example *use* the branch result: a live loop-carried accumulator read each iteration,
8//! or an intermediate whose value a later op consumes. Only [`runtime_if_value`] reassigns a result it
9//! never reads (a dead carry), kept as the single example of that shape.
10//!
11//! EDF status: a value-`if` (`result = if ..`, a tensor phi merge) lowers through VISA and runs on the
12//! simulator, but the scheduler cannot yet lower the merge to EDF (it fails `BeamSearch`), so
13//! [`runtime_if_value`], [`runtime_if_accumulate`], [`runtime_if_chain`], and [`runtime_if_const`] stop
14//! at the `visa` stage in the regression snapshot and their `compare_edf!` tests are `#[ignore]`d. Only
15//! the statement form [`runtime_if_two_outputs`] (each arm writes a distinct output, no value phi)
16//! compiles end to end to EDF.
17
18use furiosa_opt_std::prelude::*;
19
20// A 4-PE device is 1 cluster x 256 slices, so `Cluster = m![1]` matches. `W / 4` x `W % 4` gives
21// `Row` an 8-byte-aligned in-slice extent.
22axes![W = 1024];
23
24type Chip = m![1];
25type Cluster = m![1];
26type Row = DmTensor<i32, Chip, Cluster, m![W / 4], m![W % 4]>;
27
28/// The single example whose branch result is reassigned each iteration but never read (a dead carry).
29///
30/// `result = if take_then { .. } else { .. }` overwrites `result` without consuming its prior value,
31/// so the loop-carried initial arriving at each iteration is dead on arrival. The condition is a
32/// `bool` derived from the loop index. `for i in 0..2`, `i == 1` at the last iteration, so the
33/// surviving value is the then-arm.
34///
35/// Oracle: `output = input + 1`.
36#[device(chip = 1, pe = 4)]
37pub fn runtime_if_value(ctx: &mut Context, input: &HbmTensor<i32, Chip, m![W]>) -> HbmTensor<i32, Chip, m![W]> {
38    let input_dm: Row = input.to_dm(&mut ctx.tdma);
39    let mut result: Row = DmTensor::new();
40
41    for i in 0..2 {
42        let take_then = i == 1;
43        result = if take_then {
44            add_const(ctx, &input_dm, 1)
45        } else {
46            add_const(ctx, &input_dm, 2)
47        };
48    }
49
50    result.to_hbm(&mut ctx.tdma)
51}

53/// Branch result USED as a live loop-carried accumulator: each arm reads `result` and adds onto it, so
54/// the carry is genuinely consumed (unlike [`runtime_if_value`]). The condition is a `bool` from the
55/// index (`i > 0`).
56///
57/// `result` starts as `input`; `i == 0` (else) adds 2, `i == 1` (then, `i > 0`) adds 1.
58/// Oracle: `output = input + 3`.
59#[device(chip = 1, pe = 4)]
60pub fn runtime_if_accumulate(ctx: &mut Context, input: &HbmTensor<i32, Chip, m![W]>) -> HbmTensor<i32, Chip, m![W]> {
61    let mut result: Row = input.to_dm(&mut ctx.tdma);
62
63    for i in 0..2 {
64        let take_then = i > 0;
65        result = if take_then {
66            add_const(ctx, &result, 1)
67        } else {
68            add_const(ctx, &result, 2)
69        };
70    }
71
72    result.to_hbm(&mut ctx.tdma)
73}

75/// Branch result USED by a further op: the `if` selects an intermediate `mid` (reading the live carry
76/// `result`), then a second op consumes `mid` to update `result`. The condition is a `bool` from the
77/// index (`i == 1`).
78///
79/// `result` starts as `input`; each iteration `mid = result + (1 if take_then else 2)` then
80/// `result = mid + 10`. `i == 0` (else): `mid = input + 2`, `result = input + 12`. `i == 1` (then):
81/// `mid = input + 13`, `result = input + 23`.
82/// Oracle: `output = input + 23`.
83#[device(chip = 1, pe = 4)]
84pub fn runtime_if_chain(ctx: &mut Context, input: &HbmTensor<i32, Chip, m![W]>) -> HbmTensor<i32, Chip, m![W]> {
85    let mut result: Row = input.to_dm(&mut ctx.tdma);
86
87    for i in 0..2 {
88        let take_then = i == 1;
89        let mid = if take_then {
90            add_const(ctx, &result, 1)
91        } else {
92            add_const(ctx, &result, 2)
93        };
94        add_const_view_mut(ctx, &mid, 10, result.view_mut());
95    }
96
97    result.to_hbm(&mut ctx.tdma)
98}

100/// Statement-form runtime `if`: each arm writes into a *different* externally-defined output tensor's
101/// view, so both arms' results are used (as the two kernel outputs) rather than a reassigned carry.
102/// The condition is a `bool` from the index (`i == 0`).
103///
104/// `for i in 0..2`: `i == 0` takes the then-arm (`input + 1` into `out_then`), else `input + 2` into
105/// `out_else`. Oracle: `out_then = input + 1`, `out_else = input + 2`.
106#[device(chip = 1, pe = 4)]
107pub fn runtime_if_two_outputs(
108    ctx: &mut Context,
109    input: &HbmTensor<i32, Chip, m![W]>,
110) -> (HbmTensor<i32, Chip, m![W]>, HbmTensor<i32, Chip, m![W]>) {
111    let input_dm: Row = input.to_dm(&mut ctx.tdma);
112    let mut out_then: Row = DmTensor::new();
113    let mut out_else: Row = DmTensor::new();
114
115    for i in 0..2 {
116        if i == 0 {
117            add_const_view_mut(ctx, &input_dm, 1, out_then.view_mut());
118        } else {
119            add_const_view_mut(ctx, &input_dm, 2, out_else.view_mut());
120        }
121    }
122
123    (out_then.to_hbm(&mut ctx.tdma), out_else.to_hbm(&mut ctx.tdma))
124}

126/// Constant branch condition (`if true`): the branch limit folds to the constant `1`, so the then-arm
127/// is always taken. The result is used (returned) rather than reassigned across a loop.
128///
129/// Oracle: `output = input + 1`.
130#[device(chip = 1, pe = 4)]
131pub fn runtime_if_const(ctx: &mut Context, input: &HbmTensor<i32, Chip, m![W]>) -> HbmTensor<i32, Chip, m![W]> {
132    let mut result: Row = input.to_dm(&mut ctx.tdma);
133
134    result = if true {
135        add_const(ctx, &result, 1)
136    } else {
137        add_const(ctx, &result, 2)
138    };
139
140    result.to_hbm(&mut ctx.tdma)
141}

143/// Fetch `input`, add `c` on the vector engine, and commit a fresh `[W]` result.
144fn add_const(ctx: &mut Context, input_dm: &Row, c: i32) -> Row {
145    ctx.main
146        .begin(input_dm.view())
147        .fetch::<m![1], m![W % 4]>()
148        .fetch_cast::<i32>()
149        .collect::<m![1], m![W % 4 # 8]>()
150        .vector_init()
151        .vector_intra_slice_tag(TagMode::Zero)
152        .vector_fxp(FxpBinaryOp::AddFxp, c)
153        .vector_final()
154        .commit_trim::<m![W % 4]>()
155        .commit()
156}
157
158fn add_const_view_mut(
159    ctx: &mut Context,
160    input_dm: &Row,
161    c: i32,
162    output_view_mut: DmTensorViewMut<'_, i32, Chip, Cluster, m![W / 4], m![W % 4]>,
163) {
164    ctx.main
165        .begin(input_dm.view())
166        .fetch::<m![1], m![W % 4]>()
167        .fetch_cast::<i32>()
168        .collect::<m![1], m![W % 4 # 8]>()
169        .vector_init()
170        .vector_intra_slice_tag(TagMode::Zero)
171        .vector_fxp(FxpBinaryOp::AddFxp, c)
172        .vector_final()
173        .commit_trim::<m![W % 4]>()
174        .commit_view(output_view_mut)
175}
