1//! Fills an under-filled partition to the slice count with a non-target-`Tile` `Broadcast`.
2
3use furiosa_opt_std::prelude::*;
4
5// `Feat = 128` is the real partition axis; `Quad = 4` is a real inner (8 bf16 bytes).
6axes![Feat = 128, Quad = 4];
7
8type Chip = m![1];
9// `Feat`(128) * `Broadcast`(2) = 256, the slice count; the bare-`2` is the partition fill.
10type Slice = m![Feat, 2];
11type Cl = m![2];
12
13/// Stages `[Feat, Quad]` onto a 128-wide `Slice` and writes the 256-slice DM back.
14#[device(chip = 1)]
15pub fn bias_partition_broadcast(
16    ctx: &mut Context,
17    bias: &HbmTensor<bf16, Chip, m![Feat, Quad]>,
18) -> HbmTensor<bf16, Chip, m![Feat, 2, Quad]> {
19    // The output keeps the `2` axis, so the read-only fill needs no write-side collapse.
20    let bias_dm: DmTensor<bf16, Chip, Cl, Slice, m![Quad]> = bias.to_dm(&mut ctx.tdma);
21    bias_dm.to_hbm(&mut ctx.tdma)
22}
