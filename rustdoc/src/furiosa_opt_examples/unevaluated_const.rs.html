1//! Regression fixture for the `Const::Unevaluated` arm of `translate_const`
2//! (npu-visa-translate). The kernel body reads `<m![B]>::SIZE` as a *value*
3//! and guards it with a `const { assert!(..) }`. Both reach MIR as a
4//! monomorphizable anonymous/associated const, which MIR->vISA translation must
5//! const-eval rather than panic on. See furiosa-ai#18502.
6
7use furiosa_opt_std::prelude::*;
8
9axes![A = 512, B = 32];
10
11/// Transpose `[A, B]` to `[B, A]`, driving the per-row loop bound from
12/// `<m![B]>::SIZE` (read as a value) and asserting it in a `const { .. }` block.
13///
14/// `clippy::assertions_on_constants` fires because a `const { .. }` shape guard's
15/// condition is, by design, a compile-time constant; that is the exact idiom this
16/// fixture pins, so the lint is allowed locally.
17#[device(chip = 1)]
18#[allow(clippy::assertions_on_constants)]
19pub fn unevaluated_const(
20    ctx: &mut Context,
21    input: HbmTensorView<'_, i8, m![1], m![A, B]>,
22) -> HbmTensor<i8, m![1], m![B, A]> {
23    const { assert!(<m![B]>::SIZE % 32 == 0, "B must be a multiple of 32") };
24    let mut output = unsafe { HbmTensor::<i8, m![1], m![B, A]>::from_addr(0x3000) };
25    for b in 0..<m![B]>::SIZE {
26        let input_slice = input.tile::<m![B], 1, m![A, 1 # 32]>(b);
27        let output_slice = output.view_mut().tile::<m![B], 1, m![1 #{!} 32, A]>(b);
28        input_slice.to_hbm_view(&mut ctx.tdma, output_slice);
29    }
30    output
31}
