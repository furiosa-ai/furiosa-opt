1//! Device-function scalar-type coverage: one fixture per LIR-supported device scalar.
2//!
3//! Every scalar that `furiosa-opt-std` implements `Scalar` for and that has a matching LIR
4//! `ElementType` gets exactly one fixture here, so the per-type roster is visible at a glance and
5//! each type's mir -> vISA -> IR translation (`translate_ty_to_scalar` / `eval_scalar_type`) is
6//! pinned by a sibling test in `npu-visa-test/tests/scalar.rs`. Each fixture names its scalar in a
7//! tensor parameter, a DM staging, and the fetch cast, so it must translate through those two
8//! functions rather than be rejected as an "unsupported scalar primitive".
9//!
10//! Fixture shape per type:
11//! - **at-width identity relayout** for `f32` / `i32` / `i16` / `i8` / `u8`: the element threads
12//!   through the parameter, the DM staging, the identity fetch cast, and the transposed output.
13//! - **widen** for the narrowing floats: `bf16 -> f32`, `f8e4m3 -> f32`.
14//!
15//! `i4` is a supported LIR scalar (`eval_scalar_type` maps it), but has no fixture here: it is
16//! sub-byte (`RawInt4` packs two per byte), so a host-buffer execute-compare is infeasible and the
17//! `--device-function` build probe is the only available check. Omitted rather than carry a
18//! build-only outlier.
19//!
20//! The collect stage requires a packet of exactly one 32-byte flit in the post-cast element type,
21//! so each module sizes its innermost axis `B` to make `B * size_of::<dst>() == 32` (8 x 4B,
22//! 16 x 2B, or 32 x 1B). Each fixture lives in its own module (for a per-type `axes!`) but carries
23//! a unique leaf fn name, because the `--device-function` filter matches any path ending in
24//! `::<name>`.
25
26use furiosa_opt_std::prelude::*;
27
28type Chip = m![1];
29type Cluster = m![1 # 2];
30
31/// Emits one `src -> dst` fetch-cast fixture in its own module: the `src` element threads through
32/// the parameter and DM staging, the fetch cast maps `src -> dst`, and the `dst` result is stored
33/// transposed. An at-width identity relayout passes the same type for `$src` and `$dst`; a widen
34/// passes a narrower `$src` and a wider `$dst`. The innermost `B` is both the DMA tail
35/// (>= min_align = 8 bytes) and one collect flit (= 32 bytes in `$dst`), so the caller picks `$b`
36/// to satisfy `$b * size_of::<$dst>() == 32`.
37macro_rules! scalar_fixture {
38    ($module:ident, $fn:ident, $src:ty, $dst:ty, $b:literal) => {
39        pub mod $module {
40            use super::*;
41
42            axes![A = 4096, B = $b];
43
44            #[device(chip = 1)]
45            pub fn $fn(
46                ctx: &mut Context,
47                input: &HbmTensor<$src, m![1], m![A, B]>,
48            ) -> HbmTensor<$dst, m![1], m![B, A]> {
49                let input_dm = input.to_dm::<Cluster, m![A / 16], m![A / 8 % 2, A % 8, B]>(&mut ctx.tdma);
50
51                let result: DmTensor<$dst, Chip, Cluster, m![A / 16], m![A / 8 % 2, A % 8, B]> = ctx
52                    .main
53                    .begin(input_dm.view())
54                    .fetch::<m![A / 8 % 2], m![A % 8, B]>()
55                    .fetch_cast::<$dst>()
56                    .collect::<m![A / 8 % 2, A % 8], m![B]>()
57                    .commit_trim::<m![B]>()
58                    .commit();
59
60                result.to_hbm(&mut ctx.tdma)
61            }
62        }
63    };
64}
65
66// At-width identity relayout. f32 / i32: 8 x 4 bytes = 32-byte flit. i16: 16 x 2 bytes. i8 / u8: 32 x 1 byte.
67scalar_fixture!(f32_relayout, relayout_f32, f32, f32, 8);
68scalar_fixture!(i32_relayout, relayout_i32, i32, i32, 8);
69scalar_fixture!(i16_relayout, relayout_i16, i16, i16, 16);
70scalar_fixture!(i8_relayout, relayout_i8, i8, i8, 32);
71scalar_fixture!(u8_relayout, relayout_u8, u8, u8, 32);
72
73// Widen to a 4-byte target: 8 x 4 bytes = 32-byte flit (the narrow source's 8-element tail is
74// >= 8 bytes).
75scalar_fixture!(bf16_widen, widen_bf16, bf16, f32, 8);
76scalar_fixture!(f8e4m3_widen, widen_f8e4m3, f8e4m3, f32, 8);
