1//! Fetch-adapter table-lookup example kernels.
2//!
3//! Exercises the real `fetch_table_lookup` primitive on an `f4e2m1` (NVFP4 / MXFP4)
4//! weight stream: the fetch chain decodes each nibble to `f8e4m3` through the
5//! hardware paired-key table, then widens to `f32` with a following `fetch_cast`.
6//! See the book chapter `computing-tensors/fetch-adapter.md` ("Table Lookup").
7
8#![expect(clippy::type_complexity)]
9
10use furiosa_opt_std::prelude::*;
11
12axes![A = 4096, B = 8];
13
14type Chip = m![1];
15type Cluster = m![1 # 2];
16
17/// Decodes an `f4e2m1` weight stream to `f32` via the fetch-adapter table-lookup
18/// stage (`f4e2m1 -> f8e4m3`, the paired-key 4b->8b table) followed by a widening
19/// `fetch_cast`. This is the intended NVFP4 / MXFP4 decode path; the per-block scale
20/// is applied downstream and is intentionally not part of this kernel.
21///
22/// The decode table is a block-independent 16-entry constant, so this one kernel is
23/// the decode for both the NVFP4 (16-element block-scale) and MXFP4 (32-element
24/// block-scale) layouts. The `fetch_table_lookup_decode_e2m1_mxfp4` twin below tiles
25/// the same decode over a 32-element inner block to exercise the MXFP4 block boundary.
26#[device(chip = 1)]
27pub fn fetch_table_lookup_decode_e2m1(
28    ctx: &mut Context,
29    input: &HbmTensor<f4e2m1, m![1], m![A, B]>,
30) -> HbmTensor<f32, m![1], m![B, A]> {
31    let input_dm = input.to_dm::<Cluster, m![A / 16], m![A / 8 % 2, A % 8, B]>(&mut ctx.tdma);
32
33    let decoded: DmTensor<f32, Chip, Cluster, m![A / 16], m![A / 8 % 2, A % 8, B]> = ctx
34        .main
35        .begin(input_dm.view())
36        .fetch::<m![A / 8 % 2], m![A % 8, B]>()
37        .fetch_table_lookup::<f8e4m3>()
38        .fetch_cast::<f32>()
39        .collect::<m![A / 8 % 2, A % 8], m![B]>()
40        .commit_trim::<m![B]>()
41        .commit();
42
43    decoded.to_hbm(&mut ctx.tdma)
44}
45
46// MXFP4's 32-element scale block is twice the NVFP4 16-element block, so the MXFP4 twin doubles
47// the fetched axis (`AM = 2 * A`) to keep the same 256-slice partitioning (`AM / 32 == A / 16`)
48// while the fetch packet spans a 32-element MXFP4 block instead of a 16-element NVFP4 block.
49axes![AM = 8192];
50
51/// MXFP4-block twin of [`fetch_table_lookup_decode_e2m1`]: the same block-independent
52/// `f4e2m1 -> f8e4m3 -> f32` decode, but the fetch packet spans an MXFP4 32-element scale block
53/// (`AM % 16 = 16` inner over 2 time steps = 32) instead of the NVFP4 16-element block. Proves the
54/// packed decode is correct at the wider block granularity; the per-block scale (E8M0 for MXFP4)
55/// stays downstream, as for the NVFP4 kernel.
56#[device(chip = 1)]
57pub fn fetch_table_lookup_decode_e2m1_mxfp4(
58    ctx: &mut Context,
59    input: &HbmTensor<f4e2m1, m![1], m![AM, B]>,
60) -> HbmTensor<f32, m![1], m![B, AM]> {
61    let input_dm = input.to_dm::<Cluster, m![AM / 32], m![AM / 8 % 4, AM % 8, B]>(&mut ctx.tdma);
62
63    let decoded: DmTensor<f32, Chip, Cluster, m![AM / 32], m![AM / 8 % 4, AM % 8, B]> = ctx
64        .main
65        .begin(input_dm.view())
66        .fetch::<m![AM / 8 % 4], m![AM % 8, B]>()
67        .fetch_table_lookup::<f8e4m3>()
68        .fetch_cast::<f32>()
69        .collect::<m![AM / 8 % 4, AM % 8], m![B]>()
70        .commit_trim::<m![B]>()
71        .commit();
72
73    decoded.to_hbm(&mut ctx.tdma)
74}

76// The commit packet must be exactly one flit (32 bytes). The `bf16` value is 2 bytes, so the decode
77// example's inner element axis is 16 wide (16 * 2 = 32 bytes), unlike the NVFP4 decode whose f32
78// output makes an 8-wide axis a flit.
79axes![Bf = 16];
80
81/// Decodes an `f8e4m3` stream to `bf16` through the **non-paired** 8-bit-key baked table
82/// (`f8e4m3 -> bf16`), the general single-key counterpart to the paired NVFP4 / MXFP4 4b->8b decode
83/// above. An 8-bit key indexes one `bf16` value per entry (no paired byte-walk), so the fetch emits
84/// `bf16` directly with no following `fetch_cast`. Proves the non-paired baked table-lookup path.
85#[device(chip = 1)]
86pub fn fetch_table_lookup_decode_f8_to_bf16(
87    ctx: &mut Context,
88    input: &HbmTensor<f8e4m3, m![1], m![A, Bf]>,
89) -> HbmTensor<bf16, m![1], m![Bf, A]> {
90    let input_dm = input.to_dm::<Cluster, m![A / 16], m![A / 8 % 2, A % 8, Bf]>(&mut ctx.tdma);
91
92    let decoded: DmTensor<bf16, Chip, Cluster, m![A / 16], m![A / 8 % 2, A % 8, Bf]> = ctx
93        .main
94        .begin(input_dm.view())
95        .fetch::<m![A / 8 % 2], m![A % 8, Bf]>()
96        .fetch_table_lookup::<bf16>()
97        .collect::<m![A / 8 % 2, A % 8], m![Bf]>()
98        .commit_trim::<m![Bf]>()
99        .commit();
100
101    decoded.to_hbm(&mut ctx.tdma)
102}
