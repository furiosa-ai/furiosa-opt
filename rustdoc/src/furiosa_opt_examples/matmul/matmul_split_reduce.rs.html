1//! Matrix multiplication of size \[1024, 2048\] x \[2048\] -> \[1024\] with split accumulation pattern.
2//! This demonstrates the same split/accumulate pattern as matmul_16384 but with smaller size for testing.
3
4use furiosa_opt_std::prelude::*;
5
6axes![A = 1024, B = 2048, X = 32, I = 2];
7
8type Chip = m![1];
9type Cluster = m![A / 512 % 2];
10
11/// Accumulate two i32 partial sums into another i32 tile. The output stays i32, so no
12/// cast is needed (a `cast::<i32>` would be identity) — `A % 8` is already one i32 flit (32 B).
13fn add_split_contractions_i32(
14    ctx: &mut Context,
15    lhs: &DmTensor<i32, Chip, Cluster, m![A / 64 % 8, X], m![A % 64]>,
16    rhs: &DmTensor<i32, Chip, Cluster, m![A / 64 % 8, X], m![A % 64]>,
17    out: DmTensorViewMut<'_, i32, Chip, Cluster, m![A / 64 % 8, X], m![A % 64]>,
18) {
19    ctx.main
20        .begin_interleaved::<I, _, _, _, _, _>(rhs.view(), lhs.view())
21        .fetch::<m![A / 8 % 8, I], m![A % 8]>()
22        .fetch_cast::<i32>()
23        .collect::<m![A / 8 % 8, I], m![A % 8]>()
24        .vector_init()
25        .vector_intra_slice_unzip::<I, m![A / 8 % 8, 1 # 2], m![A / 8 % 8]>()
26        .vector_clip_zip(ClipBinaryOpI32::AddFxp)
27        .vector_final()
28        .commit_trim::<m![A % 8]>()
29        .commit_view(out)
30}

32/// Accumulate two i32 partial sums and narrow the result to i8. The narrowing `cast::<i8>`
33/// pads the 8-element packet to a full flit (`A % 8 # 32` = 32 i8 = 32 B).
34fn narrow_split_contractions_i8(
35    ctx: &mut Context,
36    lhs: &DmTensor<i32, Chip, Cluster, m![A / 64 % 8, X], m![A % 64]>,
37    rhs: &DmTensor<i32, Chip, Cluster, m![A / 64 % 8, X], m![A % 64]>,
38    out: DmTensorViewMut<'_, i8, Chip, Cluster, m![A / 64 % 8, X], m![A % 64]>,
39) {
40    ctx.main
41        .begin_interleaved::<I, _, _, _, _, _>(rhs.view(), lhs.view())
42        .fetch::<m![A / 8 % 8, I], m![A % 8]>()
43        .fetch_cast::<i32>()
44        .collect::<m![A / 8 % 8, I], m![A % 8]>()
45        .vector_init()
46        .vector_intra_slice_unzip::<I, m![A / 8 % 8, 1 # 2], m![A / 8 % 8]>()
47        .vector_clip_zip(ClipBinaryOpI32::AddFxp)
48        .vector_final()
49        .cast::<i8, m![A % 8 # 32]>()
50        .commit_trim::<m![A % 8]>()
51        .commit_view(out)
52}

54/// Multiply matrices of size \[1024, 2048\] x \[2048\] -> \[1024\]
55/// Demonstrates split/accumulate pattern: B axis is split into 4 tiles (B=512 each)
56/// and accumulated by stashing one group inside VE and later merging with the other group.
57#[device(chip = 1)]
58pub fn matmul_with_split_reduce(
59    ctx: &mut Context,
60    lhs: &HbmTensor<i8, m![1], m![A, B]>,
61    rhs: &HbmTensor<i8, m![1], m![B]>,
62) -> HbmTensor<i8, m![1], m![A]> {
63    type AccDmTensor = DmTensor<i32, Chip, Cluster, m![A / 64 % 8, X], m![A % 64]>;
64    let mut result0 = AccDmTensor::new();
65    let mut result1 = AccDmTensor::new();
66    let mut temp = AccDmTensor::new();
67    let mut output = DmTensor::<i8, Chip, Cluster, m![A / 64 % 8, X], m![A % 64]>::new();
68
69    // Split B axis into 4 tiles (B=512 each) and accumulate using vector engine
70    for j in 0..4 {
71        // Load lhs and rhs for this B tile
72        let lhs_tile = lhs
73            .view()
74            .tile::<m![B / 512], 1, m![A, 1 # 4, B % 512]>(j)
75            .to_dm::<Cluster, m![A / 64 % 8, B / 16 % 32], m![A % 64, B % 16]>(&mut ctx.tdma);
76        let rhs_tile = rhs
77            .view()
78            .tile::<m![B / 512], 1, m![1 # 4, B % 512]>(j)
79            .to_dm::<Cluster, m![A / 64 % 8, B / 16 % 32], m![B % 16]>(&mut ctx.tdma);
80        let rhs_trf: TrfTensor<i8, Chip, Cluster, m![A / 64 % 8, B / 16 % 32], m![1], m![B % 16 # 32]> = ctx
81            .sub
82            .begin(rhs_tile.view())
83            .fetch::<m![1], m![B % 16]>()
84            .fetch_cast::<i8>()
85            .collect::<m![1], m![B % 16 # 32]>()
86            .to_trf();
87
88        // Perform contraction for this tile
89        if j == 0 {
90            // First iteration: store result directly in result0
91            ctx.main
92                .begin(lhs_tile.view())
93                .fetch::<m![A / 2 % 32], m![A % 2, B % 16]>()
94                .fetch_cast::<i8>()
95                .collect::<m![A / 2 % 32], m![A % 2, B % 16]>()
96                .contract_outer::<m![A / 4 % 16], m![A % 4, B % 16], _, _, _>(&rhs_trf)
97                .contract_packet::<m![A % 4]>()
98                .contract_time::<m![A / 4 % 16]>()
99                .contract_lane::<m![A / 4 % 16], m![A % 4 # 8]>(LaneMode::Sequential)
100                .vector_init()
101                .vector_inter_slice_reduce::<m![A / 64 % 8, X], m![A / 4 % 16]>(InterSliceReduceOpI32::Add)
102                .vector_final()
103                .commit_trim::<m![A % 8]>()
104                .commit_view(result0.view_mut());
105        } else if j == 3 {
106            // Final iteration: store in output, conversion to i8 happens
107            ctx.main
108                .begin(lhs_tile.view())
109                .fetch::<m![A / 2 % 32], m![A % 2, B % 16]>()
110                .fetch_cast::<i8>()
111                .collect::<m![A / 2 % 32], m![A % 2, B % 16]>()
112                .contract_outer::<m![A / 4 % 16], m![A % 4, B % 16], _, _, _>(&rhs_trf)
113                .contract_packet::<m![A % 4]>()
114                .contract_time::<m![A / 4 % 16]>()
115                .contract_lane::<m![A / 4 % 16], m![A % 4 # 8]>(LaneMode::Sequential)
116                .vector_init()
117                .vector_inter_slice_reduce::<m![A / 64 % 8, X], m![A / 4 % 16]>(InterSliceReduceOpI32::Add)
118                .vector_final()
119                .commit_trim::<m![A % 8]>()
120                .commit_view(temp.view_mut());
121
122            // Accumulate using vector engine: stash previous partial sums and zip with new ones
123            narrow_split_contractions_i8(ctx, &result0, &temp, output.view_mut());
124        } else {
125            // Subsequent iterations: store in temp and accumulate
126            ctx.main
127                .begin(lhs_tile.view())
128                .fetch::<m![A / 2 % 32], m![A % 2, B % 16]>()
129                .fetch_cast::<i8>()
130                .collect::<m![A / 2 % 32], m![A % 2, B % 16]>()
131                .contract_outer::<m![A / 4 % 16], m![A % 4, B % 16], _, _, _>(&rhs_trf)
132                .contract_packet::<m![A % 4]>()
133                .contract_time::<m![A / 4 % 16]>()
134                .contract_lane::<m![A / 4 % 16], m![A % 4 # 8]>(LaneMode::Sequential)
135                .vector_init()
136                .vector_inter_slice_reduce::<m![A / 64 % 8, X], m![A / 4 % 16]>(InterSliceReduceOpI32::Add)
137                .vector_final()
138                .commit_trim::<m![A % 8]>()
139                .commit_view(temp.view_mut());
140
141            // Accumulate using vector engine: stash the older tile internally and zipped with the new tile
142            if j % 2 == 1 {
143                add_split_contractions_i32(ctx, &result0, &temp, result1.view_mut());
144            } else {
145                add_split_contractions_i32(ctx, &result1, &temp, result0.view_mut());
146            }
147        }
148    }
149
150    // Write back to HBM
151    let mut out = unsafe { HbmTensor::<i8, m![1], m![A]>::from_addr(0x3000) };
152    output
153        .view()
154        .slice_tile::<m![X], 1, m![A / 64 % 8, 1 # 32]>(0)
155        .to_hbm_view(&mut ctx.tdma, out.view_mut());
156    out
157}
