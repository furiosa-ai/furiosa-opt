1//! Scatter/gather test kernels.
2//!
3//! `scatter_minimal` scatters `K = 512` keys into a `C = 612` cache (sparse: `C > K`);
4//! the non-power-of-2, larger-than-`K` cache stresses non-aligned scatter coverage.
5//!
6//! `gather_minimal` gathers `G = 512` rows from the `K`-row table. Its gather count `G` is
7//! separate from the scatter cache `C` so the values partition `G / 2 = 256` equals
8//! `num_slices` (the sparse `C / 2 = 306` would round up past it and fail to lower).
9//! `gather_unaligned` covers a non-power-of-2 count (`U = 768`, 3 rows/slice) that still meets
10//! the 256-slice constraint.
11//!
12//! `gather_paged_kv` gathers over a 4-D DRAM pool whose gather-key label `NBlocks` is reused in
13//! the gathered output. That reused key is what the visa->LIR indirect-key relabeling isolates by
14//! subtracting the index's own labels.
15
16use furiosa_opt_std::prelude::*;
17
18axes![
19    K = 512, // Scatter key
20    D = 128, // Payload per key
21    C = 612, // Cache length (non-power-of-2, > K: stresses unaligned coverage)
22    G = 512, // Slice-aligned gather count (G / 2 = 256)
23    U = 768, // Unaligned gather count (non-power-of-2, > K: 3 rows/slice, U / 3 = 256)
24    CL = 2   // Real cluster partition (hardware has 2 clusters/chip): placed, not broadcast
25];
26
27// Multi-dim DRAM-pool gather axes. `NBlocks * PBlock = 256` fills the slices; the gather remaps
28// the leading `NBlocks` rows to physical pool rows via a byte-offset index, carrying a
29// `[KvHeads, HeadDim]` payload per gathered row.
30axes![
31    NBlocks = 16,  // gathered block count
32    PBlock = 16,   // block size (lane-aligned); NBlocks * PBlock = 256
33    KvHeads = 8,   // outer payload axis
34    HeadDim = 128  // inner payload axis
35];
36
37type Chip = m![1];
38type Cluster = m![1 # 2];
39
40/// Scatter values into cache at index positions.
41#[device(chip = 1)]
42pub fn scatter_minimal(
43    ctx: &mut Context,
44    data: &HbmTensor<bf16, Chip, m![K, D]>,
45    index: &HbmTensor<i32, Chip, m![K]>,
46    output: &mut HbmTensor<bf16, Chip, m![C, D]>,
47) {
48    let data_dm: DmTensor<bf16, Chip, Cluster, m![K / 2], m![K % 2, D]> = data.to_dm(&mut ctx.tdma);
49
50    data_dm.dma_scatter::<m![K], _, _>(index, output);
51}
52
53/// Gather `G` rows from `table` at positions given by `index` into a fresh HBM output.
54///
55/// Inverse of [`scatter_minimal`]. Produces a `DmTensor` via [`HbmTensor::dma_gather_scaled`], then
56/// writes it back to HBM via [`DmTensor::to_hbm`], returned by value because `dma_gather_scaled` does
57/// not write into an existing `&mut HbmTensor`. The `G / 2 = 256` values partition equals
58/// `num_slices`, so it lowers end to end through VISA -> LIR -> EDF.
59#[device(chip = 1)]
60pub fn gather_minimal(
61    ctx: &mut Context,
62    table: &HbmTensor<bf16, Chip, m![K, D]>,
63    index: &HbmTensor<i32, Chip, m![G]>,
64) -> HbmTensor<bf16, Chip, m![G, D]> {
65    let values_dm: DmTensor<bf16, Chip, Cluster, m![G / 2], m![G % 2, D]> = table.dma_gather_scaled(index);
66
67    values_dm.to_hbm(&mut ctx.tdma)
68}
69
70/// Gather an unaligned count `U = 768` (`= 3 * num_slices`, non-power-of-2, `> K`) from the
71/// `K`-row table. Unlike [`gather_minimal`]'s `2` rows/slice, the values partition here packs
72/// `3` rows/slice (`U / 3 = 256 = num_slices`), so it still lowers to EDF while covering a
73/// non-power-of-2 gather size that wraps the table.
74#[device(chip = 1)]
75pub fn gather_unaligned(
76    ctx: &mut Context,
77    table: &HbmTensor<bf16, Chip, m![K, D]>,
78    index: &HbmTensor<i32, Chip, m![U]>,
79) -> HbmTensor<bf16, Chip, m![U, D]> {
80    let values_dm: DmTensor<bf16, Chip, Cluster, m![U / 3], m![U % 3, D]> = table.dma_gather_scaled(index);
81
82    values_dm.to_hbm(&mut ctx.tdma)
83}

85/// Gather over a multi-dimensional DRAM pool `[NBlocks, KvHeads, PBlock, HeadDim]`, remapping the
86/// leading `NBlocks` rows to physical pool rows via a byte-offset index. The gathered output
87/// regroups the two sub-axes `[NBlocks, PBlock]` onto the 256 slices with a `[KvHeads, HeadDim]`
88/// payload per slice.
89///
90/// Unlike the 2-D `[K, D]` gathers above, the gather-key label `NBlocks` is reused in the gathered
91/// output (the gathered rows stay ordered by the same `NBlocks` axis the index addresses). That
92/// reuse hits the operator-verify wall (`index_axis.len() (0) != 1`) unless the visa->LIR
93/// relabeling subtracts the index's own labels before deciding which table axis is genuinely kept,
94/// see `LirBuilder::relabel_indirect_key_axis`.
95#[device(chip = 1)]
96pub fn gather_paged_kv(
97    ctx: &mut Context,
98    pool: &HbmTensor<bf16, Chip, m![NBlocks, KvHeads, PBlock, HeadDim]>,
99    block_table: &HbmTensor<i32, Chip, m![NBlocks]>,
100) -> HbmTensor<bf16, Chip, m![NBlocks, KvHeads, PBlock, HeadDim]> {
101    // HBM write-back past the pool; only distinctness matters for a lowering probe.
102    let gathered: DmTensor<bf16, Chip, Cluster, m![NBlocks, PBlock], m![KvHeads, HeadDim]> =
103        pool.dma_gather_scaled(block_table);
104
105    gathered.to_hbm(&mut ctx.tdma)
106}

108/// Unscaled twin of [`gather_minimal`]: the index holds RAW row positions instead of byte offsets.
109/// The unscaled path reads the index straight off SPM, so the kernel first stages the DRAM index
110/// on-chip with [`HbmTensor::to_dm`] and hands the resulting `DmTensor` to
111/// [`HbmTensor::dma_gather_unscaled`]. This is the embed / paged-attention block-table idiom. Same
112/// `G / 2 = 256` partitioning as [`gather_minimal`] so it lowers end to end through VISA -> LIR.
113#[device(chip = 1)]
114pub fn gather_aligned_unscaled(
115    ctx: &mut Context,
116    table: &HbmTensor<bf16, Chip, m![K, D]>,
117    index: &HbmTensor<i32, Chip, m![G]>,
118) -> HbmTensor<bf16, Chip, m![G, D]> {
119    let index_dm: DmTensor<i32, Chip, Cluster, m![G / 2], m![G % 2]> = index.to_dm(&mut ctx.tdma);
120    let values_dm: DmTensor<bf16, Chip, Cluster, m![G / 2], m![G % 2, D]> = table.dma_gather_unscaled(&index_dm);
121
122    values_dm.to_hbm(&mut ctx.tdma)
123}
124
125/// Placed-cluster twin of [`gather_aligned_unscaled`]: the staged SPM index is distributed across
126/// the two clusters as a REAL partition (`m![CL]`, a `LabelStride` cluster) instead of a broadcast
127/// (`m![1 # 2]`). Each cluster gathers its own `G` rows through its own slice of the index, so the
128/// index's cluster axis carries genuine per-cluster data rather than a replicated copy. Exercises
129/// the unscaled gather lowering with a non-broadcast SPM index cluster.
130#[device(chip = 1)]
131pub fn gather_placed_unscaled(
132    ctx: &mut Context,
133    table: &HbmTensor<bf16, Chip, m![K, D]>,
134    index: &HbmTensor<i32, Chip, m![CL, G]>,
135) -> HbmTensor<bf16, Chip, m![CL, G, D]> {
136    let index_dm: DmTensor<i32, Chip, m![CL], m![G / 2], m![G % 2]> = index.to_dm(&mut ctx.tdma);
137    let values_dm: DmTensor<bf16, Chip, m![CL], m![G / 2], m![G % 2, D]> = table.dma_gather_unscaled(&index_dm);
138
139    values_dm.to_hbm(&mut ctx.tdma)
140}
