10/// Configuration for the switch (slice/time topology) engine: the caller picks the topology and its
11/// dimensions, and `config_switch` validates it against the input/output slice/time shapes. Lives in
12/// `furiosa-opt-lower-types` (the engine re-exports it) so the `CustomBroadcast` divide-algebra stays
13/// off the public engine surface. `#[primitive]` tags the variants for the visa MIR plugin.
14#[primitive(SwitchConfig)]
15#[repr(C)]
16#[derive(StableAbi, Debug, Clone, Copy, PartialEq, Eq)]
17pub enum SwitchConfig {
18    /// Passes each slice's data through unchanged (`ring_size = 1`); `Slice` and `Time` preserved.
19    Forwarding,
20    /// Replicates data across slices along slice dimensions 0 and 1.
21    Broadcast01 {
22        /// Slice dimension 1 size.
23        slice1: usize,
24        /// Slice dimension 0 size.
25        slice0: usize,
26        /// Time dimension 0 size.
27        time0: usize,
28    },
29    /// Replicates data across slices along slice dimension 1.
30    Broadcast1 {
31        /// Slice dimension 1 size.
32        slice1: usize,
33        /// Slice dimension 0 size.
34        slice0: usize,
35    },
36    /// Swaps slice1 and slice0 in the slice dimension; time unchanged.
37    Transpose {
38        /// Slice dimension 1 size.
39        slice1: usize,
40        /// Slice dimension 0 size.
41        slice0: usize,
42    },
43    /// Swaps and transposes between the slice and time dimensions.
44    InterTranspose {
45        /// Slice dimension 1 size.
46        slice1: usize,
47        /// Slice dimension 0 size.
48        slice0: usize,
49        /// Time dimension 0 size.
50        time0: usize,
51    },
52    /// Routes data across slices using a custom snoop bitmap, computed by the compiler from the input
53    /// shape and topology parameters.
54    CustomBroadcast {
55        /// Ring group size for the custom routing. Must be a power of two: it feeds the SFR's
56        /// `outer_dim0_size_log = ring_size.trailing_zeros()`, so a non-power-of-two misencodes.
57        ring_size: usize,
58    },
59    /// Swaps slice1/slice0 and replicates along slice dimension 0 (Transpose + Broadcast1 at once).
60    TransposedBroadcast1 {
61        /// Slice dimension 1 size.
62        slice1: usize,
63        /// Slice dimension 0 size.
64        slice0: usize,
65    },
66}
67
68/// A named slice sub-axis of the `(slice2, slice1, slice0)` partition a switch config
69/// splits its input into. Labels which component failed the power-of-two check.
70#[repr(C)]
71#[derive(StableAbi, Debug, Clone, Copy, PartialEq, Eq)]
72#[allow(missing_docs)] // The names are self-describing.
73pub enum SwitchAxis {
74    Slice0,
75    Slice1,
76    Slice2,
77}
78
79impl fmt::Display for SwitchAxis {
80    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
81        let s = match self {
82            SwitchAxis::Slice0 => "slice0",
83            SwitchAxis::Slice1 => "slice1",
84            SwitchAxis::Slice2 => "slice2",
85        };
86        f.write_str(s)
87    }
88}

90/// The output frame a mismatched component lives in (the only frames an error reports).
91#[repr(C)]
92#[derive(StableAbi, Debug, Clone, Copy, PartialEq, Eq)]
93#[allow(missing_docs)] // The names are self-describing.
94pub enum SwitchFrame {
95    OutSlice,
96    OutTime,
97}
98
99impl fmt::Display for SwitchFrame {
100    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
101        let s = match self {
102            SwitchFrame::OutSlice => "OutSlice",
103            SwitchFrame::OutTime => "OutTime",
104        };
105        f.write_str(s)
106    }
107}

109/// Why a switch is not realizable on the Switch engine. Named configs verify constructively,
110/// so a positional failure is one [`SwitchError::OutputMismatch`] naming the offending frame;
111/// `CustomBroadcast` keeps the general sequencer variants (ring size, broadcast newness, moved
112/// order).
113///
114/// `#[repr(C)]` + `StableAbi` so the full diagnostic crosses the verifier `extern "C-unwind"`
115/// boundary into `furiosa-opt-std` (like [`crate::FetchError`]) rather than a string.
116#[repr(C)]
117#[derive(StableAbi, Debug, Clone, PartialEq, Eq, thiserror::Error)]
118pub enum SwitchError {
119    /// Input and output slice volumes differ.
120    #[error("Switch input and output slice sizes must match, got {in_slice} and {out_slice}")]
121    SliceSizeMismatch {
122        /// InSlice volume.
123        in_slice: usize,
124        /// OutSlice volume.
125        out_slice: usize,
126    },
127    /// A `(slice1, slice0, time0)` dimension was zero.
128    #[error("All dimensions must be greater than 0")]
129    ZeroDimension,
130    /// A slice sub-dimension is not a power of two (the switch network requires it).
131    #[error("Switch {axis} must be a power of 2, got {value}")]
132    NotPowerOfTwo {
133        /// Which sub-axis.
134        axis: SwitchAxis,
135        /// Its size.
136        value: usize,
137    },
138    /// InSlice volume is not a multiple of `slice1 * slice0`.
139    #[error(
140        "InSlice::SIZE must be divisible by (slice1 * slice0), got InSlice::SIZE {in_slice_size} (slice1 {slice1}, slice0 {slice0})"
141    )]
142    SliceNotDivisible {
143        /// InSlice volume.
144        in_slice_size: usize,
145        /// Slice dimension 1 size.
146        slice1: usize,
147        /// Slice dimension 0 size.
148        slice0: usize,
149    },
150    /// `Broadcast01`'s InTime volume is not a multiple of its `time0` split.
151    #[error("Broadcast01 InTime::SIZE must be divisible by time0, got InTime::SIZE {in_time_size} (time0 {time0})")]
152    Broadcast01TimeNotDivisible {
153        /// InTime volume.
154        in_time_size: usize,
155        /// Time dimension 0 size.
156        time0: usize,
157    },
158    /// `InterTranspose`'s InTime volume is not a multiple of its `slice1 * time0` split.
159    #[error(
160        "InterTranspose InTime::SIZE must be divisible by (slice1 * time0), got InTime::SIZE {in_time_size} (slice1 {slice1}, time0 {time0})"
161    )]
162    InterTransposeTimeNotDivisible {
163        /// InTime volume.
164        in_time_size: usize,
165        /// Slice dimension 1 size.
166        slice1: usize,
167        /// Time dimension 0 size.
168        time0: usize,
169    },
170    /// The actual output frame does not match the one the config reassembles from the
171    /// input components (a moved/preserved component differs, or a tile slot is not a
172    /// fresh broadcast). One comparison per output frame; `expected` shows tiles as
173    /// `broadcast(n)`.
174    #[error("Switch {frame} mismatch: expected {expected}, got {got}")]
175    OutputMismatch {
176        /// The output frame.
177        frame: SwitchFrame,
178        /// The output the `(input, config)` recipe reassembles (tiles shown as `broadcast(n)`).
179        expected: Mapping,
180        /// The actual output.
181        got: Mapping,
182    },
183    /// The asserted ring size is not a power of two (the switch network requires it).
184    #[error("Switch ring size must be a power of 2, got {ring_size}")]
185    RingSizeNotPowerOfTwo {
186        /// The asserted ring size.
187        ring_size: usize,
188    },
189    /// The asserted ring size differs from the slot-sweep value.
190    #[error("Switch ring size mismatch: the slot sweep computed {computed}, but {asserted} was asserted")]
191    RingSizeMismatch {
192        /// The slot-sweep-derived ring size.
193        computed: usize,
194        /// The asserted ring size.
195        asserted: usize,
196    },
197    /// The `(in_slice, in_time) → (out_slice, out_time)` read does not sequence: an
198    /// output slot has no input source, or a live input cell is dropped (padding may be
199    /// trimmed, live data may not).
200    #[error("Switch read is not sequenceable: {0:?}")]
201    Unsequenceable(SequencerError),
202    /// A broadcast axis re-used an InSlice / InTime identifier.
203    #[error("Switch broadcast axis {offending_ident} must be a new axis (not present in input Slice or Time).")]
204    BroadcastNotNew {
205        /// The identifier that re-used an input Slice/Time axis.
206        offending_ident: Ident,
207    },
208    /// An OutSlice slot is sourced from InTime (a time→slice cross), not from InSlice or a
209    /// fresh broadcast tile. Reported by its source mapping rather than an `Ident`, since a
210    /// padding/broadcast time axis carries no identifier to name.
211    #[error(
212        "Switch OutSlice slot must source from InSlice or a fresh broadcast tile, not InTime; offending axis: {offending_axis}"
213    )]
214    OutSliceSourcedFromTime {
215        /// The InTime-sourced axis that appeared in OutSlice.
216        offending_axis: Mapping,
217    },
218    /// A broadcast axis appeared more than once in OutSlice.
219    #[error("Switch broadcast axis {ident} must be used exactly once in OutSlice")]
220    BroadcastUsedMoreThanOnce {
221        /// The identifier that appeared on more than one tile.
222        ident: Ident,
223    },
224    /// Slice→time axes are not at OutTime's innermost positions.
225    #[error(
226        "Switch axes moving from input slice to output time must be at the output time innermost positions (inner to InTime {time}); offending moved axis: {offending_axis}"
227    )]
228    MovedAxesNotInnermost {
229        /// The InTime layout the moved axes must sit inner to.
230        time: Mapping,
231        /// The single InSlice-sourced axis that landed outer to an InTime axis.
232        offending_axis: Mapping,
233    },
234    /// Slice→time axes lost their relative InSlice order in OutTime, reported as the
235    /// offending pair of InSlice strides (an axis may be padding/broadcast, which has no
236    /// `Term`, so the strides, not the terms, name the violation).
237    #[error(
238        "Switch axes moving from input Slice to output Time must preserve their relative order: {landed_inner} (InSlice stride {landed_inner_stride}) wrongly landed inner to {nested_under} (stride {nested_under_stride}); the larger stride must stay outer"
239    )]
240    MovedOrderNotPreserved {
241        /// The moved axis that wrongly landed inner (the larger InSlice stride).
242        landed_inner: Mapping,
243        /// InSlice stride of `landed_inner`.
244        landed_inner_stride: usize,
245        /// The axis it nested under (the smaller InSlice stride).
246        nested_under: Mapping,
247        /// InSlice stride of `nested_under`.
248        nested_under_stride: usize,
249    },
250    /// A switch carries InTime through to OutTime unchanged (including padding); only the
251    /// axes moved from InSlice are added, at OutTime's innermost positions.
252    #[error(
253        "Switch input Time must be carried to output Time unchanged, got {in_time} but output Time carries {carried}"
254    )]
255    InTimeNotPreserved {
256        /// InTime layout.
257        in_time: Mapping,
258        /// The InTime portion OutTime actually carries (OutTime with the moved-axes block
259        /// stripped), not the full OutTime.
260        carried: Mapping,
261    },
262    /// The sequencer-derived snoop bitmap disagreed with an independent enumeration at
263    /// output slice `out_slice_lane`. This is the always-on soundness backstop on the snoop
264    /// lanes that reach hardware, so a mismatch is a real attribution bug, surfaced as an
265    /// error rather than a panic in the fallible bitmap path.
266    #[error("Switch snoop bitmap disagrees with the (OutSlice x OutTime) enumeration at output slice {out_slice_lane}")]
267    BitmapAttributionMismatch {
268        /// The first output slice whose snooped input-lane set differs between the two derivations.
269        out_slice_lane: usize,
270    },
