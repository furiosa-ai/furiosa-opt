3use abi_stable::StableAbi;
4use furiosa_mapping_types::{Mapping, SequencerError};

6use crate::MAX_SEQUENCER_ENTRIES;

8/// Why a fetch is not realizable on the Fetch engine — one variant per `config_fetch` / frontend
9/// `verify_fetch` check.
10#[repr(C)]
11#[derive(StableAbi, Debug, Clone, PartialEq, Eq, thiserror::Error)]
12pub enum FetchError {
13    /// Cluster size is not 1 or 2.
14    #[error("Fetch: Cluster size must be 1 or 2, got {0}")]
15    ClusterSize(usize),
16    /// Slice size is not one of 64/128/256.
17    #[error("Fetch: Slice size must be one of 64/128/256, got {0}")]
18    SliceSize(usize),
19    /// The matcher could not place an output axis against the DM memories.
20    #[error("Fetch: cannot read an output axis from DM ({0:?})")]
21    Unreadable(SequencerError),
22    /// A live input axis was left unread (the carved-down DM remainders).
23    #[error("Fetch: a live input axis is left unread (Time {time}, Packet {packet})")]
24    Unread {
25        /// The carved-down `Time` memory remainder.
26        time: Mapping,
27        /// The carved-down `Packet` memory remainder.
28        packet: Mapping,
29    },
30    /// The packet and time descriptors need more entries than the shared table holds.
31    #[error("Fetch: needs {needed} sequencer entries, but the table holds {MAX_SEQUENCER_ENTRIES}")]
32    TooManyEntries {
33        /// The number of entries the two descriptors need.
34        needed: usize,
35    },
36}
