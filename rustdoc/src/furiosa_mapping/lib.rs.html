36/// Methods for [`Mapping`] backed by the impl crate over FFI. Pure `Mapping` operations live as
37/// inherent methods in `furiosa-mapping-types`; this trait carries only the ones that need the impl.
38pub trait MappingExt: Sized {
39    /// Normalizes to canonical form.
40    fn normalize(&self) -> Self;
41    /// Returns true if `self` is a resize (innermost prefix) of `original`.
42    fn is_resize_of(&self, original: &Self) -> bool;
43    /// Splits at buffer `target` into `(outer, inner)`: `outer` strides past the first `target`
44    /// cells, `inner` keeps them, and `outer.pair(inner)` reads the same buffer. `target` must
45    /// divide the size.
46    fn split_at(&self, target: usize) -> (Self, Self);
47    /// The read at buffer `position`, an [`Index`] of per-axis contributions with composites kept
48    /// WHOLE. Call [`IndexExt::finalize`] to decode them (`mapping.index(i).finalize()`).
49    fn index(&self, position: usize) -> Index;
50    /// The raw read at every buffer position `0..size`, composites kept WHOLE, in one FFI crossing. The
51    /// batch form of `index`; call [`IndexExt::finalize`] on a cell to decode it.
52    fn indexes(&self) -> Vec<Index>;
53    /// The live axis terms (its [`Term`]s), padding excluded.
54    fn axes(&self) -> Vec<Term>;
55    /// Carves `piece` out of `self` through the matcher, returning the leftover the
56    /// scatter/gather/reduce/broadcast sites need. Each matched cell becomes a `Top` pad (the carved
57    /// hole), each unmatched cell stays live (a broadcast at `memory_stride` 0). Panics unless `piece`
58    /// is contained in `self`.
59    fn carve(&self, piece: &Self) -> Self;
60}

62/// Matches each of `streams` against the `memories`, each its own address space (the fetch engine's
63/// `Time` and `Packet`), and returns one coalesced [`SequencerConfig`] per stream — each entry's
64/// `memory_index` says which memory it reads. The streams' segments are pooled and carved together
65/// (term priority is global across streams), so a `Broadcast` never claims a pad a `Term` needs.
66///
67/// Coverage is enforced: every memory must end fully consumed for `mode` (a live cell left unread under
68/// Read / unwritten under Write is [`SequencerError::Unconsumed`], carrying the carved-down memories so
69/// the caller can name the offender). Inputs are read-only; the carving happens on internal copies.
70pub fn sequence(
71    memories: &[&Mapping],
72    streams: &[&Mapping],
73    mode: SequencerMode,
74) -> Result<Vec<SequencerConfig>, SequencerError> {
75    let memories: RVec<Mapping> = memories.iter().map(|m| (*m).clone()).collect();
76    let streams: RVec<Mapping> = streams.iter().map(|m| (*m).clone()).collect();
77    let configs = unsafe { sys::mapping_sequence(memories.as_rslice(), streams.as_rslice(), mode) }.into_result()?;
78    Ok(configs.into_iter().map(SequencerConfigExt::coalesce).collect())
79}
80
81impl MappingExt for Mapping {
82    fn normalize(&self) -> Self {
83        unsafe { sys::mapping_normalize(self) }
84    }
85
86    fn is_resize_of(&self, original: &Self) -> bool {
87        let n = self.size();
88        n <= original.size() && self.normalize() == original.clone().resize(n).normalize()
89    }
90
91    fn split_at(&self, target: usize) -> (Self, Self) {
92        let Tuple2(outer, inner) = unsafe { sys::mapping_split_at(self, target) };
93        (outer, inner)
94    }
95
96    fn index(&self, position: usize) -> Index {
97        unsafe { sys::mapping_index(self, position) }
98    }
99
100    fn indexes(&self) -> Vec<Index> {
101        unsafe { sys::mapping_indexes(self) }.into_iter().collect()
102    }
103
104    fn axes(&self) -> Vec<Term> {
105        unsafe { sys::mapping_axes(self) }.into_iter().collect()
106    }
107
108    fn carve(&self, piece: &Self) -> Self {
109        let configs = sequence(&[piece], &[self], SequencerMode::Read).expect("carve: piece must be contained in self");
110        let mut acc = Mapping::identity();
111        for (_key, entry) in configs[0].0.iter() {
112            let seg = if entry.memory_stride == 0 {
113                // Unbacked stream cell, a live leftover axis (a broadcast vs the carved memory).
114                entry.mapping.clone()
115            } else {
116                // Matched cell, a carved hole kept as Top padding so positions are preserved.
117                Mapping::identity().padding(entry.mapping.size(), PaddingKind::Top)
118            };
119            acc = seg.pair(acc);
120        }
121        acc
122    }
123}

125/// Methods for [`SequencerConfig`].
126pub trait SequencerConfigExt {
127    /// Coalesces entries that sit back-to-back in the memory buffer into one, to minimize the
128    /// sequencer entry count. Result entries are keyed by cumulative-product stream stride, so
129    /// adjacent entries are always stream-contiguous; the merge condition is memory-contiguity
130    /// alone (`inner.memory_stride * inner.size == outer.memory_stride`). The merged entry keeps
131    /// the inner's key and memory stride, with the run paired into one mapping.
132    ///
133    /// Apply this PER descriptor (e.g. packet and time separately), never across, or it would
134    /// coalesce two entries that must stay distinct.
135    fn coalesce(self) -> Self;
136}
137
138impl SequencerConfigExt for SequencerConfig {
139    fn coalesce(self) -> SequencerConfig {
140        let mut out: Vec<(usize, SequencerEntry)> = Vec::new();
141        for (key, entry) in self.0 {
142            match out.last_mut() {
143                // Same memory and memory-contiguous: extend the run. This (outer) entry wraps the
144                // inner run; the run keeps the inner's key and memory stride, growing only its paired
145                // mapping. Entries in different memories (different spaces) never merge.
146                Some((_, run))
147                    if run.memory_index == entry.memory_index
148                        && run.memory_stride * run.mapping.size() == entry.memory_stride =>
149                {
150                    let inner = std::mem::replace(&mut run.mapping, Mapping::identity());
151                    run.mapping = entry.mapping.pair(inner);
152                }
153                _ => out.push((key, entry)),
154            }
155        }
156        SequencerConfig(out.into_iter().collect())
157    }
158}

160/// Methods for [`Index`].
161pub trait IndexExt: Sized {
162    /// Marks this index as invalid.
163    fn mark_invalid(&mut self);
164    /// Adds a mapping to this index.
165    fn add_mapping<I: crate::M>(&mut self, value: usize);
166    /// The terminal read: decodes every composite and returns each symbol's absolute coordinate
167    /// (`Ident` -> coordinate, coord-0 dropped), or the [`PaddingKind`] a pad cell lands on.
168    fn finalize(self) -> RResult<RSortedMap<Ident, usize>, PaddingKind>;
169    /// Generates all possible indexes based on the given mapping.
170    fn gen_indexes(&self, mapping: Mapping) -> RVec<Self>;
171}
172
173impl IndexExt for Index {
174    fn mark_invalid(&mut self) {
175        self.0 = RResult::RErr(PaddingKind::Bottom);
176    }
177    fn add_mapping<I: crate::M>(&mut self, value: usize) {
178        self.add(I::to_value().index(value));
179    }
180    fn finalize(self) -> RResult<RSortedMap<Ident, usize>, PaddingKind> {
181        unsafe { sys::index_finalize(self) }
182    }
183    fn gen_indexes(&self, mapping: Mapping) -> RVec<Self> {
184        unsafe { sys::index_gen_indexes(self, &mapping) }
185    }
186}
