42/// Methods for [`Mapping`] backed by the impl crate over FFI. Pure `Mapping` operations live as
43/// inherent methods in `furiosa-mapping-types`; this trait carries only the ones that need the impl.
44pub trait MappingExt: Sized {
45    /// Normalizes to canonical form.
46    fn normalize(&self) -> Self;
47    /// The DMA burst tails for copying source `self` into destination `dst`, as
48    /// `(src_packet, dst_packet, common)`.
49    ///
50    /// `common` is the innermost volume both sides read the same way, which is the volume either side
51    /// can SEQUENCE. Nothing is subtracted from it: padding the two sides share is part of the addressing
52    /// they share, so it is transferred like any other cell. Only a `Bottom` write hole or a `Zero` that
53    /// must read as zero caps it, since no burst may cross one. A DRAM sink pins its tail to it.
54    ///
55    /// Each packet grows `common` through THAT side's own trailing `Top` padding, so the two may differ (a
56    /// padded source against a dense sink) and are reconciled by the tail alignment rather than by a
57    /// shared peel size. A packet need not divide its buffer; the caller that peels one checks.
58    ///
59    /// `(1, 1, 1)` when the two share nothing but the origin cell, a transpose being the plain case.
60    fn dma_tails(&self, dst: &Self) -> (usize, usize, usize);
61    /// Returns true if `self` is a resize (innermost prefix) of `original`.
62    fn is_resize_of(&self, original: &Self) -> bool;
63    /// Splits at buffer `target` into `(outer, inner)`: `outer` strides past the first `target`
64    /// cells, `inner` keeps them, and `outer.pair(inner)` reads the same buffer. `target` must
65    /// divide the size.
66    fn split_at(&self, target: usize) -> (Self, Self);
67    /// The read at buffer `position`, an [`Index`] of per-axis contributions with composites kept
68    /// WHOLE. Call [`IndexExt::finalize`] to decode them (`mapping.index(i).finalize()`).
69    fn index(&self, position: usize) -> Index;
70    /// The raw read at every buffer position `0..size`, composites kept WHOLE, in one FFI crossing. The
71    /// batch form of `index`; call [`IndexExt::finalize`] on a cell to decode it.
72    fn indexes(&self) -> Vec<Index>;
73    /// The live axis terms (its [`AxisTerm`]s, resolved symbols), padding excluded.
74    fn axes(&self) -> Vec<AxisTerm>;
75    /// A lazy iterator over where this mapping's cells land in a buffer laid out by `axes`, each offset
76    /// shifted by `base`: one `Option<usize>` per physical cell in canonical order (`Some(off)` live,
77    /// `None` padding). `padding` picks the traversal: `true` visits every physical cell (the wire order
78    /// the buffer seam needs), `false` skips the padding cells (faster on padded mappings, for relayouts).
79    ///
80    /// `axes` is the *target* buffer's layout (resolved symbol terms, e.g. from [`MappingExt::axes`]),
81    /// not necessarily this mapping's own axes: the mapping is projected
82    /// onto it. A symbol in both contributes its coordinate; a symbol only in the mapping contributes
83    /// nothing (it broadcasts, e.g. a dst axis absent from the src); a symbol only in `axes` stays at
84    /// its origin. The offsets land inside the buffer only if `axes` covers the mapping's reach: each
85    /// shared axis's extent must hold the mapping's coordinate for it (an out-of-range coordinate
86    /// wraps, like an out-of-range `base`).
87    ///
88    /// The returned [`MappingIter`] is an [`Iterator`]; it is fully owned (borrows nothing), so a
89    /// consumer just walks it (`.zip`, `.map`, `.flatten`) without ever naming the type.
90    fn iter(&self, axes: &[AxisTerm], base: &Index, padding: bool) -> MappingIter;
91    /// The leftover `self − piece` (the scatter/gather/reduce/broadcast residue): `self` is sequenced as
92    /// the STREAM against `piece` as memory, so a `self` cell `piece` backs becomes a `Top` pad and a cell
93    /// it does not back stays live (the broadcast). Runs in [`SequencerMode::Carve`], so a `Bottom` pad in
94    /// `self` (a `view_mut().tile()` hole) is tolerated as padding. Panics unless `piece` is in `self`.
95    fn carve(&self, piece: &Self) -> Self;
96}

98/// Matches each of `streams` against the `memories`, each its own address space (the fetch engine's
99/// `Time` and `Packet`), and returns one coalesced [`SequencerConfig`] per stream — each entry's
100/// `memory_index` says which memory it reads. The streams' segments are pooled and carved together
101/// (term priority is global across streams), so a `Broadcast` never claims a pad a `Term` needs.
102///
103/// Coverage is enforced: every memory must end fully consumed for `mode` (a live cell left unread under
104/// Read / unwritten under Write is [`SequencerError::Unconsumed`], carrying the carved-down memories so
105/// the caller can name the offender). Inputs are read-only; the carving happens on internal copies.
106pub fn sequence(
107    memories: &[&Mapping],
108    streams: &[&Mapping],
109    mode: SequencerMode,
110) -> Result<Vec<SequencerConfig>, SequencerError> {
111    let memories: RVec<Mapping> = memories.iter().map(|m| (*m).clone()).collect();
112    let streams: RVec<Mapping> = streams.iter().map(|m| (*m).clone()).collect();
113    let configs = unsafe { sys::mapping_sequence(memories.as_rslice(), streams.as_rslice(), mode) }.into_result()?;
114    Ok(configs.into_iter().map(SequencerConfigExt::coalesce).collect())
115}

117impl MappingExt for Mapping {
118    fn normalize(&self) -> Self {
119        unsafe { sys::mapping_normalize(self) }
120    }
121
122    fn dma_tails(&self, dst: &Self) -> (usize, usize, usize) {
123        let Tuple3(src_packet, dst_packet, common) = unsafe { sys::mapping_dma_tails(self, dst) };
124        (src_packet, dst_packet, common)
125    }
126
127    fn is_resize_of(&self, original: &Self) -> bool {
128        let n = self.size();
129        n <= original.size() && self.normalize() == original.clone().resize(n).normalize()
130    }
131
132    fn split_at(&self, target: usize) -> (Self, Self) {
133        let Tuple2(outer, inner) = unsafe { sys::mapping_split_at(self, target) };
134        (outer, inner)
135    }
136
137    fn index(&self, position: usize) -> Index {
138        unsafe { sys::mapping_index(self, position) }
139    }
140
141    fn indexes(&self) -> Vec<Index> {
142        unsafe { sys::mapping_indexes(self) }.into_iter().collect()
143    }
144
145    fn axes(&self) -> Vec<AxisTerm> {
146        unsafe { sys::mapping_axes(self) }.into_iter().collect()
147    }
148
149    fn iter(&self, axes: &[AxisTerm], base: &Index, padding: bool) -> MappingIter {
150        unsafe { sys::mapping_iter(self, RSlice::from(axes), base, padding) }
151    }
152
153    fn carve(&self, piece: &Self) -> Self {
154        let configs =
155            sequence(&[piece], &[self], SequencerMode::Carve).expect("carve: piece must be contained in self");
156        let mut acc = Mapping::identity();
157        for (_key, entry) in configs[0].0.iter() {
158            let seg = if entry.memory_stride == 0 {
159                entry.mapping.clone() // unbacked: a live broadcast, or a Top hole left as padding
160            } else {
161                Mapping::identity().padding(entry.mapping.size(), PaddingKind::Top) // matched: Top pad, positions kept
162            };
163            acc = seg.pair(acc);
164        }
165        acc
166    }
167}

169/// Methods for [`SequencerConfig`].
170pub trait SequencerConfigExt {
171    /// Coalesces entries that sit back-to-back in the memory buffer into one, to minimize the
172    /// sequencer entry count. Result entries are keyed by cumulative-product stream stride, so
173    /// adjacent entries are always stream-contiguous; the merge condition is memory-contiguity
174    /// alone (`inner.memory_stride * inner.size == outer.memory_stride`). The merged entry keeps
175    /// the inner's key and memory stride, with the run paired into one mapping.
176    ///
177    /// Apply this PER descriptor (e.g. packet and time separately), never across, or it would
178    /// coalesce two entries that must stay distinct.
179    fn coalesce(self) -> Self;
180}
181
182impl SequencerConfigExt for SequencerConfig {
183    fn coalesce(self) -> SequencerConfig {
184        let mut out: Vec<(usize, SequencerEntry)> = Vec::new();
185        for (key, entry) in self.0 {
186            match out.last_mut() {
187                // Same memory and memory-contiguous: extend the run. This (outer) entry wraps the
188                // inner run; the run keeps the inner's key and memory stride, growing only its paired
189                // mapping. Entries in different memories (different spaces) never merge.
190                Some((_, run))
191                    if run.memory_index == entry.memory_index
192                        && run.memory_stride * run.mapping.size() == entry.memory_stride =>
193                {
194                    let inner = std::mem::replace(&mut run.mapping, Mapping::identity());
195                    run.mapping = entry.mapping.pair(inner);
196                }
197                _ => out.push((key, entry)),
198            }
199        }
200        SequencerConfig(out.into_iter().collect())
201    }
202}

204/// Methods for [`Index`].
205pub trait IndexExt: Sized {
206    /// Marks this index as invalid.
207    fn mark_invalid(&mut self);
208    /// Adds a mapping to this index.
209    fn add_mapping<I: crate::M>(&mut self, value: usize);
210    /// The terminal read: decodes every composite and returns each symbol's absolute coordinate
211    /// (`Ident` -> coordinate, coord-0 dropped), or the [`PaddingKind`] a pad cell lands on.
212    fn finalize(self) -> RResult<RSortedMap<Ident, usize>, PaddingKind>;
213}
214
215impl IndexExt for Index {
216    fn mark_invalid(&mut self) {
217        self.0 = RResult::RErr(PaddingKind::Bottom);
218    }
219    fn add_mapping<I: crate::M>(&mut self, value: usize) {
220        self.add(I::to_value().index(value));
221    }
222    fn finalize(self) -> RResult<RSortedMap<Ident, usize>, PaddingKind> {
223        unsafe { sys::index_finalize(self) }
224    }
