42/// Methods for [`Mapping`] backed by the impl crate over FFI. Pure `Mapping` operations live as
43/// inherent methods in `furiosa-mapping-types`; this trait carries only the ones that need the impl.
44pub trait MappingExt: Sized {
45    /// Normalizes to canonical form.
46    fn normalize(&self) -> Self;
47    /// The DMA burst tails for copying source `self` into destination `dst`, as
48    /// `(src_packet, dst_packet, valid)`: each side's own contiguous packet (its innermost padded
49    /// segment for the shared axis, padding included) plus the shared live size. Each side peels its own
50    /// packet, so the two may differ (a padded source vs a dense sink); `valid` is the largest run both
51    /// cover with live cells, no trailing padding, which a DRAM sink pins its tail to. `(1, 1, 1)` for a
52    /// transpose.
53    fn dma_tails(&self, dst: &Self) -> (usize, usize, usize);
54    /// Returns true if `self` is a resize (innermost prefix) of `original`.
55    fn is_resize_of(&self, original: &Self) -> bool;
56    /// Splits at buffer `target` into `(outer, inner)`: `outer` strides past the first `target`
57    /// cells, `inner` keeps them, and `outer.pair(inner)` reads the same buffer. `target` must
58    /// divide the size.
59    fn split_at(&self, target: usize) -> (Self, Self);
60    /// The read at buffer `position`, an [`Index`] of per-axis contributions with composites kept
61    /// WHOLE. Call [`IndexExt::finalize`] to decode them (`mapping.index(i).finalize()`).
62    fn index(&self, position: usize) -> Index;
63    /// The raw read at every buffer position `0..size`, composites kept WHOLE, in one FFI crossing. The
64    /// batch form of `index`; call [`IndexExt::finalize`] on a cell to decode it.
65    fn indexes(&self) -> Vec<Index>;
66    /// The live axis terms (its [`AxisTerm`]s, resolved symbols), padding excluded.
67    fn axes(&self) -> Vec<AxisTerm>;
68    /// A lazy iterator over where this mapping's cells land in a buffer laid out by `axes`, each offset
69    /// shifted by `base`: one `Option<usize>` per physical cell in canonical order (`Some(off)` live,
70    /// `None` padding). `padding` picks the traversal: `true` visits every physical cell (the wire order
71    /// the buffer seam needs), `false` skips the padding cells (faster on padded mappings, for relayouts).
72    ///
73    /// `axes` is the *target* buffer's layout (resolved symbol terms, e.g. from [`MappingExt::axes`]),
74    /// not necessarily this mapping's own axes: the mapping is projected
75    /// onto it. A symbol in both contributes its coordinate; a symbol only in the mapping contributes
76    /// nothing (it broadcasts, e.g. a dst axis absent from the src); a symbol only in `axes` stays at
77    /// its origin. The offsets land inside the buffer only if `axes` covers the mapping's reach: each
78    /// shared axis's extent must hold the mapping's coordinate for it (an out-of-range coordinate
79    /// wraps, like an out-of-range `base`).
80    ///
81    /// The returned [`MappingIter`] is an [`Iterator`]; it is fully owned (borrows nothing), so a
82    /// consumer just walks it (`.zip`, `.map`, `.flatten`) without ever naming the type.
83    fn iter(&self, axes: &[AxisTerm], base: &Index, padding: bool) -> MappingIter;
84    /// The leftover `self − piece` (the scatter/gather/reduce/broadcast residue): `self` is sequenced as
85    /// the STREAM against `piece` as memory, so a `self` cell `piece` backs becomes a `Top` pad and a cell
86    /// it does not back stays live (the broadcast). Runs in [`SequencerMode::Carve`], so a `Bottom` pad in
87    /// `self` (a `view_mut().tile()` hole) is tolerated as padding. Panics unless `piece` is in `self`.
88    fn carve(&self, piece: &Self) -> Self;
89}

91/// Matches each of `streams` against the `memories`, each its own address space (the fetch engine's
92/// `Time` and `Packet`), and returns one coalesced [`SequencerConfig`] per stream — each entry's
93/// `memory_index` says which memory it reads. The streams' segments are pooled and carved together
94/// (term priority is global across streams), so a `Broadcast` never claims a pad a `Term` needs.
95///
96/// Coverage is enforced: every memory must end fully consumed for `mode` (a live cell left unread under
97/// Read / unwritten under Write is [`SequencerError::Unconsumed`], carrying the carved-down memories so
98/// the caller can name the offender). Inputs are read-only; the carving happens on internal copies.
99pub fn sequence(
100    memories: &[&Mapping],
101    streams: &[&Mapping],
102    mode: SequencerMode,
103) -> Result<Vec<SequencerConfig>, SequencerError> {
104    let memories: RVec<Mapping> = memories.iter().map(|m| (*m).clone()).collect();
105    let streams: RVec<Mapping> = streams.iter().map(|m| (*m).clone()).collect();
106    let configs = unsafe { sys::mapping_sequence(memories.as_rslice(), streams.as_rslice(), mode) }.into_result()?;
107    Ok(configs.into_iter().map(SequencerConfigExt::coalesce).collect())
108}

110impl MappingExt for Mapping {
111    fn normalize(&self) -> Self {
112        unsafe { sys::mapping_normalize(self) }
113    }
114
115    fn dma_tails(&self, dst: &Self) -> (usize, usize, usize) {
116        let Tuple3(src_packet, dst_packet, valid) = unsafe { sys::mapping_dma_tails(self, dst) };
117        (src_packet, dst_packet, valid)
118    }
119
120    fn is_resize_of(&self, original: &Self) -> bool {
121        let n = self.size();
122        n <= original.size() && self.normalize() == original.clone().resize(n).normalize()
123    }
124
125    fn split_at(&self, target: usize) -> (Self, Self) {
126        let Tuple2(outer, inner) = unsafe { sys::mapping_split_at(self, target) };
127        (outer, inner)
128    }
129
130    fn index(&self, position: usize) -> Index {
131        unsafe { sys::mapping_index(self, position) }
132    }
133
134    fn indexes(&self) -> Vec<Index> {
135        unsafe { sys::mapping_indexes(self) }.into_iter().collect()
136    }
137
138    fn axes(&self) -> Vec<AxisTerm> {
139        unsafe { sys::mapping_axes(self) }.into_iter().collect()
140    }
141
142    fn iter(&self, axes: &[AxisTerm], base: &Index, padding: bool) -> MappingIter {
143        unsafe { sys::mapping_iter(self, RSlice::from(axes), base, padding) }
144    }
145
146    fn carve(&self, piece: &Self) -> Self {
147        let configs =
148            sequence(&[piece], &[self], SequencerMode::Carve).expect("carve: piece must be contained in self");
149        let mut acc = Mapping::identity();
150        for (_key, entry) in configs[0].0.iter() {
151            let seg = if entry.memory_stride == 0 {
152                entry.mapping.clone() // unbacked: a live broadcast, or a Top hole left as padding
153            } else {
154                Mapping::identity().padding(entry.mapping.size(), PaddingKind::Top) // matched: Top pad, positions kept
155            };
156            acc = seg.pair(acc);
157        }
158        acc
159    }
160}

162/// Methods for [`SequencerConfig`].
163pub trait SequencerConfigExt {
164    /// Coalesces entries that sit back-to-back in the memory buffer into one, to minimize the
165    /// sequencer entry count. Result entries are keyed by cumulative-product stream stride, so
166    /// adjacent entries are always stream-contiguous; the merge condition is memory-contiguity
167    /// alone (`inner.memory_stride * inner.size == outer.memory_stride`). The merged entry keeps
168    /// the inner's key and memory stride, with the run paired into one mapping.
169    ///
170    /// Apply this PER descriptor (e.g. packet and time separately), never across, or it would
171    /// coalesce two entries that must stay distinct.
172    fn coalesce(self) -> Self;
173}
174
175impl SequencerConfigExt for SequencerConfig {
176    fn coalesce(self) -> SequencerConfig {
177        let mut out: Vec<(usize, SequencerEntry)> = Vec::new();
178        for (key, entry) in self.0 {
179            match out.last_mut() {
180                // Same memory and memory-contiguous: extend the run. This (outer) entry wraps the
181                // inner run; the run keeps the inner's key and memory stride, growing only its paired
182                // mapping. Entries in different memories (different spaces) never merge.
183                Some((_, run))
184                    if run.memory_index == entry.memory_index
185                        && run.memory_stride * run.mapping.size() == entry.memory_stride =>
186                {
187                    let inner = std::mem::replace(&mut run.mapping, Mapping::identity());
188                    run.mapping = entry.mapping.pair(inner);
189                }
190                _ => out.push((key, entry)),
191            }
192        }
193        SequencerConfig(out.into_iter().collect())
194    }
195}

197/// Methods for [`Index`].
198pub trait IndexExt: Sized {
199    /// Marks this index as invalid.
200    fn mark_invalid(&mut self);
201    /// Adds a mapping to this index.
202    fn add_mapping<I: crate::M>(&mut self, value: usize);
203    /// The terminal read: decodes every composite and returns each symbol's absolute coordinate
204    /// (`Ident` -> coordinate, coord-0 dropped), or the [`PaddingKind`] a pad cell lands on.
205    fn finalize(self) -> RResult<RSortedMap<Ident, usize>, PaddingKind>;
206}
207
208impl IndexExt for Index {
209    fn mark_invalid(&mut self) {
210        self.0 = RResult::RErr(PaddingKind::Bottom);
211    }
212    fn add_mapping<I: crate::M>(&mut self, value: usize) {
213        self.add(I::to_value().index(value));
214    }
215    fn finalize(self) -> RResult<RSortedMap<Ident, usize>, PaddingKind> {
216        unsafe { sys::index_finalize(self) }
217    }
