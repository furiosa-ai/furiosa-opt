1//! Hardware constraints shared across engine tensor constructors.
2//!
3//! Each engine's `Tensor::new` validates its inputs against these. The values
4//! and checks live here, in one place, so the rules are stated once and every
5//! engine refers to the same definitions instead of repeating literals. Hardware parameters shared
6//! with the published verifiers are re-exported from [`furiosa_opt_lower`] so a value is defined once
7//! across both crates.
8//!
9//! Each check is a plain `fn` whose body is a single `const { ... }` block, so the compile-time
10//! enforcement lives at the leaf and call sites stay plain calls (`assert_cluster_size::<Cluster>()`).
11//! Because each check is its own `const` item, several violated checks all report in one compile
12//! (a single shared `const {}` block would stop at the first panic).
13//!
14//! Interface (the checks call sites invoke) is at the top; the per-axis leaf helpers the
15//! dimension-preserving checks build on, and the byte-size utility, are at the bottom.
16

21/// Bits in a byte / flit byte size, single-sourced from the published verifier crate.
22pub(crate) use furiosa_opt_lower::{BITS_PER_BYTE, FLIT_BYTES};

24/// Supported `Cluster` dimension sizes.
25pub(crate) const CLUSTER_SIZES: [usize; 2] = [1, 2];
26
27/// Byte granularity a packet must align to in the [`Fetch`](crate::engine) and
28/// [`Switch`](crate::engine) engines, which pass packets through unchanged.
29pub(crate) const PACKET_ALIGN_BYTES: usize = 8;
30
31/// Supported `Slice` dimension sizes, in bytes.
32pub(crate) const SLICE_SIZES: [usize; 3] = [64, 128, 256];
33
34/// Asserts the `Cluster` dimension is one of the [`CLUSTER_SIZES`].
35pub(crate) fn assert_cluster_size<Cluster: M>() {
36    const {
37        let mut i = 0;
38        let mut ok = false;
39        while i < CLUSTER_SIZES.len() {
40            ok |= CLUSTER_SIZES[i] == Cluster::SIZE;
41            i += 1;
42        }
43        assert!(ok, "Cluster size must be 1 or 2");
44    };
45}
46
47/// Asserts the `Slice` dimension is one of the [`SLICE_SIZES`].
48pub(crate) fn assert_slice_size<Slice: M>() {
49    const {
50        let mut i = 0;
51        let mut ok = false;
52        while i < SLICE_SIZES.len() {
53            ok |= SLICE_SIZES[i] == Slice::SIZE;
54            i += 1;
55        }
56        assert!(ok, "Slice size must be one of 64 | 128 | 256");
57    };
58}
59
60/// Asserts a packet is a whole number of [`PACKET_ALIGN_BYTES`]-byte access words (Switch input).
61pub(crate) fn assert_packet_aligned_by_access_width<D: Scalar, Packet: M>() {
62    const {
63        assert!(
64            size_in_bytes(D::BITS, Packet::SIZE).is_multiple_of(PACKET_ALIGN_BYTES),
65            "Packet size must be 8-byte aligned"
66        );
67    };
68}
69
70/// Asserts a packet is a non-zero multiple of [`PACKET_ALIGN_BYTES`] bytes, at most one flit
71/// (Commit trim output: 8 | 16 | 24 | 32 bytes).
72pub(crate) fn assert_packet_aligned_by_access_width_max_flit<D: Scalar, Packet: M>() {
73    const {
74        let bytes = size_in_bytes(D::BITS, Packet::SIZE);
75        assert!(
76            bytes != 0 && bytes.is_multiple_of(PACKET_ALIGN_BYTES) && bytes <= FLIT_BYTES,
77            "Packet size must be 8, 16, 24, or 32 bytes"
78        );
79    };
80}
81
82/// Asserts a packet is exactly one flit ([`FLIT_BYTES`] bytes) (Collect, the compute engines, and
83/// the Contraction lane output).
84pub(crate) fn assert_packet_one_flit<D: Scalar, Packet: M>() {
85    const {
86        assert!(
87            size_in_bytes(D::BITS, Packet::SIZE) == FLIT_BYTES,
88            "Packet size must be exactly one flit (32 bytes)"
89        );
90    };
91}
92
93/// Asserts a packet is one or two flits (Contraction `outer` output: 32 or 64 bytes).
94pub(crate) fn assert_packet_one_or_two_flit<D: Scalar, Packet: M>() {
95    const {
96        let bytes = size_in_bytes(D::BITS, Packet::SIZE);
97        assert!(
98            bytes == FLIT_BYTES || bytes == 2 * FLIT_BYTES,
99            "Packet size must be one or two flits (32 or 64 bytes)"
100        );
101    };
102}

104/// Asserts a DM → DM transfer preserves its whole `Chip` / `Cluster` / `Slice` partition.
105///
106/// Shared by every `DmTensor`(`View`/`ViewMut`) → `DmTensor`(`View`/`ViewMut`) primitive. Each leaf
107/// check is its own `const` item, so a mismatch on any axis reports its own error (all in one
108/// compile); an axis carried through unchanged passes its check trivially.
109pub(crate) fn assert_dm_to_dm_dimension_preserved<Chip: M, Chip2: M, Cluster: M, Cluster2: M, Slice: M, Slice2: M>() {
110    assert_chip_preserved::<Chip, Chip2>();
111    assert_cluster_preserved::<Cluster, Cluster2>();
112    assert_slice_preserved::<Slice, Slice2>();
113}
114
115/// Asserts a `reshape` preserves its whole `Chip` / `Cluster` / `Slice` partition (it relabels the
116/// mapping, it moves no data). Shares the per-axis leaf checks with [`assert_dm_to_dm_dimension_preserved`].
117pub(crate) fn assert_reshape_dimension_preserved<Chip: M, Chip2: M, Cluster: M, Cluster2: M, Slice: M, Slice2: M>() {
118    assert_chip_preserved::<Chip, Chip2>();
119    assert_cluster_preserved::<Cluster, Cluster2>();
120    assert_slice_preserved::<Slice, Slice2>();
121    // TODO: also assert the `Element` size is preserved (add `Element` / `Element2` params and
122    // `assert_element_preserved`). Deferred because some current examples reshape with a
123    // mismatched `Element`; enable once those are fixed.
124}
125
126/// Asserts a HBM `reshape` preserves its whole `Chip` / `Element` partition (it relabels the
127/// mapping, it moves no data). HBM has no `Cluster`/`Slice` axis, and (unlike
128/// [`assert_reshape_dimension_preserved`]'s DM case) no existing example reshapes a mismatched
129/// `Element`, so both axes are checked here.
130pub(crate) fn assert_hbm_reshape_dimension_preserved<Chip: M, Chip2: M, Element: M, Element2: M>() {
131    assert_chip_preserved::<Chip, Chip2>();
132    assert_element_preserved::<Element, Element2>();
133}

135/// Asserts the `Chip` size is preserved across a transfer / reshape.
136fn assert_chip_preserved<Chip: M, Chip2: M>() {
137    const { assert!(Chip::SIZE == Chip2::SIZE, "Chip size must be preserved") };
138}

155/// Byte size of `length` elements of a `bits`-wide scalar.
156///
157/// `const fn` twin of [`Scalar::size_in_bytes_from_length`](crate::scalar::Scalar::size_in_bytes_from_length),
158/// callable from the `const { ... }` constraint checks. Panics if the total bit count is not byte-aligned.
159pub(crate) const fn size_in_bytes(bits: usize, length: usize) -> usize {
160    assert!(
161        (length * bits).is_multiple_of(BITS_PER_BYTE),
162        "total bits must be byte-aligned"
163    );
164    (length * bits) / BITS_PER_BYTE
165}
