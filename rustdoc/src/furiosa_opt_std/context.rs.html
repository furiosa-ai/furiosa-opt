82impl Context {
83    /// Acquire the tensor units. Chain [`Acquired::bind`] to fix the NPU device before any host
84    /// I/O (`to_hbm`/`from_hbm`) initializes the runtime.
85    pub fn acquire() -> Acquired {
86        static SINGLETON: LazyLock<Mutex<Context>> = LazyLock::new(|| {
87            Mutex::new(Context {
88                main: TuContext::<{ Tu::Main }> { _marker: PhantomData },
89                sub: TuContext::<{ Tu::Sub }> { _marker: PhantomData },
90                tdma: DmaContext::<{ Dma::Tensor }> { _marker: PhantomData },
91                pdma: DmaContext::<{ Dma::Pcie }> { _marker: PhantomData },
92            })
93        });
94        Acquired(SINGLETON.lock().unwrap_or_else(|poisoned| poisoned.into_inner()))
95    }
96}

115impl Acquired {
116    /// Binds the process to a kernel's logical [`Device`] (`kernel.device()`) before any host I/O
117    /// initializes the runtime. First bind wins; a conflicting one panics. Non-NPU backends ignore
118    /// the device.
119    pub fn bind(self, device: Device) -> Self {
120        crate::runtime::CurrentBackend::bind_device(device);
121        self
122    }

125impl<const T: Tu> TuContext<{ T }> {
126    /// Begin a tensor unit operation in this context.
127    #[primitive(TuContext::begin)]
128    pub fn begin<'l, D: MaterializableScalar, Chip: M, Cluster: M, Slice: M, Element: M>(
129        &'l mut self,
130        tensor: DmTensorView<'l, D, Chip, Cluster, Slice, Element>,
131    ) -> BeginTensor<'l, { T }, D, Chip, Cluster, Slice, Identity, Element> {
132        // The mappings differ only by `Identity`; `transmute` is a safe rewrap after the storage
133        // mapping erasure.
134        BeginTensor::new(self, tensor.inner.read().transmute())
135    }
136
137    /// Begin a tensor unit operation in this context with interleaved tensors.
138    #[primitive(TuContext::begin_interleaved)]
139    pub fn begin_interleaved<'l, I: AxisName, D: Scalar, Chip: M, Cluster: M, Slice: M, Element: M>(
140        &'l mut self,
141        lhs: DmTensorView<'l, D, Chip, Cluster, Slice, Element>,
142        rhs: DmTensorView<'l, D, Chip, Cluster, Slice, Element>,
143    ) -> BeginTensor<'l, { T }, D, Chip, Cluster, Slice, Symbol<I>, Element> {
144        let mut output = Tensor::<D, m![{ Chip }, { Cluster }, { Slice }, { Symbol<I> }, { Element }]>::uninit();
145
146        for (i, input) in [lhs, rhs].into_iter().enumerate() {
147            output
148                .view_mut()
149                .tile::<Symbol<I>, m![{ Chip }, { Cluster }, { Slice }, 1 #{!} 2, { Element }], 1>(i)
150                .transpose(input.inner, false);
151        }
152
153        BeginTensor::new(self, output)
154    }
155}

157impl TuContext<{ Tu::Sub }> {
158    /// Asymmetric cluster slice via ParallelCopy (stos): each cluster selects its own slice
159    /// position from `slice_indices` (one per cluster) — e.g. `[1, 0]` gives cluster 0 slice 1,
160    /// cluster 1 slice 0.
161    ///
162    /// `AxisToSlice` must be the outermost axis in `Element`.
163    #[primitive(TuContext::parallel_copy_cluster_slice)]
164    pub fn parallel_copy_cluster_slice<
165        'l,
166        const CLUSTER_DIM: usize,
167        AxisToSlice: M,
168        AxisSlicedElement: M,
169        Element2: M,
170        D: Scalar,
171        Chip: M,
172        Cluster: M,
173        Slice: M,
174        Element: M,
175    >(
176        &mut self,
177        tensor: DmTensorView<'l, D, Chip, Cluster, Slice, Element>,
178        slice_indices: &[usize; CLUSTER_DIM],
179    ) -> super::tensor::memory::DmTensor<D, Chip, Cluster, Slice, Element2> {
180        let mut sliced = unsafe { DmTensor::from_addr(0) };
181
182        for (cluster_idx, slice_idx) in slice_indices.iter().enumerate() {
183            let cluster_slice = tensor.cluster_tile::<Cluster, 1, Padding<Identity, CLUSTER_DIM>>(cluster_idx);
184            cluster_slice
185                .tile::<AxisToSlice, 1, AxisSlicedElement>(*slice_idx)
186                .to_dm_view_pcopy(
187                    self,
188                    sliced
189                        .view_mut()
190                        .cluster_tile::<Cluster, 1, Padding<Identity, CLUSTER_DIM, { PaddingKind::Bottom }>>(
191                            cluster_idx,
192                        ),
193                );
194        }
195
196        sliced
197    }
198
199    /// Asymmetric chip slice via ParallelCopy (stos): each chip selects its own slice position
200    /// from `slice_indices` (one per chip) — e.g. `[3, 0, 1, 2]` gives chip 0 slice 3, chip 1
201    /// slice 0, chip 2 slice 1, chip 3 slice 2.
202    #[primitive(TuContext::parallel_copy_chip_slice)]
203    pub fn parallel_copy_chip_slice<
204        'l,
205        const CHIP_DIM: usize,
206        AxisToSlice: M,
207        AxisSlicedElement: M,
208        Element2: M,
209        D: Scalar,
210        Chip: M,
211        Cluster: M,
212        Slice: M,
213        Element: M,
214    >(
215        &mut self,
216        tensor: DmTensorView<'l, D, Chip, Cluster, Slice, Element>,
217        slice_indices: &[usize; CHIP_DIM],
218    ) -> DmTensor<D, Chip, Cluster, Slice, Element2> {
219        let mut sliced = unsafe { DmTensor::from_addr(0) };
220
221        for (chip_idx, slice_idx) in slice_indices.iter().enumerate() {
222            let chip_slice = tensor.chip_tile::<Chip, 1, Padding<Identity, CHIP_DIM>>(chip_idx);
223            chip_slice
224                .tile::<AxisToSlice, 1, AxisSlicedElement>(*slice_idx)
225                .to_dm_view_pcopy(
226                    self,
227                    sliced
228                        .view_mut()
229                        .chip_tile::<Chip, 1, Padding<Identity, CHIP_DIM, { PaddingKind::Bottom }>>(chip_idx),
230                );
231        }
232
233        sliced
234    }
