13// ANCHOR: contract_time_def
14impl<'l, const T: Tu, D: Scalar, Chip: M, Cluster: M, Slice: M, Lane: M, Time: M, Packet: M, B: Backend>
15    ContractPacketTensor<'l, T, D, Chip, Cluster, Slice, Lane, Time, Packet, B>
16{
17    /// Accumulates per-cycle contractions over the `Time` dimension via the shared
18    /// accumulator buffer, shrinking input `Time` to `OutTime`. The axes present in
19    /// `Time` but absent from `OutTime` are reduce-added.
20    #[primitive(ContractPacketTensor::contract_time)]
21    pub fn contract_time<OutTime: M>(
22        self,
23    ) -> ContractTimeTensor<'l, T, D, Chip, Cluster, Slice, Lane, OutTime, Packet, B> {
24        verify_contract_time(Time::to_value(), OutTime::to_value());
25        ContractTimeTensor {
26            ctx: self.ctx,
27            inner: self.inner.reduce_add(),
28            pre_reduce_time: Time::to_value(),
29        }
30    }
31}

34/// Validates the `OutTime ⊆ Time` constraint for `.contract_time()`.
35///
36/// The axes in `OutTime` must be a subset of the axes in `Time`, with their
37/// relative order preserved. Axes present in `Time` but absent from `OutTime`
38/// are summed away by `reduce_add`. Retained axes must also preserve their
39/// padding from `Time`.
40pub(crate) fn verify_contract_time(time: Mapping, out_time: Mapping) {
41    // The outer portion of `Time` should divide `Time`.
42    // Some axes can be reduced in temporal accumulation.
43    let division_terms = config_divide_exact(&time, &out_time).unwrap_or_else(|_| {
44        panic!(
45            "contract_time: OutTime mismatch. Some axes present in Time are not present in OutTime: {time}, {out_time}"
46        )
47    });
48    // Non-reduced axes must have their order preserved in `OutTime`.
49    assert!(
50        division_terms
51            .windows(2)
52            .all(|w| w[0].divisor_stride > w[1].divisor_stride),
53        "contract_time: OutTime axes must follow the same order as the Time axes"
54    );
55
56    // Each retained axis in `out_time` must preserve its padding from `time`: the padded extent at
57    // each cumulative stride (`padding_per_stride`) must line the retained axes up edge-to-edge.
58    let time_padding_per_stride = padding_per_stride(&time);
59
60    // Sort retained axes inner-to-outer by their position in `out_time` (divisor).
61    let mut sorted_divisions: Vec<&DivideTerm> = division_terms.iter().collect();
62    sorted_divisions.sort_by_key(|d| d.divisor_stride);
63
64    // The first divisor should have a stride of 1.
65    if let Some(first) = sorted_divisions.first() {
66        assert_eq!(
67            first.divisor_stride, 1,
68            "contract_time: Padding mismatch. \
69             OutTime {out_time} has unexpected leading padding not present in Time {time}"
70        );
71    }
72
73    // For each retained axis, its padding end must equal the start of the next retained axis.
74    for (pos, d) in sorted_divisions.iter().enumerate() {
75        let expected_end = d.divisor_stride
76            * time_padding_per_stride
77                .get(&d.dividend_stride)
78                .copied()
79                .unwrap_or(d.resize);
80        let end = sorted_divisions
81            .get(pos + 1)
82            .map_or(out_time.size(), |next| next.divisor_stride);
83        assert_eq!(
84            expected_end, end,
85            "contract_time: Padding mismatch. \
86             Non-reduced axes in OutTime {out_time} do not preserve padding from Time {time}"
87        );
88    }
89}
