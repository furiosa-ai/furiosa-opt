1//! Contraction Engine: Outer -> Packet Reducer -> Time Reducer -> Lane Folder.
2//!
3//! Submodules:
4//! - [`outer`]: stashes the two un-broadcast operands as a [`LazyContraction`] (`.contract_outer`).
5//! - [`packet`]: Packet Reducer (`.contract_packet`).
6//! - [`time`]: Time Reducer (`.contract_time`).
7//! - [`lane`]: Lane Folder (`.contract_lane`).
8//!
9//! Output: a [`ContractTensor`], at [`PositionContraction`].
10//!
11//! The engine always defers: a [`LazyContraction`] carrier threads through all four stages untouched and
12//! is finalized once, at the Lane Folder, with one fused [`crate::backend::Backend::contraction`].
13//! See [`LazyContraction`].
14

36/// The two un-broadcast operands (already widened to the contraction output type) plus the mappings
37/// the Lane Folder needs to fuse them, stashed by the Outer stage in place of a materialized product.
38/// This is the only inner value a contraction stage carries: the Packet/Time reducers pass it through
39/// untouched, and [`lane::contract_lane`] (via [`crate::backend::Backend::contraction`]) finalizes it
40/// with one fused multiply-reduce, allocating zero product cells — so a wide-output GEMM never
41/// materializes the hundreds-of-MB `pre_reduce`-shaped outer product.
42#[derive(Debug, Clone)]
43pub(crate) struct LazyContraction<D: Scalar, B: Backend> {
44    pub(crate) lhs: B::Storage<D>,
45    pub(crate) rhs: B::Storage<D>,
46    /// `lhs`'s own compact layout. [`crate::storage::buf::BufStorage`] reads its strides from this;
47    /// [`crate::storage::math::MathStorage`] carries its own axes and ignores it.
48    pub(crate) lhs_map: MappingValue,
49    /// See `lhs_map`.
50    pub(crate) rhs_map: MappingValue,
51    /// The full (un-reduced) index space the fold reduces onto the Lane Folder's rebuilt `out`.
52    pub(crate) pre_reduce: MappingValue,
53}

55/// Number of columns in the temporal accumulator buffer, single-sourced from the published verifier crate.
56pub(crate) use furiosa_opt_lower::TEMPORAL_ACCUMULATOR_COLS;
57
58/// Asserts a reduce packet's element count is a power of two and at most
59/// [`TEMPORAL_ACCUMULATOR_COLS`] (`packet_reduce` / `time_reduce` output).
60fn assert_packet_pow2_within_accumulator_cols<Packet: M>() {
61    const {
62        let size = Packet::SIZE;
63        assert!(
64            size != 0 && size & (size - 1) == 0 && size <= TEMPORAL_ACCUMULATOR_COLS,
65            "Packet element count must be a power of two and at most the accumulator column count (32)"
66        );
67    };
68}

76/// Intermediate tensor after the Packet Reducer (reduce-add within `Packet`),
77/// before the Time Reducer.
78#[derive(Debug)]
79pub struct ContractPacketTensor<
80    'l,
81    const T: Tu,
82    D: Scalar,
83    Chip: M,
84    Cluster: M,
85    Slice: M,
86    Lane: M,
87    Time: M,
88    Packet: M,
89    B: Backend = CurrentBackend,
90> {
91    pub(crate) ctx: &'l mut TuContext<{ T }>,
92    /// The deferred carrier (see [`LazyContraction`]); this stage only re-types it to the post-Packet `OutPacket`.
93    pub(crate) inner: LazyContraction<D, B>,
94    /// Axis typestate the stage transitions carry; `inner` is mapping-free.
95    pub(crate) _axes: PhantomData<(Chip, Cluster, Slice, Lane, Time, Packet)>,
96}

115    pub(crate) inner: LazyContraction<D, B>,

119    /// Axis typestate the stage transitions carry; `inner` is mapping-free.
120    pub(crate) _axes: PhantomData<(Chip, Cluster, Slice, Lane, Time, Packet)>,
121}

123impl<'l, const T: Tu, D: Scalar, Chip: M, Cluster: M, Slice: M, Lane: M, Time: M, Packet: M, B: Backend>

126    fn check_constraints() {

136            ctx,
137            inner,
138            _axes: PhantomData,
139        }
140    }
