1//! Outer stage: the entry to the ONE deferred contraction algorithm.
2//!
3//! `contract_outer` validates both operand sides ([`stream_adapter::verify_stream_adapter`],
4//! [`trf_sequencer::verify_trf_sequencer`]) and stashes the two un-broadcast operands (widened to the
5//! contraction output type) plus their layouts in a [`super::LazyContraction`], WITHOUT materializing the
6//! `[.., Lane, ..]`-shaped outer product. The Packet/Time reducers thread it through unchanged, and
7//! [`super::lane`] finalizes it with one fused [`crate::backend::Backend::contraction`].
8//!
9//! The widen happens *entering* Outer: `contract_outer` takes storage-typed operands `D` and stashes
10//! them cast to the widened accumulator [`ContractionCast::Output`]. That widened type is the carrier's
11//! *own* type parameter, so every downstream stage carries the accumulator width verbatim; the narrow
12//! back to a storage dtype happens at the next [`crate::engine::cast::CastTensor`] the consumer applies.
13
14pub(super) mod stream_adapter;
15pub(super) mod trf_sequencer;

33/// Output of the Outer stage: the deferred contraction carrier (the two un-broadcast operands, widened
34/// to the accumulator type), threaded through the reducers and fused at the Lane Folder.
35///
36/// Two dtype params, related by the `Storage: ContractionCast<Output = D>` bound that pins them:
37/// - `D` is the *widened accumulator* the operands are carried in (`i4`/`i8` -> `i32`, `f8`/`bf16` -> `f32`).
38/// - `Storage` is the pre-widen operand dtype. It is retained only so the Packet Reducer's physical-flit
39///   check ([`super::packet::verify_contract_packet`]) can size the DPE *input* packet in storage bytes,
40///   which the widened `D` no longer encodes (`Storage` is not recoverable from `D` — the widen is
41///   many-to-one, so this second param is load-bearing, not redundant).
42///
43/// See the module docs for where the widen and the (downstream) narrow happen.
44#[derive(Debug)]
45pub struct ContractOuterTensor<
46    'l,
47    const T: Tu,
48    D: Scalar,
49    Storage: ContractionCast<Output = D>,
50    Chip: M,
51    Cluster: M,
52    Slice: M,
53    Lane: M,
54    Time: M,
55    Packet: M,
56    B: Backend = CurrentBackend,
57> {
58    pub(crate) ctx: &'l mut TuContext<{ T }>,
59    /// The deferred dense-GEMM carrier: the two compact operands (already widened to the contraction
60    /// output type) plus the mappings [`super::lane::contract_lane`] fuses them with. `D` is already
61    /// the widened accumulator (`Storage: ContractionCast<Output = D>`), so the carrier is keyed on `D`.
62    pub(crate) inner: LazyContraction<D, B>,
63    /// The pre-widen storage dtype, retained for the Packet Reducer's flit check (see struct doc).
64    pub(crate) _storage: PhantomData<Storage>,
65    /// Axis typestate the stage transitions carry; `inner` is mapping-free ([`LazyContraction`]).
66    pub(crate) _axes: PhantomData<(Chip, Cluster, Slice, Lane, Time, Packet)>,
67}
68
69impl<
70    'l,
71    const T: Tu,
72    D: Scalar,
73    Storage: ContractionCast<Output = D>,
74    Chip: M,
75    Cluster: M,
76    Slice: M,
77    Lane: M,
78    Time: M,
79    Packet: M,
80    B: Backend,
81> ContractOuterTensor<'l, T, D, Storage, Chip, Cluster, Slice, Lane, Time, Packet, B>
82{
83    fn check_constraints() {
84        constraints::assert_cluster_size::<Cluster>();
85        constraints::assert_slice_size::<Slice>();
86        // The DPE reads storage-width input flits, then widens; size the packet in `Storage` bytes.
87        constraints::assert_packet_one_or_two_flit::<Storage, Packet>();
88    }
89
90    #[doc(hidden)]
91    pub(crate) fn new(ctx: &'l mut TuContext<{ T }>, inner: LazyContraction<D, B>) -> Self {
92        Self::check_constraints();
93        Self {
94            ctx,
95            inner,
96            _storage: PhantomData,
97            _axes: PhantomData,
98        }
99    }
100}

102// ANCHOR: contract_outer_def
103impl<
104    'l,
105    const T: Tu,
106    P: CanApplyContractOuter,
107    D: Scalar + ContractionCast,
108    Chip: M,
109    Cluster: M,
110    Slice: M,
111    Time: M,
112    Packet: M,
113    B: Backend,
114> TuTensor<'l, T, P, D, Chip, Cluster, Slice, Time, Packet, B>
115{
116    /// Runs the Outer stage: stashes the two un-broadcast operands (widened to the accumulator type)
117    /// and the layouts [`super::lane::contract_lane`] needs to fuse them into a [`LazyContraction`]. No
118    /// materializing alternative, no per-backend branch -- every backend fuses the same way.
119    #[primitive(TuTensor::contract_outer)]
120    pub fn contract_outer<OutTime: M, OutPacket: M, Lane: M, TrfElement: M, TrfD>(
121        self,
122        trf_tensor: &TrfTensor<TrfD, Chip, Cluster, Slice, Lane, TrfElement, B>,
123    ) -> ContractOuterTensor<'l, T, <D as ContractionCast>::Output, D, Chip, Cluster, Slice, Lane, OutTime, OutPacket, B>
124    where
125        D: Cast<<D as ContractionCast>::Output>,
126        // The weight (TRF) type must form a valid contraction-engine operand pair with the
127        // stream type `D`: same type, or a mixed integer precision within a
128        // family (i4/i5 x i4/i5, i8/i9 x i8/i9). Both operands widen to the
129        // stream's accumulator for the multiply.
130        TrfD: Scalar + ContractionWeight<D> + Cast<<D as ContractionCast>::Output>,
131    {
132        type Out<D> = <D as ContractionCast>::Output;
133
134        // Skipping the broadcast transpose does not skip its validity contract -- a malformed
135        // contraction would otherwise slip past these asserts and panic far downstream instead.
136        stream_adapter::verify_stream_adapter::<D, Lane, Time, Packet, OutTime, OutPacket>();
137        trf_sequencer::verify_trf_sequencer::<TrfD, Lane, TrfElement, OutTime, OutPacket>();
138
139        // The operands keep their own compact layouts: lhs is `[Chip, Cluster, Slice, Time, Packet]`
140        // (`self.inner`), rhs is `[Chip, Cluster, Slice, Lane, TrfElement]` (`trf_tensor`). A
141        // bare-buffer backend reads its strides from these; `MathStorage` ignores them (its axes live
142        // in the storage).
143        let lhs_map = <m![{ Chip }, { Cluster }, { Slice }, { Time }, { Packet }]>::to_value();
144        let rhs_map = <m![{ Chip }, { Cluster }, { Slice }, { Lane }, { TrfElement }]>::to_value();
145        let pre_reduce = <m![{ Chip }, { Cluster }, { Slice }, { Lane }, { OutTime }, { OutPacket }]>::to_value();
146
147        // Widen each operand to the `Out<D>` accumulator up front, at the operand's own (compact) size,
148        // not pre_reduce's -- the fold at `contract_lane` then runs entirely in `Out<D>`. Parity-identical
149        // to the old per-cell widen (same `Cast::cast` per element), it just never allocates the
150        // pre_reduce-shaped broadcast this stage used to build.
151        //
152        // `map_bounded`, NOT the plain `map`: `D` (the stream side) and `TrfD` (the TRF side) may each
153        // legitimately be a non-`MaterializableScalar` staging type (`i5`/`i9`, produced by
154        // `fetch_zero_point_sub` -- "an i5/i9 may still be a contraction weight resident in the TRF" per
155        // its own doc). Such a type's storage-native length recovery over-reports (its `BITS` names a
156        // real-hardware wire width, disconnected from its host in-memory size), so a plain `map`'s
157        // internal whole-buffer walk reads/writes past the buffer once the count crosses the true
158        // element count -- this is the ONE place either operand is read back as a whole tensor before
159        // `contract_lane`'s fused `Backend::contraction` (its only other lifetime is written-once by
160        // `fetch_zero_point_sub`, never read back that way), so it is also the one place this matters.
161        let contraction = LazyContraction {
162            lhs: self.inner.map_bounded(|v| -> Out<D> { v.cast() }).inner,
163            rhs: trf_tensor.inner.map_bounded(|v| -> Out<D> { v.cast() }).inner,
164            lhs_map,
165            rhs_map,
166            pre_reduce,
167        };
168        ContractOuterTensor::new(self.ctx, contraction)
169    }
170}
