1//! Type-level occupancy of the VE stash slot, and the single source of truth for its lifecycle.
2//!
3//! RNGD's stash (operand register) is write-once and read-once per VE pass. [`StashSlot`] encodes
4//! that at compile time as a three-state machine, with exactly these transitions and no others:
5//!
6//! ```text
7//! Fresh --vector_stash--> Occupied<D, Mapping> --Stash read--> Spent
8//! ```
9//!
10//! - [`Fresh`]: never written. `vector_stash` is defined only here, so the write happens at most
11//!   once.
12//! - [`Occupied`]: holds a `Tensor<D, Mapping>`. The [`Stash`](crate::prelude::Stash) operand reads
13//!   it, and that read is defined only on `Occupied`, so the read happens at most once.
14//! - [`Spent`]: written and already read. Empty like `Fresh`, but a read lands here rather than
15//!   back in `Fresh`, so neither a second write (no `vector_stash` on `Spent`) nor a second read
16//!   (no read transition from `Spent`) has an impl to match.
17//!
18//! Docs elsewhere (`vector_stash`, `StashTransition`, `VeState::consume_stash`) reference this
19//! module rather than restating the machine.
20
21use std::fmt::Debug;
22
23use crate::engine::vector::scalar::VeScalar;
24use crate::tensor::Tensor;
25use furiosa_mapping::M;

27/// Occupancy of the stash slot at compile time. `D` ties an occupied slot's scalar to the
28/// pipeline's, so a stash read is checked against the current `D`. See the [module docs] for the
29/// full lifecycle.
30///
31/// [module docs]: crate::engine::vector::stash_slot
32pub trait StashSlot<D: VeScalar>: Debug {
33    /// Clones the slot's tensor (transposed to `TargetMapping`); `None` for the empty states
34    /// ([`Fresh`] / [`Spent`]). Only [`Occupied`] overrides the default.
35    fn clone_tensor_as<TargetMapping: M>(&self) -> Option<Tensor<D, TargetMapping>> {
36        None
37    }
38}
39
40/// The stash slot is empty and write-armed: never written this pass, so `vector_stash` is
41/// available here (and only here). Named `Fresh` rather than a second "empty" word so a call site
42/// reads which empty state still accepts a write ([`Fresh`]) versus which does not ([`Spent`]). See
43/// the [module docs].
44///
45/// [module docs]: crate::engine::vector::stash_slot
46#[derive(Debug)]
47pub struct Fresh;
48impl<D: VeScalar> StashSlot<D> for Fresh {}
49
50/// The stash slot was written and already read this pass. See the [module docs] for why a read
51/// lands here rather than in [`Fresh`].
52///
53/// [module docs]: crate::engine::vector::stash_slot
54#[derive(Debug)]
55pub struct Spent;
56impl<D: VeScalar> StashSlot<D> for Spent {}
57
58/// The stash slot holds a [`Tensor<D, Mapping>`]. See the [module docs].
59///
60/// [module docs]: crate::engine::vector::stash_slot
61#[derive(Debug)]
62pub struct Occupied<D: VeScalar, Mapping: M> {
63    data: Tensor<D, Mapping>,
64}
65
66impl<D: VeScalar, Mapping: M> Occupied<D, Mapping> {
67    /// Wraps a tensor into an occupied slot.
68    pub(crate) fn new(tensor: Tensor<D, Mapping>) -> Self {
69        Self { data: tensor }
70    }
71}
72
73impl<D: VeScalar, Mapping: M> StashSlot<D> for Occupied<D, Mapping> {
74    fn clone_tensor_as<TargetMapping: M>(&self) -> Option<Tensor<D, TargetMapping>> {
75        // `transpose` already allocates a fresh tensor, so no separate clone is needed.
76        Some(self.data.transpose::<TargetMapping>(true))
77    }
78}
