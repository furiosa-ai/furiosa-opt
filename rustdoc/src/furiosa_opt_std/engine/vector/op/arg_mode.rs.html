1//! Argument mode types for VE operations.
2//!
3//! Defines how operands are mapped to operation arguments.
4
5use std::fmt::{self, Display, Formatter};
6
7use crate::engine::vector::scalar::VeScalar;
8use furiosa_opt_macro::primitive;
9
10// ============================================================================
11// ArgMode types
12// ============================================================================
13
14/// Arg mode: what operand to use as each argument of the operator.
15///
16/// Unary ops carry no mode — they always run as `op(mainstream)`. The previous "Mode1"
17/// (`op(operand0)`) shape is now expressed by adding a dedicated op variant rather than
18/// by overloading the unary path with an operand list.
19#[derive(Debug, Clone, Copy)]
20pub enum ArgMode {
21    /// Unary argument mode: `op(mainstream)`.
22    Unary,
23    /// Binary argument mode.
24    Binary(BinaryArgMode),
25    /// Ternary argument mode.
26    Ternary(TernaryArgMode),
27}
28
29/// Binary arg mode.
30/// ModeXY: op(argX, argY) where 0=mainstream, 1=operand0
31#[primitive(op::BinaryArgMode)]
32#[derive(Debug, Clone, Copy)]
33pub enum BinaryArgMode {
34    /// op(mainstream, mainstream).
35    Mode00,
36    /// op(mainstream, operand0).
37    Mode01,
38    /// op(operand0, mainstream).
39    Mode10,
40    /// op(operand0, operand0).
41    Mode11,
42}
43
44impl Display for BinaryArgMode {
45    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
46        match self {
47            Self::Mode00 => write!(f, "BinaryArgMode::Mode00"),
48            Self::Mode01 => write!(f, "BinaryArgMode::Mode01"),
49            Self::Mode10 => write!(f, "BinaryArgMode::Mode10"),
50            Self::Mode11 => write!(f, "BinaryArgMode::Mode11"),
51        }
52    }
53}
54
55impl BinaryArgMode {
56    /// Applies arg mode (operand selection) to a bare binary operation. `Uninit` propagation is the
57    /// caller's `zip_with`'s responsibility, this only picks which operands feed `op`.
58    pub(crate) fn apply<D: VeScalar>(self, op: impl Fn(D, D) -> D) -> impl Fn(D, D) -> D {
59        move |a, b| match self {
60            Self::Mode00 => op(a, a),
61            Self::Mode01 => op(a, b),
62            Self::Mode10 => op(b, a),
63            Self::Mode11 => op(b, b),
64        }
65    }
66}

68/// Ternary arg mode.
69/// ModeXYZ: op(argX, argY, argZ) where 0=mainstream, 1=operand0, 2=operand1
70#[derive(Debug, Clone, Copy)]
71pub enum TernaryArgMode {
72    /// op(mainstream, operand0, operand1).
73    Mode012,
74    /// op(mainstream, mainstream, operand1).
75    Mode002,
76    /// op(operand0, mainstream, operand1).
77    Mode102,
78    /// op(operand0, operand0, operand1).
79    Mode112,
80    /// op(mainstream, operand1, mainstream).
81    Mode020,
82    /// op(mainstream, operand1, operand0).
83    Mode021,
84    /// op(operand0, operand1, mainstream).
85    Mode120,
86}
87
88impl Display for TernaryArgMode {
89    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
90        match self {
91            Self::Mode012 => write!(f, "TernaryArgMode::Mode012"),
92            Self::Mode002 => write!(f, "TernaryArgMode::Mode002"),
93            Self::Mode102 => write!(f, "TernaryArgMode::Mode102"),
94            Self::Mode112 => write!(f, "TernaryArgMode::Mode112"),
95            Self::Mode020 => write!(f, "TernaryArgMode::Mode020"),
96            Self::Mode021 => write!(f, "TernaryArgMode::Mode021"),
97            Self::Mode120 => write!(f, "TernaryArgMode::Mode120"),
98        }
99    }
100}
101
102impl TernaryArgMode {
103    /// Applies arg mode (operand selection) to a bare ternary operation. `Uninit` propagation is the
104    /// caller's responsibility, this only picks which of `(m, o0, o1)` feed `op`.
105    pub(crate) fn apply<D: VeScalar>(self, op: impl Fn(D, D, D) -> D) -> impl Fn(D, D, D) -> D {
106        move |m, o0, o1| match self {
107            Self::Mode012 => op(m, o0, o1),
108            Self::Mode002 => op(m, m, o1),
109            Self::Mode102 => op(o0, m, o1),
110            Self::Mode112 => op(o0, o0, o1),
111            Self::Mode020 => op(m, o1, m),
112            Self::Mode021 => op(m, o1, o0),
113            Self::Mode120 => op(o0, o1, m),
114        }
115    }
116}
