1//! VeScalar trait for Vector Engine scalar types.
2
3use std::any::Any;
4
5use crate::scalar::MaterializableScalar;
6
7/// The kind of scalar type supported by Vector Engine.
8#[derive(Debug, PartialEq, Eq)]
9pub enum VeScalarKind {
10    /// 32-bit floating point.
11    F32,
12    /// 32-bit integer.
13    I32,
14}
15
16/// Marker trait for scalar types supported by Vector Engine.
17/// Only i32 and f32 are supported.
18///
19/// This trait is sealed and cannot be implemented outside this module. `MaterializableScalar` is a
20/// supertrait, not just incidentally true of the two real impls: the VE paths already exclude i5/i9
21/// separately (per `MaterializableScalar`'s own doc), so declaring it here lets every VE-generic
22/// whole-tensor op (`Tensor::map`/`zip_with`/`transmute`, all `MaterializableScalar`-gated) stay
23/// callable under a bare `D: VeScalar` bound, with no separate restatement at each call site.
24pub trait VeScalar: MaterializableScalar + Any {
25    /// The kind of this scalar type.
26    const KIND: VeScalarKind;
27
28    /// Reinterprets `self` as `D2` when the two concrete scalar types match (equivalently
29    /// `Self::KIND == D2::KIND`, since only i32/f32 exist), else `None`. Safe: a checked [`Any`]
30    /// downcast, no `transmute`. Used by the cross-type stash read to relabel `Tensor<StashD>` as
31    /// the pipeline's `Tensor<D>` once the KIND precondition holds.
32    fn reinterpret<D2: VeScalar>(self) -> Option<D2> {
33        (&self as &dyn Any).downcast_ref::<D2>().copied()
34    }
35}
36
37impl VeScalar for i32 {
38    const KIND: VeScalarKind = VeScalarKind::I32;
39}
40
41impl VeScalar for f32 {
42    const KIND: VeScalarKind = VeScalarKind::F32;
43}
