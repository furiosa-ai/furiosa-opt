15/// VE state that tracks stash and ALU usage.
16///
17/// The type parameter `D` ties the stash's scalar type to the pipeline's current scalar type,
18/// ensuring at compile time that stash reads match the pipeline's `D`. The `Stash` parameter is one
19/// of the stash-slot states ([`Fresh`] / [`Occupied`] / [`Spent`]); see the stash-slot
20/// [module docs] for the state machine.
21///
22/// [module docs]: crate::engine::vector::stash_slot
23#[derive(Debug)]
24pub struct VeState<D: VeScalar, Stash: StashSlot<D>> {
25    /// Stash-slot state: [`Fresh`], [`Occupied`] (holding the tensor), or [`Spent`].
26    pub(crate) stash: Stash,
27    /// Set of ALUs that have been used.
28    pub(crate) used_alus: HashSet<RngdAlu>,
29    /// Marker for the scalar type `D`.
30    _marker: PhantomData<D>,
31}
32
33impl<D: VeScalar, Stash: StashSlot<D>> VeState<D, Stash> {
34    /// Checks if ALU is available and marks it as used.
35    /// Panics if ALU is already in use.
36    pub(crate) fn use_alu(&mut self, alu: RngdAlu) {
37        assert!(!self.used_alus.contains(&alu), "{alu:?} is already in use");
38        self.used_alus.insert(alu);
39    }
40
41    /// Clones the stash data, transposing to target mapping and relabeling its scalar to the
42    /// reading pipeline's `D2`. The stashed `D` and the requested `D2` must share a KIND (a
43    /// cross-KIND stash read is rejected upstream by the typestate); both are i32 or both f32, so
44    /// the relabel is a per-element identity via [`VeScalar::reinterpret`] (an exact-type `Any`
45    /// downcast).
46    ///
47    /// The per-element downcast is `unwrap_unchecked` for perf (no panic landing-pad on the hot
48    /// map); its soundness rests on the two-`VeScalar` invariant — see the SAFETY note below and the
49    /// `# Panics` assert.
50    ///
51    /// TODO: fold the KIND equality into the typestate so this stays statically total; that also
52    /// retires the `unsafe`.
53    ///
54    /// # Panics
55    /// Panics if the requested `D2` KIND does not match the stashed `D` KIND.
56    pub(crate) fn force_clone_stash_as<D2: VeScalar, TargetMapping: M>(&self) -> Option<Tensor<D2, TargetMapping>> {
57        assert!(
58            D::KIND == D2::KIND,
59            "stash type mismatch: stashed as {:?}, requested as {:?}",
60            D::KIND,
61            D2::KIND
62        );
63
64        self.stash
65            .clone_tensor_as::<TargetMapping>()
66            // SAFETY: the `assert!(D::KIND == D2::KIND)` above holds, and `VeScalar` is implemented only for
67            // `i32` and `f32`, whose KINDs are distinct — so equal KIND means the SAME concrete type, and
68            // `reinterpret::<D2>()` (an exact-type `Any` downcast) is always `Some` here. If a third
69            // `VeScalar` sharing a KIND is ever added, this precondition breaks (the assert would pass but the
70            // downcast return `None`); fold the KIND equality into the typestate (see the TODO) before then.
71            .map(|t| t.map(|x| unsafe { x.reinterpret::<D2>().unwrap_unchecked() }))
72    }
73}

75impl<D: VeScalar, StashMapping: M> VeState<D, Occupied<D, StashMapping>> {
76    /// Empties the stash, moving the state to [`Spent`] and preserving ALU usage. `D` here is the
77    /// stashed scalar type, which may differ from the reading pipeline's scalar (cross-type stash).
78    /// The read lands in `Spent` rather than `Fresh` so the write cannot be re-armed; see the
79    /// stash-slot [module docs] for the state machine. The stashed tensor's data was already read by
80    /// the op that calls this.
81    ///
82    /// [module docs]: crate::engine::vector::stash_slot
83    pub(crate) fn consume_stash(self) -> VeState<D, Spent> {
84        VeState {
85            stash: Spent,
86            used_alus: self.used_alus,
87            _marker: PhantomData,
88        }
89    }
90}

92impl<D: VeScalar> VeState<D, Fresh> {
93    /// Creates a new empty VeState.
94    pub(crate) fn new() -> Self {
95        Self {
96            stash: Fresh,
97            used_alus: HashSet::new(),
98            _marker: PhantomData,
99        }
100    }
101
102    /// Stores the tensor data in the stash, returning a new VeState with the stash's mapping.
103    /// Consumes self to change the `StashSlot` parameter.
104    pub(crate) fn stash<NewMapping: M>(self, data: &Tensor<D, NewMapping>) -> VeState<D, Occupied<D, NewMapping>> {
105        VeState {
106            stash: Occupied::new(data.clone()),
107            used_alus: self.used_alus,
108            _marker: PhantomData,
109        }
110    }
111
112    /// Consumes another VeState and merges its ALU usage into this one.
113    /// Used when combining two groups in binary operations.
114    /// Since both groups share the same ALU state (global resource),
115    /// this simply performs a union of used ALUs.
116    pub(crate) fn merge(&mut self, other: VeState<D, Fresh>) {
117        self.used_alus.extend(other.used_alus);
118    }
119
120    /// Converts VeState to different scalar type parameters, preserving ALU tracking.
121    pub(crate) fn retype<NewD: VeScalar>(self) -> VeState<NewD, Fresh> {
122        VeState {
123            stash: Fresh,
124            used_alus: self.used_alus,
125            _marker: PhantomData,
126        }
127    }
