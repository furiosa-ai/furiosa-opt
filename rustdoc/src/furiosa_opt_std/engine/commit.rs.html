1//! Commit Engine: Tensor Unit stream to DM.
2//!
3//! Drains a post-Commit-Adapter `TuTensor` (from `commit_trim` onwards) into a
4//! [`DmTensor`] (or an existing mutable view). `verify_commit` synthesizes the
5//! write sequencer with `SMapping::sequence` (Write), the dual of the fetch read.
6
7use furiosa_mapping::*;
8use furiosa_opt_macro::primitive;
9
10use furiosa_opt_lower::config_commit;
11
12use crate::backend::Backend;
13use crate::context::*;
14use crate::engine::CanApplyCommit;
15use crate::scalar::*;
16use crate::tensor::memory::{Address, DmTensor, DmTensorViewMut};
17use crate::tensor::tu::TuTensor;

19// ANCHOR: commit_impl
20impl<'l, const T: Tu, P: CanApplyCommit, D: Scalar, Chip: M, Cluster: M, Slice: M, Time: M, Packet: M, B: Backend>
21    TuTensor<'l, T, P, D, Chip, Cluster, Slice, Time, Packet, B>
22{
23    /// Commits to data memory.
24    #[primitive(TuTensor::commit)]
25    pub fn commit<Element: M>(self) -> DmTensor<D, Chip, Cluster, Slice, Element, B> {
26        verify_commit::<D, Time, Packet, Element>();
27        DmTensor::new(self.inner.transpose(false), None)
28    }
29
30    /// Commits to data memory at `address`.
31    #[primitive(TuTensor::commit_at)]
32    pub fn commit_at<Element: M>(self, address: Address) -> DmTensor<D, Chip, Cluster, Slice, Element, B> {
33        verify_commit::<D, Time, Packet, Element>();
34        DmTensor::new(self.inner.transpose(false), Some(address))
35    }
36
37    /// Commits to a mutable tensor view in data memory.
38    #[primitive(TuTensor::commit_view)]
39    pub fn commit_view<Element: M>(self, mut dst: DmTensorViewMut<'l, D, Chip, Cluster, Slice, Element, B>) {
40        verify_commit::<D, Time, Packet, Element>();
41        dst.inner.transpose(self.inner.view(), false);
42    }
43}
44// ANCHOR_END: commit_impl

46/// Validates the Commit engine and synthesizes the write config via [`config_commit`], panicking
47/// with the commit error on failure. The resolved descriptors are not consumed yet, so they're
48/// explicitly discarded. The dual of `verify_fetch`: it writes the time-ordered `(InTime, InPacket)`
49/// stream into the DM `Element` layout.
50pub(crate) fn verify_commit<D: Scalar, InTime: M, InPacket: M, Element: M>() {
51    let _ = config_commit(
52        &InTime::to_value(),
53        &InPacket::to_value(),
54        &Element::to_value(),
55        D::BITS,
56    )
57    .unwrap_or_else(|e| panic!("{e}"));
58}

64    /// Each accepts a commit across an orthogonal phenomenon; the matcher's write synthesis is
65    /// exhaustively tested in `furiosa-mapping-impl`.
66    mod commit_valid {
67        use super::*;
68
69        axes![N = 8, A = 4, B = 3, C = 4];
70
71        /// Full trim, then a clean multi-axis commit.
72        #[test]
73        fn full_trim_then_commit() {
74            verify_commit::<i8, m![A, B, C], m![N], m![A, B, C, N]>();
75        }
76
77        /// Packet keeps trailing flit padding (partial trim).
78        #[test]
79        fn partial_trim_then_commit() {
80            verify_commit::<i8, m![A], m![N # 16], m![A, N # 16]>();
81        }
82
83        /// Time axes are transposed across the commit.
84        #[test]
85        fn time_transpose() {
86            verify_commit::<i8, m![A # 32, B], m![N], m![B, A # 32, N]>();
87        }
88
89        /// The stream emits the live block four times (`1#2 ⋈ 1#2`); overlapping
90        /// dummy writes collapse onto the DM's smaller dummy region — realizable
91        /// even though an injective placement could not fit.
92        #[test]
93        fn interleaved_time_padding_overlaps_into_dm_padding() {
94            verify_commit::<i8, m![1 # 2, A, 1 # 2], m![N], m![A, 1 # 3, N]>();
95        }
96    }
