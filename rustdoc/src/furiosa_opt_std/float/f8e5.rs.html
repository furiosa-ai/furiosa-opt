1//! E5M2 (1 sign, 5-bit exponent, 2-bit mantissa) 8-bit float conversions.
2//!
3//! E5M2 is IEEE-754-like, so conversion to/from f32 is a direct bit reshuffle
4//! (rebias the exponent and align the mantissa) rather than a lookup table.
5
6pub(crate) const F8E5_ZERO: u8 = 0x00;
7pub(crate) const F8E5_NEG_ZERO: u8 = 0x80;
8pub(crate) const F8E5_ONE: u8 = 0x3c; // sign 0, exp 15, mantissa 0
9
10pub(crate) fn f8_e5_to_f32(bits: u8) -> f32 {
11    let sign = ((bits >> 7) & 1) as u32;
12    let exp = ((bits >> 2) & 0x1f) as u32;
13    let mant = (bits & 0x3) as u32;
14    match exp {
15        // Zero or subnormal: value = (-1)^s * mant * 2^-16.
16        0 => {
17            let m = mant as f32 / 65536.0;
18            if sign == 1 { -m } else { m }
19        }
20        // Inf (mantissa 0) or NaN.
21        0x1f => {
22            let f = if mant == 0 { f32::INFINITY } else { f32::NAN };
23            if sign == 1 { -f } else { f }
24        }
25        // Normal: rebias exponent (15 -> 127) and left-align the mantissa.
26        // `exp + 127 - 15` (not `exp - 15 + 127`) so the u32 never underflows.
27        _ => f32::from_bits((sign << 31) | ((exp + 127 - 15) << 23) | (mant << 21)),
28    }
29}
30
31pub(crate) fn f8_e5_from_f32(v: f32) -> u8 {
32    if v.is_nan() {
33        return 0x7e; // qNaN: exp 0x1f, mantissa nonzero
34    }
35    let sign = if v.is_sign_negative() { 0x80u8 } else { 0 };
36    let a = v.abs();
37    if a.is_infinite() {
38        return sign | 0x7c; // inf: exp 0x1f, mantissa 0
39    }
40    if a == 0.0 {
41        return sign;
42    }
43    let n = a.to_bits();
44    let unbiased_exp = ((n >> 23) & 0xff) as i32 - 127;
45    let man = n & 0x7fffff;
46    let biased = unbiased_exp + 15;
47    if biased >= 0x1f {
48        return sign | 0x7c; // overflow saturates to inf
49    }
50    if biased <= 0 {
51        // Subnormal: round a / 2^-16 to the nearest representable mantissa.
52        let sub = (a * 65536.0).round() as u32;
53        return sign | (sub.min(3) as u8);
54    }
55    // Normal: keep the top 2 mantissa bits, round half up (carry into exponent).
56    let mut mant2 = man >> 21;
57    let mut exp = biased as u32;
58    if (man >> 20) & 1 == 1 {
59        mant2 += 1;
60        if mant2 == 4 {
61            mant2 = 0;
62            exp += 1;
63            if exp >= 0x1f {
64                return sign | 0x7c;
65            }
66        }
67    }
68    sign | ((exp as u8) << 2) | (mant2 as u8)
69}
70
71pub(crate) fn f8_e5_is_zero(bits: u8) -> bool {
72    bits == F8E5_ZERO || bits == F8E5_NEG_ZERO
73}
