1//! E4M3 (1 sign, 4-bit exponent, 3-bit mantissa) 8-bit float conversions.
2//!
3//! E4M3 is not a plain bit reshuffle of f32 (its bias and saturation differ), so
4//! `to_f32` uses a 256-entry lookup table and `from_f32` a hand-tuned rounding.
5
6pub(crate) const F8E4_ZERO: u8 = 0x00;
7pub(crate) const F8E4_NEG_ZERO: u8 = 0x80;
8pub(crate) const F8E4_ONE: u8 = 0x38;
9
10pub(crate) fn f8_e4_to_f32(bits: u8) -> f32 {
11    f32::from_bits(F8E4_TO_F32[bits as usize])
12}
13
14pub(crate) fn f8_e4_from_f32(v: f32) -> u8 {
15    if v.is_nan() {
16        return 0x7f; // qNaN
17    }
18    let n = v.to_bits();
19    let exp = (n >> 23) & 0xff;
20    let man = n & 0x7fffff;
21    let abs_value = if exp > 127 - 7 {
22        let valid = (n >> 20) & 0x7ff;
23        let val = if (n & 0xfffff) + (valid & 1) > 0x80000 {
24            valid + 1
25        } else {
26            valid
27        };
28        if val >= (((127 + 8) << 3) | 0x6) {
29            0x7e_u32 // MAX
30        } else {
31            val - ((127 - 7) << 3)
32        }
33    } else if exp == 127 - 7 {
34        if man >= 0x700000 {
35            8
36        } else if man > 0x500000 {
37            7
38        } else if man >= 0x300000 {
39            6
40        } else if man > 0x100000 {
41            5
42        } else {
43            4
44        }
45    } else if exp == 127 - 8 {
46        if man >= 0x600000 {
47            4
48        } else if man > 0x200000 {
49            3
50        } else {
51            2
52        }
53    } else if exp == 127 - 9 {
54        if man >= 0x400000 { 2 } else { 1 }
55    } else if exp == 127 - 10 && man != 0 {
56        1
57    } else {
58        0
59    };
60    (((n >> 31) << 7) | abs_value) as u8
61}
62
63pub(crate) fn f8_e4_is_zero(bits: u8) -> bool {
64    bits == F8E4_ZERO || bits == F8E4_NEG_ZERO
65}
