14/// `tracing` target and category for on-device profile spans, mirroring
15/// npu-executor's `TRACING_TARGET_NPU`/`TRACING_CATEGORY_NPU` so the shared
16/// furiosa-telemetry chrome exporter consumes VISA kernel spans identically.
17const TRACING_TARGET_NPU: &str = "span::npu";
18const TRACING_CATEGORY_NPU: &str = "NPU";
19
20/// Run a loaded kernel over its I/O buffers, arming on-device profiling only when
21/// a subscriber listens on the `span::npu` target. Returns the runtime status
22/// (0 on success). The enable check is the sole hot-path cost when profiling is
23/// off; read-back and decode are deferred by the runtime, so the launch itself
24/// never blocks.
25pub(super) fn run(kernel: *mut Kernel, inputs: &[*const NpuBuffer], outputs: &[*const NpuBuffer]) -> i32 {
26    if !tracing::enabled!(target: TRACING_TARGET_NPU, tracing::Level::INFO) {
27        // SAFETY: the kernel handle and pointer arrays stay live across this synchronous call.
28        return unsafe {
29            furiosa_kernel_run(
30                kernel,
31                rt(),
32                inputs.as_ptr(),
33                inputs.len(),
34                outputs.as_ptr(),
35                outputs.len(),
36            )
37        };
38    }
39
40    // `sink` emits each decoded span as an `info_span!` on `span::npu` (name +
41    // cycle window), matching npu-executor's schema; `done` is a noop since the
42    // ctx is `null`. The runtime calls them off the launch hot path during
43    // deferred read-back. They are `extern "C" fn` (not closures) because the C
44    // ABI callback slots require the C ABI, which Rust closures cannot carry.
45    unsafe extern "C" fn sink(_ctx: *mut c_void, name: Str, begin: u64, end: u64) {
46        // SAFETY: the runtime passes a valid name pointer and length for this call.
47        let bytes = unsafe { std::slice::from_raw_parts(name.ptr.cast::<u8>(), name.len) };
48        tracing::info_span!(
49            target: TRACING_TARGET_NPU,
50            "NPU",
51            cat = TRACING_CATEGORY_NPU,
52            name = String::from_utf8_lossy(bytes).as_ref(),
53            begin_cycle = begin,
54            end_cycle = end,
55        );
56    }
57    unsafe extern "C" fn done(_ctx: *mut c_void) {}
58
59    // SAFETY: the handle and pointer arrays stay live across the launch; the sink/done callbacks
60    // are `'static` and the ctx is `null`, so nothing must outlive this call.
61    unsafe {
62        furiosa_profiled_run(
63            rt(),
64            kernel,
65            inputs.as_ptr(),
66            inputs.len(),
67            outputs.as_ptr(),
68            outputs.len(),
69            Some(sink),
70            Some(done),
71            std::ptr::null_mut(),
72        )
73    }
74}

76pub(crate) fn rt() -> *mut Runtime {
77    struct Handle(*mut Runtime);
78    unsafe impl Send for Handle {}
79    unsafe impl Sync for Handle {}
80
81    static RT: OnceLock<Handle> = OnceLock::new();
82    RT.get_or_init(|| {
83        let ver = unsafe { std::ffi::CStr::from_ptr(furiosa_version()) };
84        log::info!("device-runtime: {}", ver.to_string_lossy());
85        let Device { chip: chips, pe: pes } = *DEVICE.get_or_init(|| Device { chip: 1, pe: 8 });
86
87        // Optimistic acquisition: `furiosa_runtime_init`'s exclusive open (null when a
88        // chip is busy) is the atomic arbiter, so try each candidate set of `chips`
89        // and take the first that acquires. No check-then-use, hence no TOCTOU race;
90        // the chips are whatever the host exposes and need not be contiguous.
91        let available = available_chips();
92        for group in available.windows(chips as usize) {
93            let npus: Vec<NpuDesc> = group
94                .iter()
95                .flat_map(|&chip| match pes {
96                    8 => vec![
97                        NpuDesc {
98                            chip,
99                            pe_start: 0,
100                            pe_end: 3,
101                        },
102                        NpuDesc {
103                            chip,
104                            pe_start: 4,
105                            pe_end: 7,
106                        },
107                    ],
108                    n => vec![NpuDesc {
109                        chip,
110                        pe_start: 0,
111                        pe_end: n - 1,
112                    }],
113                })
114                .collect();
115            let ptr = unsafe { furiosa_runtime_init(npus.as_ptr(), npus.len()) };
116            if !ptr.is_null() {
117                return Handle(ptr);
118            }
119        }
120        panic!("no NPU available: {} chip(s) present, all in use", available.len());
121    })
122    .0
123}

125/// Chip ids the host exposes, read from the `/dev/rngd/npu<N>mgmt` device nodes, narrowed to
126/// [`visible_chips`] when it is set.
127fn available_chips() -> Vec<u8> {
128    let mut chips: Vec<u8> = std::fs::read_dir("/dev/rngd")
129        .into_iter()
130        .flatten()
131        .flatten()
132        .filter_map(|entry| {
133            let name = entry.file_name();
134            name.to_str()?.strip_prefix("npu")?.strip_suffix("mgmt")?.parse().ok()
135        })
136        .collect();
137    chips.sort_unstable();
138    if let Some(visible) = visible_chips() {
139        chips.retain(|chip| visible.contains(chip));
140    }
141    chips
142}

144/// Chip ids `FURIOSA_VISIBLE_DEVICES` (comma-separated) restricts acquisition to; `None` when unset.
145///
146/// Acquisition skips busy chips but cannot reserve one, so processes starting together race for the
147/// same free chip. Listing disjoint chips pins them apart.
148fn visible_chips() -> Option<Vec<u8>> {
149    let spec = std::env::var("FURIOSA_VISIBLE_DEVICES").ok()?;
150    let spec = spec.trim();
151    if spec.is_empty() {
152        return None;
153    }
154    Some(
155        spec.split(',')
156            .map(|id| {
157                id.trim()
158                    .parse()
159                    .unwrap_or_else(|_| panic!("FURIOSA_VISIBLE_DEVICES: `{id}` is not a chip index (0-255)"))
160            })
161            .collect(),
162    )
163}

167/// Binds the `#[device(chip, pe)]` the process runs on, via `Context::acquire().bind(..)`.
168/// First bind wins; a conflicting one panics.
169pub(crate) fn bind_device(device: Device) {
170    let cur = *DEVICE.get_or_init(|| device);
171    assert_eq!(
172        cur, device,
173        "conflicting NPU device in one process: {cur:?} vs {device:?}"
174    );
175}
