1//! Backend abstraction: the storage-op seam for tensor operations.
2//!
3//! A [`Backend`] picks the concrete tensor storage type and implements every tensor operation
4//! against it. The concrete impls are [`Emulation`] (host-side buffer emulator), [`Npu`] (real
5//! device), and [`Typecheck`] (shape/mapping validation only). The cfg-selected
6//! [`crate::runtime::CurrentBackend`] alias picks which one the crate compiles against.
7//! Device-function execution (`launch`, `DeviceFn`, `DeviceSend`) lives in the sibling
8//! [`crate::runtime`] module.

30/// Backend for tensor operations.
31///
32/// A backend picks the concrete tensor storage type (`Self::Storage<D>`) and the DMA
33/// protocol. The storage type is opaque (`Clone + Debug` only); every operation on it, both the
34/// lifecycle constructors/serializers and the value-ops (map, reduce, zip_with, scatter, ...),
35/// is a method on this trait, delegating to the concrete storage's inherent methods.
36///
37/// `Emulation`: interprets ops on `BufStorage`'s physical staging buffer (default).
38/// `Npu`: owns native staging buffers and does real PCIe DMA; the host never interprets ops.
39/// `Typecheck`: shape/mapping validation only, no host values (`--cfg backend="typecheck"`).
40///
41/// The host backend (Emulation) implements the DMA as a plain host-side `transpose`
42/// passthrough; `Npu` provides real device transfers. DMA stays a required method (no default) so
43/// each backend states its choice explicitly.
44pub trait Backend: Sized + 'static {
45    /// Concrete per-backend tensor type (`BufStorage` / `PhantomStorage`). Opaque:
46    /// only `Clone + Debug + PartialEq` (so the memory-tier `#[derive(Debug)]` structs and the
47    /// `Tensor<D, Mapping, B>` wrapper's `Debug`/`PartialEq` resolve without a per-backend `where`
48    /// clause). `PartialEq` (not `Eq`) because `f32` tensors must compare. All operations on the
49    /// storage are methods on this trait.
50    type Storage<D: Scalar>: Clone + Debug + PartialEq;
51
52    /// Build a storage from a flat `D` buffer in `mapping`-order, delegating to the concrete
53    /// storage's inherent constructor. The runtime `mapping` carries the layout, construction is
54    /// the type→runtime boundary, so the wrapper converts its type param to a value here.
55    fn from_vec<D: Scalar>(mapping: &MappingValue, data: impl IntoIterator<Item = D>) -> Self::Storage<D>;
56
57    /// Build a storage from its dense physical device byte image (`mapping`-order, packed on `D::BITS`):
58    /// the inverse of [`Self::into_buf`]. The default decodes the packed bytes to a logical `Vec<D>` and
59    /// packs via [`Self::from_vec`]; a backend whose storage already IS the packed image (Emulation /
60    /// Npu's [`BufStorage`]) overrides this to store the bytes directly, no decode / re-pack. Pre-packed
61    /// data (fp4 / f4e2m1 weights) enters through here. Backs [`crate::tensor::Tensor::from_buf`].
62    fn from_buf<D: Scalar>(mapping: &MappingValue, buf: Vec<u8>) -> Self::Storage<D> {
63        Self::from_vec(mapping, (0..mapping.size()).map(|i| D::load(&buf, i)))
64    }
65
66    /// An uninitialized storage of this layout, delegating to the concrete storage's inherent
67    /// constructor.
68    fn uninit<D: Scalar>(mapping: &MappingValue) -> Self::Storage<D>;
69
70    /// Serialize the storage to a flat `D` buffer in `mapping`-order (the physical / wire layout),
71    /// consuming the storage, delegating to the concrete storage's inherent serializer.
72    fn into_vec<D: MaterializableScalar>(storage: Self::Storage<D>, mapping: &MappingValue) -> Vec<D>;
73
74    /// Serialize the storage to its dense physical device byte image (`mapping`-order, packed on
75    /// `D::BITS`): the exact bytes that sit in HBM / DM, half the logical length for a 4-bit `D`. The
76    /// default decodes to a logical `Vec<D>` and re-packs via [`Scalar::to_buf`]; a backend whose
77    /// storage already IS the packed image (Emulation / Npu's [`BufStorage`]) overrides this to move the
78    /// internal bytes out directly. Backs [`crate::tensor::memory::HbmTensor::to_buf`].
79    fn into_buf<D: MaterializableScalar>(storage: Self::Storage<D>, mapping: &MappingValue) -> Vec<u8> {
80        D::to_buf(&Self::into_vec(storage, mapping))
81    }
82
83    /// Element-wise map to a new scalar, preserving layout. The closure is bare-`D`; each backend
84    /// applies its own per-storage strategy, Emulation/Npu walk the `BufStorage` buffer over bare
85    /// `D`, Typecheck no-ops on the phantom. Mapping-free: elementwise ops preserve the source
86    /// layout, so no mapping type parameter is needed.
87    /// The closure must be `Fn + Sync` (not `FnMut`): an elementwise map has no cross-cell state, and
88    /// [`Emulation`] folds the cells across the rayon pool.
89    fn map<D: MaterializableScalar, D2: Scalar>(
90        src: &Self::Storage<D>,
91        f: impl Fn(D) -> D2 + Sync,
92    ) -> Self::Storage<D2>;
93
94    /// [`Self::map`], but bounded by a caller-supplied `len` instead of the storage's own length
95    /// recovery. Backs [`crate::tensor::Tensor::map_bounded`]; see its doc for why plain [`Self::map`]
96    /// is unsound for a non-`MaterializableScalar` staging type (`i5`/`i9`).
97    fn map_bounded<D: Scalar, D2: Scalar>(
98        src: &Self::Storage<D>,
99        len: usize,
100        f: impl Fn(D) -> D2 + Sync,
101    ) -> Self::Storage<D2>;
102
103    /// Element-wise zip of two same-layout tensors to a new scalar. See [`Self::map`]; the result is
104    /// defined only where both inputs are.
105    fn zip_with<D: MaterializableScalar, D2: MaterializableScalar, D3: Scalar>(
106        a: &Self::Storage<D>,
107        b: &Self::Storage<D2>,
108        f: impl Fn(D, D2) -> D3 + Sync,
109    ) -> Self::Storage<D3>;
110
111    /// Element-wise zip of three same-layout tensors to a new scalar, the ternary peer of
112    /// [`Self::map`] (1-ary) and [`Self::zip_with`] (2-ary). The result is defined only where all
113    /// three inputs are. The VE engine uses it for branch-conditional combine: it passes the
114    /// execution-id `tag` as the third operand and resolves `TagFilter`/select inside the closure,
115    /// so the backend stays free of VE concepts.
116    fn zip3_with<D: MaterializableScalar, D2: MaterializableScalar, D3: MaterializableScalar, D4: Scalar>(
117        a: &Self::Storage<D>,
118        b: &Self::Storage<D2>,
119        c: &Self::Storage<D3>,
120        f: impl Fn(D, D2, D3) -> D4 + Sync,
121    ) -> Self::Storage<D4>;
122
123    /// Writes a transposed/broadcast view of `src` into `dst`. `src_offset` / `dst_offset` allow
124    /// operating on partial views; `src_map` / `dst_map` are each storage's base (live-axis) mapping,
125    /// where a tiled axis the `Src` / `Dst` *type* carries as padding stays a live `Symbol`, needed
126    /// to resolve a partial-view offset's physical wire base. Each backend applies its own per-storage
127    /// strategy: Emulation/Npu resolve the offset bases against the maps and drive a
128    /// sequencer over the `BufStorage` buffer, Typecheck runs only the structural check.
129    #[allow(clippy::too_many_arguments)]
130    fn transpose<D: Scalar, Src: M, Dst: M>(
131        dst: &mut Self::Storage<D>,
132        src: &Self::Storage<D>,
133        src_offset: &Index,
134        dst_offset: &Index,
135        src_map: &MappingValue,
136        dst_map: &MappingValue,
137        allow_broadcast: bool,
138    );
139
140    /// Reduces the factors of `src`'s mapping absent in `Dst`, returning the result. Axes that `Dst`
141    /// adds beyond the reduced source are broadcast (the reduced value repeats across them).
142    /// `allow_broadcast` gates that, exactly like [`Self::transpose`]'s flag: `false` is a
143    /// plain reduce that asserts `Dst` is a pure factor and *rejects* an unintended extra axis;
144    /// `true` permits the broadcast axes. The reduce fn works over bare scalars; the backend lifts
145    /// it to its cell representation.
146    ///
147    /// `reduce_fn` should be associative: the parallel backends may regroup the fold across threads. An
148    /// associative fn gives a deterministic, serial-identical result; a non-associative one (saturating
149    /// integer add, float add/mul) can differ from a serial left-fold and vary run-to-run on
150    /// [`Emulation`], which regroups the fold across the rayon pool.
151    fn reduce<D: MaterializableScalar, Src: M, Dst: M>(
152        src: &Self::Storage<D>,
153        reduce_fn: impl Fn(D, D) -> D + Sync,
154        identity: D,
155        allow_broadcast: bool,
156    ) -> Self::Storage<D>;
157
158    /// Contracts `lhs` and `rhs` into an `out`-shaped result: a generalized matmul
159    /// `out[..] = sum over the axes of pre_reduce absent from out of (lhs * rhs)`. `pre_reduce` is the
160    /// full (un-reduced) index space; `lhs_map` / `rhs_map` / `out` are its sub-mappings (the contracted
161    /// axes are those in `pre_reduce` but not `out`).
162    ///
163    /// A required primitive, not a default: every backend fuses this into a single accumulating pass
164    /// over `out`-many cells (`O(out)` memory), never materializing the `pre_reduce`-shaped outer
165    /// product (`O(pre_reduce)`, which for a wide-output GEMM is the hundreds-of-MB dense buffer this
166    /// primitive exists to avoid). The mappings arrive as runtime values because the engine-level
167    /// contraction pipeline ([`crate::engine::contraction`]) only has them as values at the point it
168    /// defers the fold; `lhs`/`rhs` keep their own compact (un-broadcast) layout, read through
169    /// `lhs_map`/`rhs_map`.
170    ///
171    /// `pre_reduce` is not simply "the union of `lhs_map`'s and `rhs_map`'s axes": the engine-level
172    /// caller can reconcile the two operands' contracted portions through DIFFERENT symbols (e.g. one
173    /// operand's `Time`/`Packet` against the other's `TrfElement`, per the hardware layout-expansion
174    /// contract `contract_outer` validates), which a naive per-symbol union of the two maps cannot
175    /// recover. So it is a required, independently-computed argument, not derived here.
176    ///
177    /// Like [`Self::reduce`], [`Emulation`]'s per-output-cell sum regroups across threads, so its
178    /// `f32` result can differ from serial and vary run-to-run.
179    fn contraction<D: ContractionCast + MaterializableScalar>(
180        lhs: &Self::Storage<D>,
181        rhs: &Self::Storage<D>,
182        lhs_map: &MappingValue,
183        rhs_map: &MappingValue,
184        pre_reduce: &MappingValue,
185        out: &MappingValue,
186    ) -> Self::Storage<D>;
187
188    /// Scatters `src` into `dst` at positions read from the same-backend `i32` index tensor.
189    fn scatter<D: Scalar, Src: M, Key: M, Dst: M, Idx: M>(
190        src: &Self::Storage<D>,
191        dst: &mut Self::Storage<D>,
192        index: &Self::Storage<i32>,
193        scaled: bool,
194    );
195
196    /// Gathers from `src` (table) into `dst` at positions read from the index tensor.
197    fn gather<D: MaterializableScalar, Src: M, Dst: M, Idx: M>(
198        src: &Self::Storage<D>,
199        dst: &mut Self::Storage<D>,
200        index: &Self::Storage<i32>,
201        scaled: bool,
202    );
203
204    /// Reinterprets `src`'s buffer under a new mapping `Dst`, returning the result. A relabeling, not
205    /// a data move: `BufStorage` keeps the same physical buffer, pairing wire position `i` under
206    /// `Src` with wire position `i` under `Dst`.
207    ///
208    /// Memory-safe and length-checked (asserts `Src::SIZE == Dst::SIZE`). Whether the relabel is
209    /// *meaningful* is the caller's logical precondition, discharged by the `unsafe` public `reshape`
210    /// wrappers: `Src` and `Dst` must enumerate the elements in the SAME physical (wire) order, i.e.
211    /// `Src::indexes()` and `Dst::indexes()` line the live elements up identically. Regrouping axes
212    /// (merge/split) preserves wire order and is valid; a permutation (e.g. `m![A, B]` → `m![B, A]`)
213    /// does NOT and must use `transpose`. This method itself only guarantees memory safety.
214    fn reshape<D: Scalar, Src: M, Dst: M>(src: &Self::Storage<D>) -> Self::Storage<D>;
215
216    /// Relabels `storage` from `src_map` to `dst_map` (`Tensor::transmute`). Unlike `reshape`, the two
217    /// maps may differ in size: the vector engine relabels a padded read down to its compact slot
218    /// (`…tile(k).read().transmute()`), so `dst_map.size() <= src_map.size()` is intended.
219    ///
220    /// `Typecheck` returns `storage` unchanged, its canonical storage is already
221    /// padding-free, so a smaller `dst_map` needs no data move. `Emulation` / `Npu` COMPACT the
222    /// physical buffer: they keep only the live (non-padding) cells of the `src_map` layout, in
223    /// `dst_map` order, producing a `dst_map.size()`-length buffer (a no-op when no padding is dropped).
224    fn transmute<D: MaterializableScalar, Src: M, Dst: M>(
225        storage: Self::Storage<D>,
226        src_map: &MappingValue,
227        dst_map: &MappingValue,
228    ) -> Self::Storage<D>;
229
230    /// Transfer host tensor to HBM. Host-only: invoked from test/setup code over PCIe DMA, never
231    /// from device kernel MIR, so no `#[primitive(...)]` annotation is needed.
232    fn to_hbm<D: MaterializableScalar, Element: M, Chip: M, Element2: M>(
233        host: &HostTensor<D, Element, Self>,
234    ) -> impl std::future::Future<Output = HbmTensor<D, Chip, Element2, Self>>;
235
236    /// Transfer HBM tensor to host. Host-only; see the `to_hbm` note above.
237    fn from_hbm<D: MaterializableScalar, Chip: M, Element: M, Element2: M>(
238        hbm: &HbmTensor<D, Chip, Element, Self>,
239    ) -> impl std::future::Future<Output = HostTensor<D, Element2, Self>>;
240
241    /// Bind the process to a logical [`Device`](crate::context::Device) before any host I/O. Only the
242    /// NPU backend acts on this; backends without a physical device default to a no-op.
243    fn bind_device(_device: crate::context::Device) {}
