12/// Typecheck backend: shape/mapping-only, no host-side computation.
13///
14/// All per-operation overrides live on [`PhantomStorage`]'s inherent methods. The Backend
15/// supplies only the host-side DMA stubs here.
16#[derive(Debug, Clone, Copy)]
17pub struct Typecheck;
18
19impl<D: Scalar, Mapping: M> Tensor<D, Mapping, Typecheck> {
20    /// Phantom tensor for Typecheck: metadata only, no values.
21    ///
22    /// `Tensor::zeroed()` is the generic constructor but reads as element-level
23    /// "zero-filled"; for Typecheck the whole tensor *is* the empty value, so this name
24    /// matches the actual semantics.
25    pub fn empty() -> Self {
26        Self::zeroed()
27    }
28}
29
30impl Backend for Typecheck {
31    type Storage<D: Scalar> = PhantomStorage<D>;
32
33    fn from_vec<D: Scalar>(mapping: &MappingValue, data: impl IntoIterator<Item = D>) -> Self::Storage<D> {
34        PhantomStorage::from_vec(mapping, data)
35    }
36
37    fn zeroed<D: Scalar>(mapping: &MappingValue) -> Self::Storage<D> {
38        PhantomStorage::zeroed(mapping)
39    }
40
41    fn into_vec<D: MaterializableScalar>(storage: Self::Storage<D>, mapping: &MappingValue) -> Vec<D> {
42        storage.into_vec(mapping)
43    }
44
45    fn map<D: MaterializableScalar, D2: Scalar>(
46        src: &Self::Storage<D>,
47        f: impl Fn(D) -> D2 + Sync,
48    ) -> Self::Storage<D2> {
49        src.map(f)
50    }
51
52    fn map_bounded<D: Scalar, D2: Scalar>(
53        src: &Self::Storage<D>,
54        _len: usize,
55        f: impl Fn(D) -> D2 + Sync,
56    ) -> Self::Storage<D2> {
57        // `PhantomStorage` carries no values (see `PhantomStorage::map`), so the explicit `len` this
58        // exists to honor for a real buffer is moot here; delegate straight to the ordinary phantom map.
59        src.map(f)
60    }
61
62    fn zip_with<D: MaterializableScalar, D2: MaterializableScalar, D3: Scalar>(
63        a: &Self::Storage<D>,
64        b: &Self::Storage<D2>,
65        f: impl Fn(D, D2) -> D3 + Sync,
66    ) -> Self::Storage<D3> {
67        a.zip_with(b, f)
68    }
69
70    fn zip3_with<D: MaterializableScalar, D2: MaterializableScalar, D3: MaterializableScalar, D4: Scalar>(
71        a: &Self::Storage<D>,
72        b: &Self::Storage<D2>,
73        c: &Self::Storage<D3>,
74        f: impl Fn(D, D2, D3) -> D4 + Sync,
75    ) -> Self::Storage<D4> {
76        a.zip3_with(b, c, f)
77    }
78
79    fn transpose<D: Scalar, Src: M, Dst: M>(
80        dst: &mut Self::Storage<D>,
81        src: &Self::Storage<D>,
82        src_offset: &Index,
83        dst_offset: &Index,
84        _src_map: &MappingValue,
85        _dst_map: &MappingValue,
86        allow_broadcast: bool,
87    ) {
88        // `PhantomStorage` holds no buffer, so it ignores `*_map` and runs only the structural check.
89        dst.transpose::<Src, Dst>(src, src_offset, dst_offset, allow_broadcast);
90    }
91
92    fn reduce<D: MaterializableScalar, Src: M, Dst: M>(
93        src: &Self::Storage<D>,
94        reduce_fn: impl Fn(D, D) -> D + Sync,
95        identity: D,
96        allow_broadcast: bool,
97    ) -> Self::Storage<D> {
98        src.reduce::<Src, Dst, _>(reduce_fn, identity, allow_broadcast)
99    }
100
101    fn contraction<D: ContractionCast + MaterializableScalar>(
102        lhs: &Self::Storage<D>,
103        rhs: &Self::Storage<D>,
104        _lhs_map: &MappingValue,
105        _rhs_map: &MappingValue,
106        _pre_reduce: &MappingValue,
107        out: &MappingValue,
108    ) -> Self::Storage<D> {
109        // `PhantomStorage` holds no values, so it neither widens nor folds; see its doc.
110        lhs.contraction(rhs, out)
111    }
112
113    fn scatter<D: Scalar, Src: M, Key: M, Dst: M, Idx: M>(
114        src: &Self::Storage<D>,
115        dst: &mut Self::Storage<D>,
116        index: &Self::Storage<i32>,
117        scaled: bool,
118    ) {
119        src.scatter::<Src, Key, Dst, Idx>(dst, index, scaled);
120    }
121
122    fn gather<D: MaterializableScalar, Src: M, Dst: M, Idx: M>(
123        src: &Self::Storage<D>,
124        dst: &mut Self::Storage<D>,
125        index: &Self::Storage<i32>,
126        scaled: bool,
127    ) {
128        src.gather::<Src, Dst, Idx>(dst, index, scaled);
129    }
130
131    fn reshape<D: Scalar, Src: M, Dst: M>(src: &Self::Storage<D>) -> Self::Storage<D> {
132        src.reshape::<Src, Dst>()
133    }
134
135    fn transmute<D: MaterializableScalar, Src: M, Dst: M>(
136        storage: Self::Storage<D>,
137        _src_map: &MappingValue,
138        _dst_map: &MappingValue,
139    ) -> Self::Storage<D> {
140        // `PhantomStorage` holds no buffer; a relabel is purely type-level.
141        storage
142    }
143
144    async fn to_hbm<D: MaterializableScalar, Element: M, Chip: M, Element2: M>(
145        _host: &HostTensor<D, Element, Self>,
146    ) -> HbmTensor<D, Chip, Element2, Self> {
147        HbmTensor::new(Tensor::empty(), 0)
148    }
149
150    async fn from_hbm<D: MaterializableScalar, Chip: M, Element: M, Element2: M>(
151        _hbm: &HbmTensor<D, Chip, Element, Self>,
152    ) -> HostTensor<D, Element2, Self> {
153        Tensor::empty().into()
154    }
155}
