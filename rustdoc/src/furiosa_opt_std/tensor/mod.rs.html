20/// Tensor with scalar type `D`, axes determined by `Mapping`, and backend determined by `B`.
21///
22/// A thin newtype over the backend's concrete tensor type ([`Backend::Storage`]). `B` defaults to
23/// [`CurrentBackend`], a cfg-aliased type that picks
24/// Emulation / Npu / Typecheck. User code writes `Tensor<D, M>` and gets the
25/// backend-appropriate storage automatically. Explicit `Tensor<D, M, SomeBackend>` overrides for
26/// testing / cross-backend code.
27pub struct Tensor<D: Scalar, Mapping: M, B: Backend = CurrentBackend> {
28    pub(crate) inner: B::Storage<D>,
29    // `B::Storage<D>` drops `Mapping` at the type level (storage-layer erasure); this phantom keeps
30    // the layout traveling at the `Tensor` level so memory-tier structs stay generic and inferable.
31    pub(crate) _marker: std::marker::PhantomData<Mapping>,
32}
33
34impl<D: Scalar, Mapping: M, B: Backend> fmt::Debug for Tensor<D, Mapping, B> {
35    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
36        f.debug_struct("Tensor").field("inner", &self.inner).finish()
37    }
38}

45        }
46    }

56    /// Wraps a backend-native concrete tensor in `Tensor`.
57    pub(crate) fn from_inner(inner: B::Storage<D>) -> Self {
58        Self {
59            inner,
60            _marker: std::marker::PhantomData,
61        }
62    }

64    /// Consumes the tensor and returns its underlying backend-native concrete tensor.
65    pub fn into_inner(self) -> B::Storage<D> {
66        self.inner
67    }

69    /// Creates a new tensor from an initialized buffer. Panics if the buffer length does not match
70    /// `Mapping::SIZE`.
71    pub fn from_vec(data: impl IntoIterator<Item = D>) -> Self {
72        // The length check is backend-independent `Mapping::SIZE` arithmetic, so it lives here in the
73        // wrapper rather than on the construct-only `Backend` seam.
74        let data: Vec<D> = data.into_iter().collect();
75        assert_eq!(
76            data.len(),
77            Mapping::SIZE,
78            "from_vec: buffer length does not match mapping size"
79        );
80        Self::from_inner(B::from_vec(&Mapping::to_value(), data))
81    }
82
83    /// Creates a tensor from a pre-packed device byte image (the packed buffer, [`Self::into_buf`]'s
84    /// inverse). Contrast [`Self::from_vec`], which packs logical values: pre-packed data (fp4 / f4e2m1
85    /// weights) comes through here so the bytes are stored as-is, not decoded to a logical `Vec` and
86    /// re-packed. Panics if the byte length does not match the mapping's packed size.
87    pub fn from_buf(buf: Vec<u8>) -> Self {
88        assert_eq!(
89            buf.len() * 8,
90            Mapping::SIZE * D::BITS,
91            "from_buf: byte length does not match the mapping's packed size"
92        );
93        Self::from_inner(B::from_buf(&Mapping::to_value(), buf))
94    }

96    /// Returns the tensor data as a flat `Vec<D>`, consuming the tensor.
97    pub fn into_vec(self) -> Vec<D>
98    where
99        D: MaterializableScalar,
100    {
101        B::into_vec(self.inner, &Mapping::to_value())
102    }

104    /// Returns the tensor's dense physical device byte image (packed on `D::BITS`), consuming the
105    /// tensor. Delegates to the backend's [`Backend::into_buf`]; Emulation / Npu move the packed
106    /// `BufStorage` bytes out directly.
107    pub fn into_buf(self) -> Vec<u8>
108    where
109        D: MaterializableScalar,
110    {
111        B::into_buf(self.inner, &Mapping::to_value())
112    }

114    /// Maps the tensor element-wise to a new scalar; same layout. The bare-scalar closure is lifted
115    /// by the backend's own [`Backend::map`] strategy (Emulation/Npu a bare buffer walk, Typecheck a
116    /// no-op), no offset-uniform primitive imposed across storages.
117    ///
118    /// Only valid for a `D` whose storage-native length recovery is exact -- every
119    /// [`crate::scalar::MaterializableScalar`] type, where `Scalar::BITS` always agrees with what
120    /// `to_buf`/`load`/`store` actually pack per element. A non-materializable staging type (`i5`/`i9`)
121    /// does NOT hold that invariant (`BITS` names a hardware-only wire width, disconnected from its host
122    /// `size_of`-based default `load`/`store`) and must use [`Self::map_bounded`] instead -- see that
123    /// method's doc for why this one is unsound for it.
124    pub fn map<D2: Scalar, F>(&self, f: F) -> Tensor<D2, Mapping, B>
125    where
126        F: Fn(D) -> D2 + Sync,
127        D: MaterializableScalar,
128    {
129        Tensor::from_inner(B::map(&self.inner, f))
130    }
131
132    /// [`Self::map`], but bounded by this tensor's own `Mapping::SIZE` instead of trusting the backend
133    /// storage's own length recovery.
134    ///
135    /// For most `D` this is equivalent to [`Self::map`] (redundant, so prefer the simpler `map`). It
136    /// exists for a non-`MaterializableScalar` staging type (`i5`/`i9`, the zero-point-subtracted
137    /// contraction operands): such a `D` is deliberately never routed through any device image, HBM, or
138    /// DM (`MaterializableScalar` gates those entry points), so its `Scalar::BITS` describes only a
139    /// real-hardware wire width -- for `i9` that's 8 bits, one physical byte, because the extra bit is
140    /// produced inside the contraction engine at consumption time and never actually crosses a wire or
141    /// sits in device memory. Host-side (Emulation), `i5`/`i9` still need a REAL, full-precision value
142    /// (`fetch_zero_point_sub` computes the exact zero-point-subtracted result eagerly), so they stay in
143    /// their full in-memory representation (`i9` is `i16`-backed) -- neither overrides `load`/`store`/
144    /// `to_buf`, so those default to `size_of::<Self>()`, twice `BITS`'s 8-bit claim. `self.len()`
145    /// (used by [`Self::map`] via [`crate::storage::buf::BufStorage::par_iter`]) derives the element
146    /// count from the buffer's byte length using `BITS`, so for `i9` it silently reports DOUBLE the
147    /// true count, and an unpaired whole-buffer walk (this widen has no shorter sibling to truncate it,
148    /// unlike a `zip_with`) reads/writes past the buffer once the count crosses the true half.
149    /// `Mapping::SIZE` is the type-level truth regardless of any of that, so bound the walk on it here.
150    pub(crate) fn map_bounded<D2: Scalar, F>(&self, f: F) -> Tensor<D2, Mapping, B>
151    where
152        F: Fn(D) -> D2 + Sync,
153    {
154        Tensor::from_inner(B::map_bounded(&self.inner, Mapping::SIZE, f))
155    }
156
157    /// Zips two same-layout tensors element-wise. Delegates to the backend's [`Backend::zip_with`];
158    /// the result is defined only where both inputs are.
159    pub fn zip_with<D2: MaterializableScalar, D3: Scalar, F>(
160        &self,
161        other: &Tensor<D2, Mapping, B>,
162        f: F,
163    ) -> Tensor<D3, Mapping, B>
164    where
165        F: Fn(D, D2) -> D3 + Sync,
166        D: MaterializableScalar,

168        Tensor::from_inner(B::zip_with(&self.inner, &other.inner, f))

171    /// Performs reduction (sum) over axes not present in `Dst` (strict: no broadcast).
172    pub fn reduce_add<Dst: M>(&self) -> Tensor<D, Dst, B>
173    where
174        D: Zero + MaterializableScalar,
175    {
176        self.reduce::<Dst>(|a, b| a + b, D::zero(), false)
177    }

179    /// Reduces the factors of `self`'s mapping that are absent in `Dst`. Axes that `Dst` adds beyond
180    /// the reduced source are broadcast (the reduced value repeats across them); `allow_broadcast`
181    /// gates that, `false` is a plain reduce that requires `Dst` to be a pure factor and rejects an
182    /// unintended extra axis, `true` permits the broadcast (like [`Self::transpose`]'s flag).
183    ///
184    /// The reduce fn works over bare scalars; the backend lifts it to its cell representation.
185    /// Emulation/Npu have no undefined cells.
186    pub fn reduce<Dst: M>(
187        &self,
188        reduce_fn: impl Fn(D, D) -> D + Sync,
189        identity: D,
190        allow_broadcast: bool,
191    ) -> Tensor<D, Dst, B>
192    where
193        D: MaterializableScalar,
194    {
195        Tensor::from_inner(B::reduce::<D, Mapping, Dst>(
196            &self.inner,
197            reduce_fn,
198            identity,
199            allow_broadcast,
200        ))
201    }

203    /// Creates a tensor with every logical cell set to `value`, a scalar broadcast ("splat").
204    /// Padding cells stay uninitialized.
205    pub fn splat(value: D) -> Self {
206        Self::from_vec(std::iter::repeat_n(value, Mapping::SIZE))
207    }

209    /// Creates a tensor with every logical cell set to zero (every [`Scalar`] is `Zero` via `Num`).
210    pub fn zero() -> Self {
211        Self::splat(num_traits::Zero::zero())
212    }

214    /// Creates a random tensor.
215    pub fn rand(rng: &mut impl Rng) -> Self
216    where
217        StandardUniform: rand::distr::Distribution<D>,
218    {
219        Self::from_vec((0..Mapping::SIZE).map(|_| rng.random::<D>()))
220    }

222    /// Performs contraction between two tensors: a generalized matmul that multiplies `lhs` and `rhs`
223    /// over the shared `Union` index space and reduces the axes absent from the output `Mapping`.
224    /// Delegates to [`Backend::contraction`], which the host backend fuses into a single accumulating
225    /// pass (Emulation does, so no full `Union` outer product is materialized).
226    pub fn contraction<Union: M, Lhs: M, Rhs: M>(lhs: &Tensor<D, Lhs, B>, rhs: &Tensor<D, Rhs, B>) -> Self
227    where
228        D: ContractionCast + MaterializableScalar,
229    {
230        Tensor::from_inner(B::contraction(
231            &lhs.inner,
232            &rhs.inner,
233            &Lhs::to_value(),
234            &Rhs::to_value(),
235            &Union::to_value(),
236            &Mapping::to_value(),
237        ))
238    }

241/// Host-side tensor methods.
242impl<D: Scalar, Mapping: M, B: Backend> Tensor<D, Mapping, B> {
243    /// The empty destination canvas for the relayout ops (transpose / scatter / gather / reduce /
244    /// reshape), which fill only the live cells; cells no op writes (padding and any gaps) are
245    /// don't-care. `pub(crate)`: no public constructor hands out an undefined-value tensor (an
246    /// external caller that wants zeros uses `splat`/`zero`).
247    pub(crate) fn zeroed() -> Self {
248        // Replace zeroed with uninit (zero-cost) once MaybeUninit rework of Buf lands.
249        Self::from_inner(B::zeroed(&Mapping::to_value()))
250    }
251
252    /// Creates a mutable view of the tensor.
253    pub fn view_mut<'l>(&'l mut self) -> TensorViewMut<'l, D, Mapping, B> {
254        TensorViewMut::new(&mut self.inner)
255    }
256
257    /// Creates an immutable view of the tensor.
258    pub fn view<'l>(&'l self) -> TensorView<'l, D, Mapping, B> {
259        TensorView::new(&self.inner)
260    }

262    /// Rewraps the tensor under a different mapping.
263    ///
264    /// After the `Mapping`-type-parameter erasure of the storage layer, `B::Storage<D>` no longer
265    /// carries the mapping at the type level, so the inner storage of `Tensor<D, Mapping, B>` and
266    /// `Tensor<D, Mapping2, B>` is literally the same type, rewrapping it is a plain, safe move.
267    ///
268    /// Deliberately NOT size-checked: `transmute` is used to relabel a padded read down to its
269    /// compact slot (the vector engine does `…tile(k).read().transmute()`, shrinking the mapping),
270    /// so `Mapping::SIZE != Mapping2::SIZE` is a legitimate, intended case. Delegates to
271    /// [`Backend::transmute`], which COMPACTS the physical `BufStorage` buffer (drops the padding
272    /// the relabel removes).
273    pub fn transmute<Mapping2: M>(self) -> Tensor<D, Mapping2, B>
274    where
275        D: MaterializableScalar,
276    {
277        Tensor::from_inner(B::transmute::<D, Mapping, Mapping2>(
278            self.inner,
279            &Mapping::to_value(),
280            &Mapping2::to_value(),
281        ))
282    }
283
284    /// Reshapes the tensor to a different mapping by reinterpreting its buffer under `Mapping2`
285    /// (delegates to [`Backend::reshape`]). Relabels axes over an unchanged buffer; the element count
286    /// is checked (`Mapping::SIZE == Mapping2::SIZE`, asserted below).
287    ///
288    /// Kept a *runtime* `assert_eq!`, not a `const` block: every
289    /// `{Dm,Hbm}Tensor(View/ViewMut)::reshape` delegates its `Element`-inclusive combined mapping here
290    /// (their own per-axis checks stop at `Chip`/`Cluster`/`Slice` -- see
291    /// [`constraints::assert_reshape_dimension_preserved`]'s TODO: some current examples reshape with
292    /// a mismatched `Element`). A `const` block is evaluated at monomorphization regardless of whether
293    /// the call is ever reached at runtime, so making this check static would trip on those dead-but-
294    /// compiled paths too, not just the ones actually exercised.
295    ///
296    /// # Safety
297    ///
298    /// `Mapping` and `Mapping2` must lay the elements out in the SAME physical (wire) order, so the
299    /// element at each wire position is unchanged by the relabel. Axis regrouping (merge/split)
300    /// preserves wire order and is valid; a permutation like `m![A, B]` → `m![B, A]` does NOT, that
301    /// reorders elements and must use `transpose`. Equal sizes alone do not guarantee this.
302    pub unsafe fn reshape<Mapping2: M>(&self) -> Tensor<D, Mapping2, B> {
303        assert_eq!(Mapping::SIZE, Mapping2::SIZE);
304        Tensor::from_inner(B::reshape::<D, Mapping, Mapping2>(&self.inner))
305    }
306
307    /// Performs transpose. The mapping division is validated inside
308    /// [`Backend::transpose`] (via `transpose_broadcast`), so this generic body works
309    /// for every backend.
310    pub fn transpose<Dst: M>(&self, allow_broadcast: bool) -> Tensor<D, Dst, B> {
311        let mut dst = Tensor::zeroed();
312        dst.view_mut().transpose(self.view(), allow_broadcast);
313        dst
314    }
315
316    /// Scatter elements from self into dst at positions given by index tensor. Delegates to
317    /// [`Backend::scatter`].
318    pub fn scatter<Key: M, Dst: M, Idx: M>(
319        &self,
320        dst: &mut Tensor<D, Dst, B>,
321        index: &Tensor<i32, Idx, B>,
322        scaled: bool,
323    ) {
324        B::scatter::<D, Mapping, Key, Dst, Idx>(&self.inner, &mut dst.inner, &index.inner, scaled);
325    }
326
327    /// Gather elements from self (table) into dst at positions given by index tensor. Delegates
328    /// to [`Backend::gather`].
329    pub fn gather<Dst: M, Idx: M>(&self, dst: &mut Tensor<D, Dst, B>, index: &Tensor<i32, Idx, B>, scaled: bool)
330    where
331        D: MaterializableScalar,
332    {
333        B::gather::<D, Mapping, Dst, Idx>(&self.inner, &mut dst.inner, &index.inner, scaled);
334    }
335}
