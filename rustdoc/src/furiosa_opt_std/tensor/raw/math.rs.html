11/// Math raw tensor (Simulation): `ArrayD<Opt<D>>` host buffer with full ALU semantics.
12///
13/// `data` only stores the math (non-padding) values: its shape is the product of the axes'
14/// moduli. Buffers in physical layout order (length = `Mapping::SIZE`, includes padding) are
15/// loaded position-by-position via [`Self::write_index`], which silently no-ops on padding
16/// positions. Constructors that want bulk-load from such a buffer should iterate
17/// `Index::new().gen_indexes(Mapping::to_value())` and call `write_index`.
18#[derive(Debug, Clone, PartialEq, Eq)]
19#[doc(hidden)]
20pub struct MathRawTensor<D: Scalar> {
21    axes: Vec<Term>,
22    data: ArrayD<Opt<D>>,
23}
24
25impl<D: Scalar> RawTensor<D> for MathRawTensor<D> {
26    fn axes(&self) -> &[Term] {
27        &self.axes
28    }
29
30    fn uninit_from_axes(axes: Vec<Term>) -> Self {
31        let shape = shape_from_axes(&axes);
32        let data = ArrayD::from_elem(IxDyn(&shape), Opt::Uninit);
33        Self { axes, data }
34    }
35
36    fn read_index(&self, index: Index) -> Opt<D> {
37        let Some(coords) = finalize_coords(&self.axes, index) else {
38            return Opt::Uninit;
39        };
40        *self.data.get(coords.as_slice()).expect("Index out of bounds.")
41    }
42
43    fn write_index(&mut self, index: Index, value: Opt<D>) {
44        let Some(coords) = finalize_coords(&self.axes, index) else {
45            return;
46        };
47        *self.data.get_mut(coords.as_slice()).expect("Index out of bounds.") = value;
48    }
49
50    fn from_buf<Mapping: M>(data: impl IntoIterator<Item = D>) -> Self {
51        let mut tensor = Self::uninit_from_axes(gen_axes::<Mapping>());
52        for (index, value) in Mapping::to_value().indexes().into_iter().zip(data) {
53            tensor.write_index(index, Opt::Init(value));
54        }
55        tensor
56    }
57
58    fn to_buf<Mapping: M>(&self) -> Vec<D> {
59        Mapping::to_value()
60            .indexes()
61            .into_iter()
62            .map(|index| match self.read_index(index) {
63                Opt::Init(value) => value,
64                Opt::Uninit => panic!(
65                    "MathRawTensor::to_buf called on a tensor containing Opt::Uninit slots; \
66                     use the logical Opt-buffer view instead."
67                ),
68            })
69            .collect()
70    }
71
72    fn to_buf_or_default<Mapping: M>(&self) -> Vec<D> {
73        self.to_buf_or_default_opt::<Mapping>()
74    }
75
76    fn map<D2: Scalar, Output: RawTensor<D2>, F>(&self, mut f: F) -> Output
77    where
78        F: FnMut(&Opt<D>) -> Opt<D2>,
79    {
80        let axes = self.axes.to_vec();
81        let mut output = Output::uninit_from_axes(axes.clone());
82        for index in Index::new().gen_indexes(Mapping::from_terms(axes.iter().cloned())) {
83            let value = self.read_index(index.clone());
84            output.write_index(index, f(&value));
85        }
86        output
87    }
88
89    fn reduce<Src: M, Dst: M, Reduce>(&self, reduce_fn: Reduce, identity: Opt<D>) -> Self
90    where
91        Reduce: Fn(Opt<D>, Opt<D>) -> Opt<D>,
92    {
93        // Carve Dst out of Src for the reduced axes, derived structurally so a partial-axis reduction
94        // survives gen_axes consolidation.
95        let reduce_residue = Src::to_value().carve(&Dst::to_value());
96        let mut output = Self::uninit_from_axes(gen_axes::<Dst>());
97        for dst_index in Index::new().gen_indexes(Dst::to_value()) {
98            let mut acc = identity;
99            for src_index in dst_index.clone().gen_indexes(reduce_residue.clone()) {
100                acc = reduce_fn(acc, self.read_index(src_index));
101            }
102            output.write_index(dst_index, acc);
103        }
104        output
105    }
106
107    fn reduce_then_broadcast<Src: M, Dst: M, Reduce>(&self, reduce_fn: Reduce, identity: Opt<D>) -> Self
108    where
109        Reduce: Fn(Opt<D>, Opt<D>) -> Opt<D>,
110    {
111        let src = Src::to_value();
112        let dst = Dst::to_value();
113        // Broadcast axes: the Dst axes built from symbols absent in Src (a symbol-level split).
114        let broadcast = broadcast_axes(&src, &dst);
115        // Kept axes: what's left of Dst after carving the broadcast out.
116        let inter = dst.carve(&broadcast);
117        // Reduced axes: what's left of Src after carving the kept axes out.
118        let reduce_residue = src.carve(&inter);
119
120        let mut output = Self::uninit_from_axes(gen_axes::<Dst>());
121        for inter_index in Index::new().gen_indexes(inter) {
122            let mut acc = identity;
123            for src_index in inter_index.clone().gen_indexes(reduce_residue.clone()) {
124                acc = reduce_fn(acc, self.read_index(src_index));
125            }
126            for dst_index in inter_index.gen_indexes(broadcast.clone()) {
127                output.write_index(dst_index, acc);
128            }
129        }
130        output
131    }
132
133    fn reshape<Mapping: M, Mapping2: M>(self) -> Self {
134        assert_eq!(Mapping::SIZE, Mapping2::SIZE);
135        Self::from_opt_buf::<Mapping2>(self.to_opt_buf::<Mapping>())
136    }
137
138    fn write_transpose<Src: M, Dst: M>(
139        &mut self,
140        src: &Self,
141        src_offset: &Index,
142        dst_offset: &Index,
143        allow_broadcast: bool,
144    ) {
145        let broadcast = transpose_broadcast::<Src, Dst>(allow_broadcast);
146        for index in Index::new().gen_indexes(Src::to_value()) {
147            let mut src_index = index.clone();
148            src_index.add(src_offset.clone());
149            let value = src.read_index(src_index);
150
151            let mut dst_index_base = index;
152            dst_index_base.add(dst_offset.clone());
153            for broadcast_index in dst_index_base.gen_indexes(broadcast.clone()) {
154                self.write_index(broadcast_index, value);
155            }
156        }
157    }
158
159    fn zip_with<D2, D3, Other, Output, F>(&self, rhs: &Other, f: F) -> Output
160    where
161        D2: Scalar,
162        D3: Scalar,
163        Other: RawTensor<D2>,
164        Output: RawTensor<D3>,
165        F: Fn(Opt<D>, Opt<D2>) -> Opt<D3>,
166    {
167        assert_zip(self.axes(), rhs.axes());
168        let axes = self.axes().to_vec();
169        let mut output = Output::uninit_from_axes(axes.clone());
170        for index in Index::new().gen_indexes(Mapping::from_terms(axes.iter().cloned())) {
171            let l = self.read_index(index.clone());
172            let r = rhs.read_index(index.clone());
173            output.write_index(index, f(l, r));
174        }
175        output
176    }

178    fn write_scatter<Src, Key, Dst, Idx, IdxRaw>(&self, dst: &mut Self, index: &IdxRaw, scaled: bool)
179    where
180        Src: M,
181        Key: M,
182        Dst: M,
183        Idx: M,
184        IdxRaw: RawTensor<i32>,
185    {
186        let key = Key::to_value();
187        let (payload, dst_term) = scatter_params(&Src::to_value(), &Dst::to_value(), &key);
188
189        let index_stride = if scaled {
190            payload.clone().remove_padding().size() * std::mem::size_of::<D>()
191        } else {
192            1
193        };
194
195        let indices: Vec<usize> = (0..Idx::SIZE)
196            .map(|i| {
197                let mut idx = Index::new();
198                idx.add_mapping::<Idx>(i);
199                let opt = index.read_index(idx);
200                let Opt::Init(v) = opt else {
201                    panic!("Scatter index must be initialized")
202                };
203                usize::try_from(v).expect("Scatter index must be non-negative") / index_stride
204            })
205            .collect();
206
207        for payload_index in Index::new().gen_indexes(payload) {
208            for (key_pos, key_index) in Index::new().gen_indexes(key.clone()).into_iter().enumerate() {
209                let mut src_index = payload_index.clone();
210                src_index.add(key_index);
211                let value = self.read_index(src_index);
212
213                let mut dst_index = payload_index.clone();
214                dst_index.add_term(dst_term.clone(), indices[key_pos]);
215                dst.write_index(dst_index, value);
216            }
217        }
218    }
219
220    fn write_gather<Src, Dst, Idx, IdxRaw>(&self, dst: &mut Self, index: &IdxRaw, scaled: bool)
221    where
222        Src: M,
223        Dst: M,
224        Idx: M,
225        IdxRaw: RawTensor<i32>,
226    {
227        let params = gather_params(&Src::to_value(), &Dst::to_value(), &Idx::to_value());
228
229        let index_stride = if scaled {
230            params.payload.clone().remove_padding().size() * std::mem::size_of::<D>()
231        } else {
232            1
233        };
234
235        let indices: Vec<usize> = (0..Idx::SIZE)
236            .map(|i| {
237                let mut idx = Index::new();
238                idx.add_mapping::<Idx>(i);
239                let opt = index.read_index(idx);
240                let Opt::Init(v) = opt else {
241                    panic!("Gather index must be initialized")
242                };
243                usize::try_from(v).expect("Gather index must be non-negative") / index_stride
244            })
245            .collect();
246
247        for payload_index in Index::new().gen_indexes(params.payload) {
248            for (idx_pos, dst_iter_index) in Index::new()
249                .gen_indexes(params.idx_residue.clone())
250                .into_iter()
251                .enumerate()
252            {
253                let mut src_index = payload_index.clone();
254                src_index.add_term(params.src_term.clone(), indices[idx_pos]);
255                let value = self.read_index(src_index);
256
257                let mut dst_index = payload_index.clone();
258                dst_index.add(dst_iter_index);
259                dst.write_index(dst_index, value);
260            }
261        }
262    }

264    fn apply_branch_operands<Mapping, Operand, TagRaw, F>(
265        &self,
266        tag: &TagRaw,
267        operands: &[Operand],
268        mut update: F,
269    ) -> Self
270    where
271        D: VeScalar,
272        Mapping: M,
273        TagRaw: RawTensor<u8>,
274        Operand: OperandTag<D, Mapping>,
275        F: FnMut(&Index, &Operand, &mut Self),
276    {
277        let mut output = self.clone();
278        for index in Index::new().gen_indexes(Mapping::to_value()) {
279            let eid = tag.read_index(index.clone());
280            let Opt::Init(_) = eid else {
281                continue;
282            };
283            for operand in operands {
284                if !operand.tag_filter().matches(eid) {
285                    continue;
286                }
287                update(&index, operand, &mut output);
288            }
289        }
290        output
291    }
292}

294impl<D: Scalar> RawTensorOpt<D> for MathRawTensor<D> {
295    fn from_opt_buf<Mapping: M>(data: impl IntoIterator<Item = Opt<D>>) -> Self {
296        // Physical-layout iteration: a padding position's finalized index is `RErr`, so
297        // `write_index` silently no-ops there. The corresponding `data` value (typically
298        // `Opt::Uninit` for padding slots) is dropped, matching `MathRawTensor`'s
299        // padding-stripped `ArrayD` representation.
300        let mut tensor = Self::uninit_from_axes(gen_axes::<Mapping>());
301        for (index, value) in Mapping::to_value().indexes().into_iter().zip(data) {
302            tensor.write_index(index, value);
303        }
304        tensor
305    }
306
307    fn to_opt_buf<Mapping: M>(&self) -> Vec<Opt<D>> {
308        Mapping::to_value()
309            .indexes()
310            .into_iter()
311            .map(|index| self.read_index(index))
312            .collect()
313    }
