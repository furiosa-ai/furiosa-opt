9/// Typecheck tensor: metadata only (axes), no host buffer. All reads return `Opt::Uninit` and all
10/// writes are silently dropped.
11///
12/// Each op runs the relevant `op_prep` helper for its panic side (so mapping errors still surface
13/// under Typecheck) and then leaves the (empty) phantom output untouched, no per-element iteration.
14#[derive(Clone, Debug)]
15pub struct PhantomStorage<D: Scalar> {
16    axes: Vec<AxisTerm>,
17    _phantom: PhantomData<D>,
18}
19
20impl<D: Scalar> PartialEq for PhantomStorage<D> {
21    fn eq(&self, other: &Self) -> bool {
22        self.axes == other.axes
23    }
24}
25impl<D: Scalar> Eq for PhantomStorage<D> {}
26
27impl<D: Scalar> PhantomStorage<D> {
28    /// An empty phantom tensor of the given `Mapping`'s layout, Typecheck carries axes only, no
29    /// values.
30    fn new<Mapping: M>() -> Self {
31        Self::new_from_axes(Mapping::to_value().axes())
32    }
33
34    /// An empty phantom tensor carrying `axes` directly. Lets elementwise ops preserve the source
35    /// axes without naming a mapping type.
36    fn new_from_axes(axes: Vec<AxisTerm>) -> Self {
37        Self {
38            axes,
39            _phantom: PhantomData,
40        }
41    }
42
43    /// Typecheck carries no values; map/zip produce an empty phantom of the SAME axes, no
44    /// per-element iteration. Back [`crate::backend::Backend::map`] / [`crate::backend::Backend::zip_with`].
45    pub(crate) fn map<D2: Scalar>(&self, _f: impl FnMut(D) -> D2) -> PhantomStorage<D2> {
46        PhantomStorage::new_from_axes(self.axes.clone())
47    }
48
49    pub(crate) fn zip_with<D2: Scalar, D3: Scalar>(
50        &self,
51        _other: &PhantomStorage<D2>,
52        _f: impl Fn(D, D2) -> D3,
53    ) -> PhantomStorage<D3> {
54        PhantomStorage::new_from_axes(self.axes.clone())
55    }
56
57    /// Ternary peer of [`Self::zip_with`]; phantom carries no values.
58    pub(crate) fn zip3_with<D2: Scalar, D3: Scalar, D4: Scalar>(
59        &self,
60        _b: &PhantomStorage<D2>,
61        _c: &PhantomStorage<D3>,
62        _f: impl Fn(D, D2, D3) -> D4,
63    ) -> PhantomStorage<D4> {
64        PhantomStorage::new_from_axes(self.axes.clone())
65    }
66
67    /// Typecheck no-op: runs only the structural check, leaving the empty phantom `self` untouched.
68    /// Backs [`crate::backend::Backend::transpose`].
69    pub(crate) fn transpose<Src: M, Mapping: M>(
70        &mut self,
71        _src: &PhantomStorage<D>,
72        _src_offset: &Index,
73        _dst_offset: &Index,
74        allow_broadcast: bool,
75    ) {
76        let _ = transpose_broadcast::<Src, Mapping>(allow_broadcast);
77    }
78
79    /// Backs [`crate::backend::Backend::reduce`].
80    pub(crate) fn reduce<Src: M, Dst: M, R: Fn(D, D) -> D>(
81        &self,
82        _reduce_fn: R,
83        _identity: D,
84        allow_broadcast: bool,
85    ) -> Self {
86        // Structural parity check with the value backends (`BufStorage::reduce`): reject an
87        // unintended broadcast, then run the same containment carves so a non-contained / over-padded
88        // reduce is rejected at type-check time exactly as it is at runtime. The carves panic on
89        // a mismatch; their offsets are unused here since Typecheck moves no values.
90        let src = Src::to_value();
91        let dst = Dst::to_value();
92        let broadcast = broadcast_axes(&src, &dst);
93        assert!(
94            allow_broadcast || broadcast.axes().is_empty(),
95            "reduce: Dst adds axes absent from the source; pass allow_broadcast=true for reduce-then-broadcast"
96        );
97        let inter = dst.carve(&broadcast);
98        let _reduce_residue = src.carve(&inter);
99        Self::new::<Dst>()
100    }
101
102    /// Backs [`crate::backend::Backend::contraction`]. Typecheck carries no values, so it neither
103    /// widens nor folds; `out` is a runtime value here (the deferred engine-level fold only has it as
104    /// one), so it is taken directly rather than rebuilt from a static mapping type.
105    pub(crate) fn contraction(&self, _rhs: &Self, out: &MappingValue) -> Self {
106        Self::new_from_axes(out.axes())
107    }
108
109    /// Backs [`crate::backend::Backend::scatter`].
110    // `Idx` is unused here (Typecheck decodes no indices) but must match the `Backend::scatter`
111    // signature that `BufStorage` needs (`Idx::SIZE`).
112    #[expect(
113        clippy::extra_unused_type_parameters,
114        reason = "signature parity with Backend::scatter"
115    )]
116    pub(crate) fn scatter<Src: M, Key: M, Dst: M, Idx: M>(
117        &self,
118        _dst: &mut PhantomStorage<D>,
119        _index: &PhantomStorage<i32>,
120        _scaled: bool,
121    ) {
122        let _ = scatter_params(&Src::to_value(), &Dst::to_value(), &Key::to_value());
123    }
124
125    /// Backs [`crate::backend::Backend::gather`].
126    pub(crate) fn gather<Src: M, Dst: M, Idx: M>(
127        &self,
128        _dst: &mut PhantomStorage<D>,
129        _index: &PhantomStorage<i32>,
130        _scaled: bool,
131    ) {
132        let _ = gather_params(&Src::to_value(), &Dst::to_value(), &Idx::to_value());
133    }
134
135    /// Backs [`crate::backend::Backend::reshape`].
136    pub(crate) fn reshape<Mapping: M, Mapping2: M>(&self) -> Self {
137        assert_eq!(Mapping::SIZE, Mapping2::SIZE);
138        Self::new::<Mapping2>()
139    }
140
141    /// Typecheck has no values, the input buffer is dropped regardless of its length. (Length is
142    /// still validated by [`crate::tensor::Tensor::from_vec`] in the wrapper, unlike the `Opt` path.)
143    pub(crate) fn from_vec(mapping: &MappingValue, _data: impl IntoIterator<Item = D>) -> Self {
144        Self::new_from_axes(mapping.axes())
145    }
146
147    /// Typecheck carries no values.
148    pub(crate) fn into_vec(self, _mapping: &MappingValue) -> Vec<D> {
149        Vec::new()
150    }
151
152    /// An empty phantom tensor of this layout.
153    pub(crate) fn zeroed(mapping: &MappingValue) -> Self {
154        Self::new_from_axes(mapping.axes())
155    }
156}
