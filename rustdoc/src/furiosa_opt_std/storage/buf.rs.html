13/// Emulation / Npu tensor: the dense device image as one packed `Vec<u8>`. Element `i` sits at position
14/// `i` (padding included, zero-initialized); a byte-multiple `D` takes its own byte run, a sub-byte `D`
15/// (`f4e2m1` / `i4`) packs two per byte. Only [`Scalar::load`] / [`Scalar::store`] know the width, so the
16/// generic-over-`D` ops address elements by logical index. The element count is byte-aligned (enforced by
17/// [`Self::from_vec`]), so it is recovered from the byte length alone.
18///
19/// Layout-free: ops take their mapping(s) from type parameters. Reads parallelize via [`Self::par_iter`];
20/// a parallel write partitions the byte image into disjoint chunks ([`Self::par_chunks_mut`]).
21#[derive(Clone, Debug)]
22pub struct BufStorage<D: Scalar> {
23    bytes: Vec<u8>,
24    _marker: PhantomData<D>,
25}
26
27/// Byte-wise equality over the packed device image: two `BufStorage`s are equal iff their raw bytes
28/// match. This is the right notion for an opaque storage buffer: identical bit patterns (a `NaN`
29/// included) compare equal, which is what tensor-image comparison wants. Byte equality is
30/// intentionally STRICTER than `D`'s value equality: `from_vec` stores each code verbatim (it does
31/// not canonicalize), so the two `f4e2m1` zero codes (`0x0` = +0, `0x8` = -0) are value-equal yet
32/// compare unequal here. (Float `PartialEq`'s `NaN != NaN` is a value-level rule and does not apply
33/// to a byte image.)
34impl<D: Scalar> PartialEq for BufStorage<D> {
35    fn eq(&self, other: &Self) -> bool {
36        self.bytes == other.bytes
37    }
38}
39impl<D: Scalar + Eq> Eq for BufStorage<D> {}

42    /// The packed byte length for `n` elements (`n * BITS / 8`, exact: `n` is byte-aligned by the
43    /// [`Self::from_vec`] invariant). A byte-multiple width divides evenly for any `n`; a 4-bit width
44    /// needs an even `n`.
45    fn byte_len(n: usize) -> usize {
46        n * D::BITS / 8
47    }

49    /// Packs a logical `Vec<D>` on `D::BITS` via [`Scalar::store`]. The single constructor and sole
50    /// byte-alignment enforcement point. No `mapping`-size check here (`BufStorage` is layout-free); the
51    /// `data.len() == Mapping::SIZE` gate lives in the `Tensor::from_vec` wrapper, matching `MathStorage`.
52    pub(crate) fn from_vec(data: impl IntoIterator<Item = D>) -> Self {
53        let vals: Vec<D> = data.into_iter().collect();
54        assert!(
55            (vals.len() * D::BITS).is_multiple_of(8),
56            "BufStorage<D>: a sub-byte element count must be byte-aligned (got {} elements at {} bits)",
57            vals.len(),
58            D::BITS,
59        );
60        // `Scalar::to_buf` is the single packer: a byte-multiple width memcpys the whole slice, a
61        // sub-byte width packs two codes per byte (zero-initializing first, since each nibble store is a
62        // read-modify-write that reads the byte to preserve its sibling). Wrapping that output is the
63        // inverse of `into_buf`.
64        Self::from_buf(D::to_buf(&vals))
65    }

77    /// The element count, recovered from the byte length (exact: the count is byte-aligned).
78    ///
79    /// Sound only if `D::BITS` actually matches what `D`'s own `to_buf`/`load`/`store` pack per
80    /// element -- true for every [`crate::scalar::MaterializableScalar`] (a byte-aligned type's `BITS`
81    /// IS `size_of::<D>() * 8` by construction; a sub-byte one, `i4`/`f4e2m1`, overrides all three to
82    /// pack exactly `BITS` bits). A non-materializable staging type (`i5`/`i9`) does NOT hold this: its
83    /// `BITS` names a real-hardware wire width, while its (unoverridden, default) `to_buf`/`load`/
84    /// `store` pack its full host `size_of` -- for `i9` (an `i16`) that's double `BITS`'s 8-bit claim.
85    /// Such a type is architecturally meant to be read/written only through a Mapping-bounded op
86    /// (transpose, reduce, [`crate::tensor::Tensor::map_bounded`]) or as the WRITE target of a widen,
87    /// never through this (or `par_iter`/`map`/`zip_with`/`zip3_with`/`to_vec`/`into_vec`, every one of
88    /// which derives its element count from this method) -- calling any of them on such a type is not
89    /// merely wrong, it silently reads/writes past the buffer once the (over-reported) count crosses
90    /// the true element count. The debug-only check below catches that class of mistake at its very
91    /// first `len()` call rather than downstream as a confusing out-of-range panic (or, worse, a
92    /// same-length sibling silently masking it, as `zip_with` did for i9 until #18934's `map` exposed
93    /// it -- see that PR's fix commit). Cached per-monomorphization (one real `to_buf` probe per `D`
94    /// ever used, not per call) since `len()` sits on the hot path.
95    pub(crate) fn len(&self) -> usize
96    where
97        D: MaterializableScalar,
98    {
99        #[cfg(debug_assertions)]
100        {
101            use std::sync::OnceLock;
102            static CHECKED: OnceLock<()> = OnceLock::new();
103            CHECKED.get_or_init(|| {
104                let bits_based = Self::byte_len(8);
105                let actual: [D; 8] = [num_traits::Zero::zero(); 8];
106                let real = D::to_buf(&actual).len();
107                assert_eq!(
108                    bits_based,
109                    real,
110                    "BufStorage<{}>: `Scalar::BITS` ({}) disagrees with `to_buf`'s own packed byte \
111                     width for 8 elements ({bits_based} vs {real} bytes). This type's whole-buffer \
112                     host ops (`len`/`par_iter`/`map`/`zip_with`/`zip3_with`/`to_vec`/`into_vec`) are \
113                     unsound: `BITS` does not describe what `to_buf`/`load`/`store` actually pack per \
114                     element (a hardware-only staging type, e.g. `i5`/`i9`, has a `BITS` naming a \
115                     real-device wire width that its host in-memory form does not match). Route this \
116                     type through a Mapping-bounded op instead (`Tensor::map_bounded`, transpose, \
117                     reduce), never a bare whole-buffer one.",
118                    std::any::type_name::<D>(),
119                    D::BITS,
120                );
121            });
122        }
123        self.bytes.len() * 8 / D::BITS
124    }

158    /// The elements as an indexed parallel iterator, each decoded via [`Scalar::load`]. Unlike a write, a
159    /// sub-byte read has no aliasing hazard (two threads decoding sibling nibbles of the same byte never
160    /// race), so a plain index range suffices; no byte-chunk partitioning needed here, unlike
161    /// [`Self::par_chunks_mut`] (its mutable, byte-owning peer). Backs [`Self::map`] and the zips.
162    pub(crate) fn par_iter(&self) -> impl IndexedParallelIterator<Item = D> + '_
163    where
164        D: MaterializableScalar,
165    {
166        (0..self.len()).into_par_iter().map(move |i| self.get(i))
167    }

169    /// The elements as a parallel iterator of disjoint, byte-aligned mutable chunks: the safe
170    /// parallel-write primitive (mutable peer of [`Self::par_iter`]). Each rayon job gets one
171    /// [`BufChunkMut`] owning a distinct element range and writes only within it via [`BufChunkMut::set`].
172    ///
173    /// Keying disjointness on the element index is unsound once `BITS < 8`: two elements share a byte,
174    /// so sibling writes from different jobs race on it. Partitioning the byte image makes each chunk
175    /// own whole bytes, race-free for every width with no `unsafe`. `min_elems` is the per-chunk floor
176    /// (rounded up to the byte-alignment quantum). A sub-window writer intersects [`BufChunkMut::range`]
177    /// with its window. Backs transpose / gather.
178    pub(crate) fn par_chunks_mut(
179        &mut self,
180        min_elems: usize,
181    ) -> impl IndexedParallelIterator<Item = BufChunkMut<'_, D>> {
182        // The alignment quantum `align = lcm(8, BITS) / BITS` is the fewest elements that fill a whole
183        // number of bytes (2 for a 4-bit width, 1 for a byte-multiple width); round `min_elems` up to it so
184        // each chunk owns a whole number of elements, byte-aligned on both ends.
185        const CACHE_LINE: usize = 64;
186        let align = lcm(8, D::BITS) / D::BITS;
187        let elems_per_chunk = min_elems.next_multiple_of(align).max(align);
188        // Round the chunk up to a cache line so two adjacent chunks never share one: byte-disjoint already
189        // guarantees correctness, but a `bytes_per_chunk` off a 64B boundary leaves neighbours sharing a
190        // boundary line, so a write in each false-shares it. 64 is a multiple of every scalar's byte
191        // quantum (1 / 2 / 4B), so the whole-byte invariant is preserved.
192        let bytes_per_chunk = Self::byte_len(elems_per_chunk).next_multiple_of(CACHE_LINE);
193        // Recompute elems from the rounded byte count so `lo = c * elems_per_chunk` stays exact: `64 * 8`
194        // is divisible by every `D::BITS` (4 / 8 / 16 / 32), so the division is lossless.
195        let elems_per_chunk = bytes_per_chunk * 8 / D::BITS;
196        self.bytes
197            .par_chunks_mut(bytes_per_chunk)
198            .enumerate()
199            .map(move |(c, bytes)| {
200                // Each chunk owns the element range starting at `c * elems_per_chunk`; the last chunk may be
201                // short, so its length is what its own bytes hold.
202                let lo = c * elems_per_chunk;
203                let hi = lo + bytes.len() * 8 / D::BITS;
204                BufChunkMut {
205                    bytes,
206                    lo,
207                    hi,
208                    _marker: PhantomData,
209                }
210            })
211    }

214    /// across the rayon pool and `from_vec` repacks the result on `D2::BITS`. Backs
215    /// [`crate::backend::Backend::map`].

232    /// Element-wise zip of two same-layout physical buffers, bare `D`, offset-aligned. Backs
233    /// [`crate::backend::Backend::zip_with`].
234    pub(crate) fn zip_with<D2: MaterializableScalar, D3: Scalar>(
235        &self,
236        other: &BufStorage<D2>,
237        f: impl Fn(D, D2) -> D3 + Sync,
238    ) -> BufStorage<D3>
239    where
240        D: MaterializableScalar,
241    {
242        // Reads never race, so zip the two buffers' parallel element iterators (the physical packing is
243        // transparent through `par_iter`). The output repacks on `D3::BITS` via `from_vec`.
244        let data = self
245            .par_iter()
246            .zip(other.par_iter())
247            .map(|(a, b)| f(a, b))
248            .collect::<Vec<_>>();
249        BufStorage::from_vec(data)
250    }

271    /// Writes a transposed/broadcast view of `src` into `self` (the destination) via a sequencer
272    /// walk over the physical buffer. `src_map` / `dst_map` are the two storages' base (live-axis)
273    /// mappings, used only to resolve a partial-view offset's wire base (see [`window_base`]). Backs
274    /// [`crate::backend::Backend::transpose`].
275    #[allow(clippy::too_many_arguments)]
276    pub(crate) fn transpose<Src: M, Mapping: M>(
277        &mut self,
278        src: &BufStorage<D>,
279        src_offset: &Index,
280        dst_offset: &Index,
281        src_map: &MappingValue,
282        dst_map: &MappingValue,
283        allow_broadcast: bool,
284    ) {
285        // Structural check (also asserts `Src` is contained in `Mapping`); `!allow_broadcast`
286        // rejects a non-padding leftover.
287        let _ = transpose_broadcast::<Src, Mapping>(allow_broadcast);
288        let src_view = &Src::to_value();
289        let dst_view = &Mapping::to_value();
290        // Relayout: sequence the `Src` buffer (memory) against the `Mapping` (dst) layout (stream)
291        // under `Carve`. Each dst buffer position reads one Src element; a broadcast axis (in dst, not Src)
292        // gets `memory_stride` 0, so one Src element feeds the whole broadcast run. `Carve` (not `Read`)
293        // tolerates a `Bottom` pad in the dst stream, a `view_mut().tile()` write hole, read as a `Top`
294        // don't-care, which the dst liveness walk then gates out of the write below.
295        let config = &sequence(&[src_view], &[dst_view], SequencerMode::Carve)
296            .expect("transpose: Src must be covered by the dst stream")[0];
297        // The src window's physical base and the dst window's base. Both default to 0 for an empty
298        // (whole-tensor) offset. The base resolves the offset against the *base* (live-axis) map, where
299        // a tiled offset axis the view carries as padding is still a live `Symbol` (the view shares the
300        // base's wire layout, so the wire position carries over).
301        let src_base = window_base(src_map, src_offset);
302        let dst_base = window_base(dst_map, dst_offset);
303        // Walk the dst window, zipping three position-aligned streams of the dst layout: the dst elements
304        // (each position writes its own, `dst_base + pos`), the src offset to read (`config`, splits via
305        // seek), and the dst liveness (`iter_positions()` yields `None` for a padding / write-hole cell), so
306        // liveness is read inline, not materialized into a mask. A live position reads `src`'s element and
307        // writes its own dst element; the stream's trailing `Bottom`-pad positions, which would address past
308        // the window, are clamped away below. The write is range-partitioned on the dst (each rayon job
309        // owns a byte-aligned element range), so the scatter is safe for every width, no branch on packing.
310        // The per-chunk position range is bounded by both stream lengths so a seek never runs past either.
311        let stream = config.stream_size().min(dst_view.iter_positions().len());
312        self.par_chunks_mut(PAR_MIN_JOB).for_each(|mut chunk| {
313            // The chunk owns the dst elements in `chunk.range()`; the positions it writes are those whose
314            // `dst_base + pos` lands in the chunk (and in range of the stream). Seek both the src offsets
315            // and the liveness to that contiguous position range and walk them position-aligned.
316            let elems = chunk.range();
317            let pos_lo = elems.start.saturating_sub(dst_base);
318            let pos_hi = elems.end.saturating_sub(dst_base).min(stream);
319            if pos_lo >= pos_hi {
320                return;
321            }
322            let src_pos = config.iter_range(pos_lo, pos_hi);
323            let live = dst_view.iter_positions().range(pos_lo, pos_hi);
324            for (i, (src_pos, live)) in src_pos.zip(live).enumerate() {
325                if live.is_some() {
326                    chunk.set(dst_base + pos_lo + i, src.get(src_base + src_pos));
327                }
328            }
329        });
330    }

332    /// Reduces the factors of `self`'s mapping that are absent in `Dst`. `Dst` must be a factor of
333    /// the source mapping (only reduce away existing axes; no broadcast). Backs [`crate::backend::Backend::reduce`].
334    ///
335    /// Per-output-cell: each output cell independently folds its block of
336    /// the reduced (residue) axes into its own slot (`par_iter_mut`), no shared accumulator. The output
337    /// cell's source read base is the cell's coordinate projected onto the source (a broadcast axis `Dst`
338    /// adds projects to 0, so its fan shares one base); the residue deltas are the reduced axes, shared
339    /// across cells. So a read is the additive `base + residue_delta` over the physical buffer.
340    ///
341    /// REASSOCIATES: one output cell's residue block is folded left-to-right in `residue` order, which is
342    /// the physical (wire) order of the reduced axes, not necessarily the serial-`iter` stream order, so
343    /// `reduce_fn` runs in an implementation-defined order. For an associative `reduce_fn` the result
344    /// equals the serial one; for a non-associative one (e.g. `f32` add/mul) the Emulation result can
345    /// differ from serial. Accepted: Emulation is not bit-reproducible for non-associative reductions.
346    /// The order is still deterministic across runs regardless of rayon's split, since each cell folds
347    /// its whole block alone.
348    pub(crate) fn reduce<Src: M, Dst: M, R: Fn(D, D) -> D + Sync>(
349        &self,
350        reduce_fn: R,
351        identity: D,
352        allow_broadcast: bool,
353    ) -> Self
354    where
355        D: MaterializableScalar,
356    {
357        let src = Src::to_value();
358        let dst = Dst::to_value();
359        // Axes `Dst` adds beyond the reduced source are broadcast; `!allow_broadcast` rejects them.
360        let broadcast = broadcast_axes(&src, &dst);
361        assert!(
362            allow_broadcast || broadcast.axes().is_empty(),
363            "reduce: Dst adds axes absent from the source; pass allow_broadcast=true for reduce-then-broadcast"
364        );
365        // The kept axes (`Dst` minus the broadcast) sit in both source and Dst; the reduced (residue)
366        // axes are the source's beyond them. A broadcast axis (in `dst`, not the source) reads nothing,
367        // so its fan copies share one source base and recompute the fold.
368        let inter = dst.carve(&broadcast);
369        let reduce_residue = src.carve(&inter);
370        let out_size = dst.size();
371        let plan = ReadPlan::new(&src, &dst, &reduce_residue);
372        // Fold each independent output cell across the rayon pool into a logical `Vec<D>`, then pack the
373        // result on `D::BITS` via `from_vec`. `with_min_len` holds each job at >= `PAR_MIN_JOB` work
374        // (cells x live residue) so small reduces stay one job. A dead (padding) cell has no reads and
375        // keeps its `identity`. A split-then-padded reduced axis can land a read past the physical
376        // buffer, so `try_get` skips an out-of-range one.
377        let mut out = vec![identity; out_size];
378        out.par_iter_mut()
379            .with_min_len(min_cells_per_job(plan.cell_work()))
380            .enumerate()
381            .for_each(|(o, slot)| {
382                let Some(reads) = plan.reads(o) else { return };
383                let mut acc = identity;
384                for p in reads {
385                    if let Some(v) = self.try_get(p) {
386                        acc = reduce_fn(acc, v);
387                    }
388                }
389                *slot = acc;
390            });
391        Self::from_vec(out)
392    }

394    /// Fused contraction (generalized matmul) for the physical buffer: each output cell independently
395    /// sums `lhs * rhs` over the contracted axes (those in `Union` but absent from `Out`). O(`Out`)
396    /// memory, no `Union` outer product. Backs [`crate::backend::Backend::contraction`] for Emulation.
397    ///
398    /// Per-output-cell: each operand offset splits into the output
399    /// cell's read base plus a contracted-coordinate delta, and each output cell folds that block into
400    /// its own slot (`par_iter_mut`), no shared accumulator. An operand axis absent from `Out` (a
401    /// contracted axis) is a residue delta; an axis the operand lacks broadcasts (its base / delta is 0).
402    ///
403    /// Accumulates in `<D as ContractionCast>::Output`: each operand
404    /// widens on load via [`Cast`] (`D -> Output`), the contracted block sums in the wider type, and the
405    /// sum narrows via [`Cast`] (`Output -> D`) into the slot. So an `i8` matmul accumulates in `i32` and
406    /// a `bf16` matmul in `f32`, instead of overflowing the narrow dtype. The bf16 test below
407    /// (`contraction_bf16_recovers_...`) pins this; the i8 fold shares this code and is deferred to
408    /// dedicated coverage (#176 int-GEMM).
409    ///
410    /// One `Backend::contraction` is a single within-slice contraction (its `Union`/`Out` carry no
411    /// `Slice` axis), so its narrow is the per-slice Lane Folder narrow. A cross-slice reduction is a
412    /// separate downstream Vector Engine `reduce` over the already-narrowed per-slice results, so the
413    /// narrow here never spans slices (pinned by `cross_slice_reduce_narrows_per_lane_fold_not_globally`).
414    ///
415    /// REASSOCIATES, like [`Self::reduce`]: one output cell's products are summed in physical (wire)
416    /// order of the contracted axes, not the serial-`iter` stream order. Exact for an associative
417    /// combine; for `f32` add/mul the Emulation result can differ from serial. Accepted; the order is
418    /// still deterministic across runs regardless of rayon's split.
419    pub(crate) fn contraction(
420        lhs: &BufStorage<D>,
421        rhs: &BufStorage<D>,
422        lhs_map: &MappingValue,
423        rhs_map: &MappingValue,
424        pre_reduce: &MappingValue,
425        out_map: &MappingValue,
426    ) -> BufStorage<D>
427    where
428        D: ContractionCast + MaterializableScalar,
429    {
430        // A bare `BufStorage` carries no axes of its own, so the operand layouts arrive as
431        // `lhs_map`/`rhs_map` rather than being read off the storage (contrast `MathStorage`).
432        // The contracted axes are `pre_reduce` beyond the kept output, exactly `reduce`'s residue.
433        let contracted = pre_reduce.carve(out_map);
434        let out_size = out_map.size();
435        // One read base per output cell into each operand (the output coordinate, contracted at 0), plus
436        // the per-contracted-coordinate operand offset deltas shared across cells; the residue is the
437        // contracted axes (an operand axis absent from `Out` reduces, one it lacks broadcasts).
438        let lhs_plan = ReadPlan::new(lhs_map, out_map, &contracted);
439        let rhs_plan = ReadPlan::new(rhs_map, out_map, &contracted);
440        let mut out = vec![num_traits::Zero::zero(); out_size];
441        // Each output cell sums the products over its contracted block (the two plans' reads zip
442        // position-aligned) in `<D as ContractionCast>::Output`: operands widen on load via `Cast`
443        // (`D -> Output`), the sum narrows via `Cast` (`Output -> D`). A dead (padding) cell has no
444        // reads and keeps its zero. A split-then-padded contracted coordinate can land past a buffer,
445        // so skip an out-of-range read.
446        out.par_iter_mut()
447            .with_min_len(min_cells_per_job(lhs_plan.cell_work()))
448            .enumerate()
449            .for_each(|(o, slot)| {
450                let (Some(lhs_reads), Some(rhs_reads)) = (lhs_plan.reads(o), rhs_plan.reads(o)) else {
451                    return;
452                };
453                // Sum the products in `ContractionCast::Output`: each MAC widens both operands via
454                // `Cast`, `filter_map` drops a split-then-padded out-of-range read, and the accumulator
455                // narrows via `Cast` after the block folds (`fold` seed = wide zero). The narrow sits
456                // outside the fold, the Lane Folder's narrow-once-after-the-reduction.
457                let acc = lhs_reads
458                    .zip(rhs_reads)
459                    .filter_map(|(lp, rp)| match (lhs.try_get(lp), rhs.try_get(rp)) {
460                        (Some(l), Some(r)) => Some(Cast::cast(l) * Cast::cast(r)),
461                        _ => None,
462                    })
463                    .fold(
464                        <<D as ContractionCast>::Output as num_traits::Zero>::zero(),
465                        |acc, prod| acc + prod,
466                    );
467                *slot = Cast::cast(acc);
468            });
469        Self::from_vec(out)
470    }

472    /// Scatters `self` into `dst` at positions read from the `i32` index tensor. Backs
473    /// [`crate::backend::Backend::scatter`].
474    pub(crate) fn scatter<Src: M, Key: M, Dst: M, Idx: M>(
475        &self,
476        dst: &mut BufStorage<D>,
477        index: &BufStorage<i32>,
478        scaled: bool,
479    ) {
480        let key = Key::to_value();
481        let (payload, dst_term) = scatter_params(&Src::to_value(), &Dst::to_value(), &key);
482        let payload = payload.remove_padding();
483        let scatter_axis = MappingValue::from_terms(std::iter::once(dst_term.to_term()));
484        // The index tensor's buffer holds exactly one element per `Idx` position, which is why
485        // `decode_indices` reads it whole.
486        debug_assert_eq!(index.len(), Idx::SIZE);
487        let indices = decode_indices(index, decode_stride::<D>(&payload, scaled));
488
489        let key_size = key.size();
490        let axis_size = scatter_axis.size();
491        let src_off: Vec<usize> = sequence(&[&Src::to_value()], &[&payload.clone().pair(key)], SequencerMode::Read)
492            .expect("scatter: Src factors into payload x key")[0]
493            .iter()
494            .collect();
495        let dst_off: Vec<usize> = sequence(&[&Dst::to_value()], &[&payload.pair(scatter_axis)], SequencerMode::Read)
496            .expect("scatter: Dst factors into payload x scatter-axis")[0]
497            .iter()
498            .collect();
499        // Serial element copy through the accessor (scatter is inherently serial: distinct keys can map to
500        // the same dst element, so the last write wins). The accessor keeps it correct for a packed dst.
501        for payload_pos in 0..src_off.len() / key_size {
502            for key_pos in 0..key_size {
503                let src_elem = src_off[payload_pos * key_size + key_pos];
504                let dst_elem = dst_off[payload_pos * axis_size + indices[key_pos]];
505                let v = self.get(src_elem);
506                dst.set(dst_elem, v);
507            }
508        }
509    }

511    /// Gathers from `self` (table) into `dst` at positions read from the index tensor. Backs
512    /// [`crate::backend::Backend::gather`].
513    pub(crate) fn gather<Src: M, Dst: M, Idx: M>(&self, dst: &mut BufStorage<D>, index: &BufStorage<i32>, scaled: bool)
514    where
515        D: MaterializableScalar,
516    {
517        let params = gather_params(&Src::to_value(), &Dst::to_value(), &Idx::to_value());
518        let payload = params.payload.remove_padding();
519        let gather_axis = MappingValue::from_terms(std::iter::once(params.src_term.to_term()));
520        // The compact residue axes are exactly the index tensor's mapping.
521        let idx_residue = Idx::to_value();
522        let indices = decode_indices(index, decode_stride::<D>(&payload, scaled));
523
524        let axis_size = gather_axis.size();
525        let residue_size = idx_residue.size();
526        let src_off: Vec<usize> = sequence(
527            &[&Src::to_value()],
528            &[&payload.clone().pair(gather_axis)],
529            SequencerMode::Read,
530        )
531        .expect("gather: Src table factors into payload x gather-axis")[0]
532            .iter()
533            .collect();
534        let dst_off: Vec<usize> = sequence(&[&Dst::to_value()], &[&payload.pair(idx_residue)], SequencerMode::Read)
535            .expect("gather: Dst factors into payload x idx-residue")[0]
536            .iter()
537            .collect();
538        // Each payload block writes its `residue_size` dst elements, and `dst_off` is a permutation of dst
539        // positions (distinct), so the blocks scatter into disjoint slots. The index only repeats *reads*
540        // (`src_off[... + indices[r]]`), never a write. The dst positions `dst_off` names are permuted, not
541        // a contiguous range, so range-partitioning the dst (the uniform safe scatter) needs the inverse:
542        // `writer[dst_elem]` is the `(payload_pos, r)` flat index that writes `dst_elem`, or `None` for a
543        // dst position the gather does not touch (which then keeps its prior value).
544        let mut writer: Vec<Option<usize>> = vec![None; dst.len()];
545        for (flat, &dst_elem) in dst_off.iter().enumerate() {
546            writer[dst_elem] = Some(flat);
547        }
548        // Range-partition the dst: each rayon job owns a contiguous, byte-aligned element range, reads its
549        // permuted source through the shared `&` source (a race-free read), and writes only positions it
550        // owns. Safe for every width, no branch on packing.
551        dst.par_chunks_mut(min_cells_per_job(residue_size).saturating_mul(residue_size))
552            .for_each(|mut chunk| {
553                for dst_elem in chunk.range() {
554                    let Some(flat) = writer[dst_elem] else { continue };
555                    let (payload_pos, r) = (flat / residue_size, flat % residue_size);
556                    let src_elem = src_off[payload_pos * axis_size + indices[r]];
557                    chunk.set(dst_elem, self.get(src_elem));
558                }
559            });
560    }

572    /// relabel drops, so later offset-addressed ops see `dst_map`'s layout. Backs
573    /// [`crate::backend::Backend::transmute`]. The vector-engine relayout is
574    /// `…tile(k).read().transmute()`: the read leaves the buffer padded in `src_map`, the transmute

577    where
578        D: MaterializableScalar,
579    {
580        // Same layout drops no padding.
581        if src_map == dst_map {

583        }

647/// The per-output-cell read plan for `reduce` / `contraction` over the physical buffer: how to find,
648/// for each output cell, the operand buffer positions whose values fold into it.
649///
650/// Built from one sequencer config over the `out`-outer / `residue`-inner stream (which fully covers the
651/// operand: `out` carries the kept axes, `residue` the reduced ones; a broadcast axis the operand lacks
652/// contributes nothing). The operand wire offset is additive in the two coordinates
653/// (`offset(o, r) = base(o) + delta(r)`), the same model the serial per-cell fold uses. The shared
654/// `delta(r)` (one residue row, small) is precomputed; each cell's `base(o)` is seeked lazily by
655/// [`Self::reads`], so there is no `out_size` base table and the fold is a single parallel pass (the
656/// seek that would fill such a table happens inside it instead). [`Self::reads`] `.get()`-guards an
657/// out-of-range sum (a split-then-padded coordinate).
658struct ReadPlan {
659    /// The operand wire offsets over the `out`-outer / `residue`-inner stream; `base(o)` is seeked from it.
660    config: SequencerConfig,
661    /// Live residue deltas added to a cell's base, in `residue` walk order (padding positions dropped).
662    delta: Vec<usize>,
663    /// The padded residue width: the stream stride between consecutive output cells.
664    rwidth: usize,
665    /// The first live residue column within a row (the base read's offset into the row).
666    r0: usize,
667    /// Total stream length; an output cell whose base position is `>= stream_size` is dead (padding).
668    stream_size: usize,
669}
670
671impl ReadPlan {
672    /// `memory` is the operand mapping, `out` the output mapping (its cell index `o` is the stream row),
673    /// `residue` the operand's reduced / contracted axes (the `carve` leftover, which marks the kept /
674    /// broadcast axes as `Top` pads).
675    fn new(memory: &MappingValue, out: &MappingValue, residue: &MappingValue) -> Self {
676        // The residue carries `Top` pads (the kept / broadcast axes `carve` marked, and any padded
677        // operand axis): a padded position re-reads its live sibling (`memory_stride` 0), so folding it
678        // would double-count. Walk the residue once to mask the live positions; only those become deltas.
679        let live_r: Vec<usize> = residue
680            .iter_positions()
681            .enumerate()
682            .filter_map(|(r, off)| off.map(|_| r))
683            .collect();
684        let rwidth = residue.size(); // the padded residue width: one full stream row below
685        // A reduce / contraction always folds at least one cell (an empty residue is degenerate), so the
686        // residue has a first live position; index it after this guard rather than panicking opaquely.
687        let &r0 = live_r
688            .first()
689            .expect("reduce/contraction: residue must have a live cell to fold");
690        // Operand wire offsets over the `out` x `residue` stream (`out` outer, residue inner). `out`
691        // carries the kept axes (a broadcast axis the operand lacks reads stride 0, so its fan shares one
692        // base) and `residue` the reduced axes, so together they consume the operand. `out` sequenced as
693        // the outer stream means stream row `o` IS output-buffer offset `o`: `offset(o, r)` sits at stream
694        // position `o * rwidth + r`.
695        let config = sequence(&[memory], &[&out.clone().pair(residue.clone())], SequencerMode::Read)
696            .expect("reduce/contraction: out x residue must factor the operand")
697            .swap_remove(0);
698        let stream_size = config.stream_size();
699        // Deltas: the live residue cells of the first reachable output cell's row, each minus that row's
700        // first live read. The first reachable cell is `o == 0` when `r0` is in range (the base position
701        // `o * rwidth + r0` grows with `o`); otherwise nothing is reachable and the deltas are unused.
702        let delta: Vec<usize> = if r0 < stream_size {
703            let row: Vec<usize> = config.iter_range(0, rwidth.min(stream_size)).collect();
704            live_r.iter().map(|&r| row[r] - row[r0]).collect()
705        } else {
706            Vec::new()
707        };
708        Self {
709            config,
710            delta,
711            rwidth,
712            r0,
713            stream_size,
714        }
715    }
716
717    /// The operand buffer positions folding into output cell `o`: its base (seeked from the stream) plus
718    /// each live residue delta. `None` for a dead (padding) output cell the stream never reaches. A
719    /// split-then-padded residue can still push an individual read past the buffer, so the caller
720    /// `.get()`-guards each position.
721    fn reads(&self, o: usize) -> Option<impl Iterator<Item = usize> + '_> {
722        let pos = o * self.rwidth + self.r0;
723        let base = (pos < self.stream_size).then(|| self.config.iter_range(pos, pos + 1).next())??;
724        Some(self.delta.iter().map(move |&d| base + d))
725    }
726
727    /// Work per output cell (the live residue width), for the rayon `with_min_len` job-size hint.
728    fn cell_work(&self) -> usize {
729        self.delta.len()
730    }
731}

733/// The physical wire base where the `offset` window starts in `base_map` (0 for an empty offset): the
734/// wire position whose canonical offset equals the offset's. `base_map` is the storage's live-axis
735/// mapping, so a tiled offset axis the relayout's `Src` / `Dst` *type* carries as padding is still a
736/// live `Symbol` here and matches; the view shares the base's wire layout, so this position is also the
737/// partial view's start in the buffer.
738fn window_base(base_map: &MappingValue, offset: &Index) -> usize {
739    if *offset == Index::new() {
740        return 0;
741    }
742    // The offset's dense (canonical) offset under `base_map`: each axis's coordinate times its
743    // canonical weight (innermost axis = 1). One `finalize` (inside `axis_coords`).
744    let axes = base_map.axes();
745    let coords = crate::storage::axis_coords(&axes, offset.clone())
746        .expect("transpose: partial-view offset must be a live coordinate of its base mapping");
747    let mut weight = 1;
748    let mut target = 0;
749    for (axis, &coord) in axes.iter().rev().zip(coords.iter().rev()) {
750        target += coord * weight;
751        weight *= axis.modulo;
752    }
753    // The wire position whose canonical offset is `target`, found by the lazy wire walk (`Some` = live,
754    // padding never matches), replacing a `finalize` of every position with the cheap odometer.
755    base_map
756        .iter(&axes, &Index::new(), true)
757        .position(|offset| offset == Some(target))
758        .expect("transpose: partial-view offset must land on a live cell of its base mapping")
759}

761/// Lays `live` elements into a fresh `dst_map`-sized buffer at its live wire positions (padding stays
762/// zero).
763fn place_live_elems<D: Scalar>(dst_map: &MappingValue, mut live: impl Iterator<Item = D>) -> Vec<D> {
764    let mut data = vec![D::zero(); dst_map.size()];
765    for (elem, offset) in data.iter_mut().zip(dst_map.iter_positions()) {
766        if offset.is_some() {
767            *elem = live
768                .next()
769                .expect("transmute: dst_map has more live elements than src_map");
770        }
771    }
772    data
773}

775/// Decodes a `BufStorage`-backed `i32` index tensor to element positions: each value divided by the
776/// stride from [`decode_stride`] (a byte stride when scaled, else 1). `BufStorage` offsets are
777/// physical and `data` already holds exactly the index mapping's cells, so the whole buffer is read.
778/// Shared by scatter and gather.
779fn decode_indices(index: &BufStorage<i32>, index_stride: usize) -> Vec<usize> {
780    // The index buffer holds exactly the index mapping's elements; read them in order (an `i32` decode is
781    // a whole-value read, so this is the cheap byte-multiple path).
782    (0..index.len())
783        .map(|i| {
784            let raw = index.get(i);
785            let pos = usize::try_from(raw)
786                .unwrap_or_else(|_| panic!("scatter/gather index at cell {i} must be non-negative, got {raw}"));
787            pos / index_stride
788        })
789        .collect()
790}
791
792/// The divisor [`decode_indices`] applies: a scaled index tensor holds byte offsets into
793/// payload-sized blocks (divide by the block's byte size); an unscaled one holds element indices.
794fn decode_stride<D>(payload: &MappingValue, scaled: bool) -> usize {
795    if scaled {
796        payload.size() * std::mem::size_of::<D>()
797    } else {
798        1
799    }
800}

803fn lcm(a: usize, b: usize) -> usize {
804    a / gcd(a, b) * b
805}

807fn gcd(mut a: usize, mut b: usize) -> usize {
808    while b != 0 {
809        (a, b) = (b, a % b);
810    }
811    a

1077    /// The `Output -> D` narrow happens once per within-slice `contraction` (the Lane Folder narrow), so
1078    /// a cross-slice reduction over the narrowed per-slice partials differs from one global wide fold.
1079    /// The hardware narrows each per-slice partial at its own Lane Folder and the Vector Engine sums the
1080    /// narrowed partials; the sim composes that as three per-slice `BufStorage::contraction`s feeding a
1081    /// downstream `BufStorage::reduce`.
1082    ///
1083    /// Each per-slice dot is `[1.0, 2^-8] . [1.0, 1.0] = 1.0 + 2^-8`, which the per-slice narrow rounds
1084    /// to `1.0` (`2^-8` is half a bf16 ulp at magnitude 1, ties-to-even). The cross-slice `reduce` over
1085    /// the three `1.0`s gives `3.0`. A global wide fold is the same six MACs as one `K=6` contraction: it
1086    /// keeps `3.0 + 3·2^-8` in `f32` and narrows once to `bf16(3.015625)`. The two outputs differ, so the
1087    /// narrow is per slice, not held wide across the cross-slice reduce.
1088    #[test]
1089    fn cross_slice_reduce_narrows_per_lane_fold_not_globally() {
1090        axes![Slice = 3, M = 1, K = 2, N = 1, GlobalK = 6];
1091        // Per-slice path: each slice is a within-slice contraction that narrows to bf16 at its lane fold.
1092        let lhs_slice = || BufStorage::from_vec(vec![bf16::from_f32(1.0), bf16::from_f32(2.0f32.powi(-8))]);
1093        let rhs_slice = || BufStorage::from_vec(vec![bf16::from_f32(1.0); 2]);
1094        let partial = BufStorage::contraction(
1095            &lhs_slice(),
1096            &rhs_slice(),
1097            &<m![M, K]>::to_value(),
1098            &<m![K, N]>::to_value(),
1099            &<m![M, K, N]>::to_value(),
1100            &<m![M, N]>::to_value(),
1101        );
1102        let partial0 = partial.get(0);
1103        assert_eq!(
1104            partial0.to_f32(),
1105            1.0,
1106            "per-slice narrow rounds 1 + 2^-8 to 1.0 in bf16"
1107        );
1108        // The Vector Engine sums the three narrowed per-slice partials downstream (a `reduce`).
1109        let three_narrowed = BufStorage::from_vec(vec![partial0; 3]);
1110        let cross_slice = three_narrowed
1111            .reduce::<m![Slice, M, N], m![M, N], _>(|a, b| a + b, bf16::from_f32(0.0), false)
1112            .get(0)
1113            .to_f32();
1114        assert_eq!(cross_slice, 3.0);
1115        // One global wide fold: the same six MACs as a single K=6 contraction, narrowing once at the end.
1116        let global = BufStorage::contraction(
1117            &BufStorage::from_vec([1.0, 2.0f32.powi(-8)].into_iter().cycle().take(6).map(bf16::from_f32)),
1118            &BufStorage::from_vec(vec![bf16::from_f32(1.0); 6]),
1119            &<m![M, GlobalK]>::to_value(),
1120            &<m![GlobalK, N]>::to_value(),
1121            &<m![M, GlobalK, N]>::to_value(),
1122            &<m![M, N]>::to_value(),
1123        );
1124        let global0 = global.get(0).to_f32();
1125        assert_eq!(global0, 3.015625, "global wide fold keeps 3 + 3·2^-8, narrows once");
1126        assert_ne!(
1127            cross_slice, global0,
1128            "per-slice Lane Folder narrow must differ from a global narrow across the cross-slice reduce"
1129        );
1130    }
