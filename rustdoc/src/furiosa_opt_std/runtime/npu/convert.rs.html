8fn to_buffer<D: Scalar, Chip: M, Element: M>(addr: u64) -> Buffer {
9    let len = Pair::<Chip, Element>::SIZE * std::mem::size_of::<D>();
10    Buffer::from_raw(unsafe { lib().furiosa_npu_buffer_from(rt(), addr, len) })
11}

37/// `Extend`-shaped trait for `Vec<Buffer>`, narrowed to this crate's needs.
38///
39/// Has the identical signature to `core::iter::Extend<A>` — same `&mut self`
40/// receiver, same `IntoIterator<Item = A>` parameter, same `extend` method
41/// name.
42pub trait ExtendBuffers<A> {
43    /// Extends `self` with one DMA buffer per leaf tensor produced by `iter`.
44    fn extend<I: IntoIterator<Item = A>>(&mut self, iter: I);
45}
46
47impl ExtendBuffers<()> for Vec<Buffer> {
48    fn extend<I: IntoIterator<Item = ()>>(&mut self, _iter: I) {}
49}

51impl<D: Scalar, Chip: M, E: M> ExtendBuffers<HbmTensor<D, Chip, E>> for Vec<Buffer> {
52    fn extend<I: IntoIterator<Item = HbmTensor<D, Chip, E>>>(&mut self, iter: I) {
53        for t in iter {
54            self.push((&t).into());
55        }
56    }
57}

117macro_rules! impl_extend_buffers_tuple {
118    () => {};
119    (($T0:ident, $t0:ident) $(, ($T:ident, $t:ident))*) => {
120        impl<$T0 $(, $T)*> ExtendBuffers<($T0, $($T,)*)> for Vec<Buffer>
121        where
122            Self: ExtendBuffers<$T0> $(+ ExtendBuffers<$T>)*
123        {
124            fn extend<__I: IntoIterator<Item = ($T0, $($T,)*)>>(&mut self, iter: __I) {
125                for ($t0, $($t,)*) in iter {
126                    ExtendBuffers::extend(self, ::std::iter::once($t0));
127                    $( ExtendBuffers::extend(self, ::std::iter::once($t)); )*
128                }
129            }
130        }
131        impl_extend_buffers_tuple!($(($T, $t)),*);
132    };
133}
