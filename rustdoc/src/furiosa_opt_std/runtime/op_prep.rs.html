1//! Per-op preparation step shared between Simulation and Typecheck.
2//!
3//! Each helper here is the "prep" that runs before an op's real work — it derives the mapping
4//! data the op needs (broadcast residue, reshape quotient, scatter payload, …) and panics on
5//! ill-formed mappings along the way. Called from exactly two places per op: the Simulation
6//! impl on `MathRawTensor` consumes the returned mapping to drive its iteration loop, and the
7//! Typecheck impl on `PhantomRawTensor` discards the value (`let _ = …`) but still triggers
8//! the panic side, so type errors surface without the value-level loop ever running. Npu /
9//! Emulation never call them — their `BufRawTensor` op bodies are `todo!()`.
10//!
11//! Most helpers are noun-style and return a derived `Mapping`; `assert_zip` is the one pure
12//! check (returns `()`). The shared file lets Simulation and Typecheck stay in lockstep on
13//! mapping algebra without duplicating the prep step.
14//!
15//! [`Backend`]: crate::runtime::backend::Backend
16//! [`RawTensor`]: crate::tensor::raw::RawTensor
17

22/// The broadcast portion of `dst` for `RawTensor::write_transpose`, the part `src` does not cover;
23/// `MathRawTensor`'s write loop iterates over it. Panics on ill-formed mappings.
24pub(crate) fn transpose_broadcast<Src: M, Dst: M>(allow_broadcast: bool) -> Mapping {
25    let src_mapping = Src::to_value();
26    let dst_mapping = Dst::to_value();
27    // Carve `src` out of `dst` via the matcher; the leftover is the broadcast portion.
28    let broadcast = dst_mapping.carve(&src_mapping);
29    if !allow_broadcast {
30        assert!(broadcast.is_padding());
31    }
32    broadcast
33}

35/// Axes-equality check for `RawTensor::zip_with`.
36pub(crate) fn assert_zip(lhs_axes: &[Term], rhs_axes: &[Term]) {
37    assert_eq!(
38        lhs_axes, rhs_axes,
39        "Tensors must have the same axes for element-wise binary operations"
40    );
41}

43/// Broadcast residue for `RawTensor::reduce_then_broadcast`. Panics on ill-formed mappings.
44pub(crate) fn reduce_broadcast(src_axes: &[Term], dst_axes: &[Term]) -> Mapping {
45    Mapping::from_terms(dst_axes.iter().cloned()).carve(&Mapping::from_terms(src_axes.iter().cloned()))
46}

48/// The broadcast portion of `dst` for `reduce_then_broadcast`: the axes built from symbols absent
49/// in `src` (a symbol-level split, since broadcast axes are always new).
50pub(crate) fn broadcast_axes(src: &Mapping, dst: &Mapping) -> Mapping {
51    let src_idents: HashSet<Ident> = src.idents().into_iter().collect();
52    let dst_axes = dst.axes();
53    let mut ids = Vec::new();
54    // The symbol-level filter is exact only when each dst axis is WHOLLY new or WHOLLY shared.
55    // A dst axis straddling both (a shared symbol partially split into a broadcast factor) would
56    // split silently; pin that this does not happen.
57    debug_assert!(
58        dst_axes.iter().all(|term| {
59            collect_term_idents(term, &mut ids);
60            let in_src = ids.iter().filter(|i| src_idents.contains(i)).count();
61            in_src == 0 || in_src == ids.len()
62        }),
63        "broadcast_axes assumes each dst axis is wholly new or wholly shared, not a straddling split"
64    );
65    let axes: Vec<Term> = dst_axes
66        .into_iter()
67        .filter(|term| {
68            collect_term_idents(term, &mut ids);
69            ids.iter().all(|ident| !src_idents.contains(ident))
70        })
71        .collect();
72    Mapping::from_terms(axes)
73}

75/// Collects `term`'s idents into `out` (cleared first), so the caller reuses one buffer.
76fn collect_term_idents(term: &Term, out: &mut Vec<Ident>) {
77    out.clear();
78    match &term.inner {
79        Atom::Symbol { symbol, .. } => out.push(*symbol),
80        Atom::Composite(inner) => out.extend(inner.idents()),
81    }
82}

84/// Scatter payload mapping and destination axis term. Used by every backend whose
85/// `RawTensor::write_scatter` body iterates the host. Panics on ill-formed scatters.
86pub(crate) fn scatter_params(src: &Mapping, dst: &Mapping, key: &Mapping) -> (Mapping, Term) {
87    let payload = src.carve(key);
88    let dst_term = dst
89        .carve(&payload)
90        .axes()
91        .into_iter()
92        .next()
93        .expect("scatter dst residue has no live target axis");
94    (payload, dst_term)
95}

97/// Derived parameters for `RawTensor::write_gather`.
98///
99/// Inverse of [`scatter_params`]:
100/// - scatter: `src ÷ key = payload`, `dst ÷ payload = dst_term` (the indexed axis on dst).
101/// - gather:  `dst ÷ idx_axes = payload`, `src ÷ payload = src_term` (the indexed axis on src/table).
102pub(crate) struct GatherParams {
103    /// Axes shared by `src` (the table) and `dst` (the gather output), iterated identically
104    /// on both sides. Derived as `dst ÷ idx_axes`.
105    pub payload: Mapping,
106    /// Axes in `dst` that replace the indexed axis on the source side. The `write_gather` loop
107    /// iterates these positions and writes each output slot sequentially. Derived as
108    /// `dst ÷ payload`.
109    pub idx_residue: Mapping,
110    /// Single-term locator on the `src` side identifying the indexed (lookup) axis. The runtime
111    /// indices tensor's values index into this axis. Derived as the surviving term in
112    /// `src ÷ payload`.
113    pub src_term: Term,
114}
115
116/// Compute [`GatherParams`] for a gather op. Used by every backend whose
117/// `RawTensor::write_gather` body iterates the host. Panics on ill-formed gathers.
118///
119/// `idx` is the full mapping of the indices tensor; the "key-axes-replacement" inside `dst` is
120/// derived as `dst ÷ payload`.
121pub(crate) fn gather_params(src: &Mapping, dst: &Mapping, idx: &Mapping) -> GatherParams {
122    let payload = dst.carve(idx);
123    let idx_residue = dst.carve(&payload);
124    let src_term = src
125        .carve(&payload)
126        .axes()
127        .into_iter()
128        .next()
129        .expect("gather src residue has no live target axis");
130    GatherParams {
131        payload,
132        idx_residue,
133        src_term,
134    }
135}

137#[cfg(test)]
138mod tests {
139    use furiosa_mapping::*;
140
141    use super::broadcast_axes;
142
143    axes![A = 4, B = 2, C = 8];
144
145    /// `broadcast_axes` is the symbol-level narrowing of `dst.carve(src)`: it keeps the SAME live axes
146    /// but drops the positional padding carve adds. Where its invariant holds (each dst axis wholly new
147    /// or wholly shared), its axes must equal carve's live axes.
148    #[test]
149    fn broadcast_axes_matches_carve() {
150        let cases: [(Mapping, Mapping); 4] = [
151            (<m![A, C]>::to_value(), <m![A, B, C]>::to_value()), // B is the new (broadcast) axis
152            (<m![A]>::to_value(), <m![A, B]>::to_value()),
153            (<m![B]>::to_value(), <m![A, B, C]>::to_value()), // two new axes (A, C)
154            (<m![A, B, C]>::to_value(), <m![A, B, C]>::to_value()), // no broadcast
155        ];
156        let sorted_axes = |m: &Mapping| {
157            let mut a = m.axes();
158            a.sort();
159            a
160        };
161        for (src, dst) in cases {
162            assert_eq!(
163                sorted_axes(&broadcast_axes(&src, &dst)),
164                sorted_axes(&dst.carve(&src)),
165                "broadcast_axes(src={src:?}, dst={dst:?})"
166            );
167        }
168    }
